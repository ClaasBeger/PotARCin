def solve(grid):

    out = [row[:] for row in grid]
    
    # Find 8

                # (2,4)=5, (1,4)=5,(1,5)=5,(1,6)=5... wait (0,6)=5
                # Pattern: L-shapes. Each L goes 1 vertical + 2 horizontal, stepping by 2
                
                # Up direction: for k=0,1,2,...
                # vertical at (r-(2k+1), c+2k), then horizontal row r-(2k+2) from c+2k to c+2(k+1)
                # and top of that: (r-(2k+3), c+2(k+1))...

                # (2,4)=5, (1,4)=5,(1,5)=5,(1,6)=5, (0,6)=5
                # That's: go up 1, then up 1+right 2, then up 1
                
                # Simpler: zigzag. Each segment is 3 long forming an L
                # Segment 1 up: (2,4),(1,4),(1,5),(1,6),(0,6)
                # That's length-2 up, length-2 right, length-1 up

                # Each L: 2 up, 2 right repeated... no.

                # The pattern from 8 upward and downward are symmetric but rotated.
                
                # Up from 8: step up 1, turn right step 2, turn up step 1, forming staircase
                # Down from 8: step down 1, turn left step 2, turn down step 1...
                
                pos_r, pos_c = r, c
                # Going up-right
                for step in range(20):
