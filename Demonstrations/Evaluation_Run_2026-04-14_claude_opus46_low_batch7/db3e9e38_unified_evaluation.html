def solve(grid):
    rows, cols = len(grid), len(grid[0])
    # Find the column of 7s and height
    col7 = None
    height = 0
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 7:
                col7 = c
                height += 1
                break
        if col7 is not None and grid[r][col7] != 7:
            break
    
    out = [row[:] for row in grid]
    
    # From bottom row of 7s upward, each row i from bottom gets expanding pattern
    for i in range(height):
        r = height - 1 - i  # row index (0=top)
        # distance i: spread i steps from col7
        for d in range(i + 1):
            # alternating 7 and 8: d=0 is 7 (original), d=1 is 8, d=2 is 7...
            val = 7 if d % 2 == 0 else 8
            for sign in [1, -1]:
                c = col7 + sign * d
                if 0 <= c < cols:
                    out[r][c] = val
    
    return out
