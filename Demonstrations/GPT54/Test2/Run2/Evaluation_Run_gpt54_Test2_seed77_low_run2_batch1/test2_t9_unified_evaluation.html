2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import deque, defaultdict

    H, W = len(grid), len(grid[0])

    # 8-connected components by color
    seen = [[False] * W for _ in range(H)]
    comps = []

    for r in range(H):
        for c in range(W):
            color = grid[r][c]
            if color == 0 or seen[r][c]:
                continue
            q = deque([(r, c)])
            seen[r][c] = True
            cells = []
            while q:
                x, y = q.popleft()
                cells.append((x, y))
                for dx in (-1, 0, 1):
                    for dy in (-1, 0, 1):
                        if dx == 0 and dy == 0:
                            continue
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < H and 0 <= ny < W and not seen[nx][ny] and grid[nx][ny] == color:
                            seen[nx][ny] = True
                            q.append((nx, ny))

            minr = min(r for r, _ in cells)
            maxr = max(r for r, _ in cells)
            minc = min(c for _, c in cells)
            maxc = max(c for _, c in cells)
            rel = {(r - minr, c - minc) for r, c in cells}
            h = maxr - minr + 1
            w = maxc - minc + 1

            comps.append({
                "color": color,
                "cells": cells,
                "rel": rel,
                "minr": minr,
                "maxr": maxr,
                "minc": minc,
                "maxc": maxc,
                "h": h,
                "w": w,
                "size": len(cells),
            })

    def classify(comp):
        rel = comp["rel"]
        h, w = comp["h"], comp["w"]
        size = comp["size"]

        if size == 1:
            return "singleton"

        # straight lines
        if h == 1:
            return "hline"
        if w == 1:
            return "vline"

        # diagonals
        if all(r == c for r, c in rel):
            return "diag"
        if all(r + c == w - 1 for r, c in rel):
            return "antidiag"

        # corner shapes
        top = {(0, c) for c in range(w)}
        bottom = {(h - 1, c) for c in range(w)}
        left = {(r, 0) for r in range(h)}
        right = {(r, w - 1) for r in range(h)}

        if rel == top | left:
            return "corner_tl"
        if rel == top | right:
            return "corner_tr"
        if rel == bottom | left:
            return "corner_bl"
        if rel == bottom | right:
            return "corner_br"

        # fallback by occupied extremes
        has_top = any(r == 0 for r, c in rel)
        has_bottom = any(r == h - 1 for r, c in rel)
        has_left = any(c == 0 for r, c in rel)
        has_right = any(c == w - 1 for r, c in rel)
        if has_top and has_left:
            return "corner_tl"
        if has_top and has_right:
            return "corner_tr"
        if has_bottom and has_left:
            return "corner_bl"
        if has_bottom and has_right:
            return "corner_br"
        return "singleton"

    for comp in comps:
        comp["kind"] = classify(comp)

    # Output size: largest intrinsic extent
    N = max(max(comp["h"], comp["w"]) for comp in comps) if comps else 0
    out = [[0] * N for _ in range(N)]

    def draw_cells(color, rel_cells, r0, c0):
        for r, c in rel_cells:
            rr, cc = r0 + r, c0 + c
            if 0 <= rr < N and 0 <= cc < N:
                out[rr][cc] = color

    # Place corner shapes
    for kind, r0_func, c0_func in [
        ("corner_tl", lambda c: 0, lambda c: 0),
        ("corner_tr", lambda c: 0, lambda c: N - c["w"]),
        ("corner_bl", lambda c: N - c["h"], lambda c: 0),
        ("corner_br", lambda c: N - c["h"], lambda c: N - c["w"]),
    ]:
        for comp in sorted([x for x in comps if x["kind"] == kind], key=lambda z: (z["minr"], z["minc"])):
            draw_cells(comp["color"], comp["rel"], r0_func(comp), c0_func(comp))

    # Place vertical lines by left/right side in original grid
    vlines = [c for c in comps if c["kind"] == "vline"]
    for comp in sorted(vlines, key=lambda z: (z["minc"], z["minr"])):
        right_side = comp["minc"] >= W / 2
        r0 = 0 if comp["minr"] <= H / 2 else N - comp["h"]
        c0 = N - 1 if right_side else 0
        draw_cells(comp["color"], {(r, 0) for r, _ in comp["rel"]}, r0, c0)

    # Place horizontal lines by top/bottom and left/right in original grid
    hlines = [c for c in comps if c["kind"] == "hline"]
    for comp in sorted(hlines, key=lambda z: (z["minr"], z["minc"])):
        bottom_side = comp["minr"] >= H / 2
        right_side = comp["minc"] >= W / 2
        r0 = N - 1 if bottom_side else 0
        c0 = N - comp["w"] if right_side else 0
        draw_cells(comp["color"], {(0, c) for _, c in comp["rel"]}, r0, c0)

    # Place diagonals
    diags = sorted([c for c in comps if c["kind"] == "diag"], key=lambda z: (z["minr"], z["minc"]))
    for i, comp in enumerate(diags):
        # right-leaning nested placement
        start_r = min(i, N - comp["h"])
        start_c = max(0, N - comp["w"] - i)
        draw_cells(comp["color"], comp["rel"], start_r, start_c)

    antidiags = sorted([c for c in comps if c["kind"] == "antidiag"], key=lambda z: (z["minr"], z["minc"]))
    for i, comp in enumerate(antidiags):
        start_r = min(i, N - comp["h"])
        start_c = N - comp["w"]
        draw_cells(comp["color"], comp["rel"], start_r, start_c)

    # Place singletons in remaining empty cells, preferring center, then scan
    singletons = sorted([c for c in comps if c["kind"] == "singleton"], key=lambda z: (z["minr"], z["minc"]))
    prefs = []
    if N:
        centers = [(N // 2, N // 2)]
        if N % 2 == 0:
            centers += [(N // 2 - 1, N // 2), (N // 2, N // 2 - 1), (N // 2 - 1, N // 2 - 1)]
        prefs.extend(centers)
        for r in range(N):
            for c in range(N):
                prefs.append((r, c))

    used = set()
    for comp in singletons:
        placed = False
        for r, c in prefs:
            if (r, c) in used:
                continue
            if out[r][c] == 0:
                out[r][c] = comp["color"]
                used.add((r, c))
                placed = True
                break
        if not placed and N:
            out[0][0] = comp["color"]

    return out
