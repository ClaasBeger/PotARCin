2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0]) if h else 0

    g = [row[:] for row in grid]

    special_cells = []
    for r in range(h):
        for c in range(w):
            if g[r][c] not in (0, 5, 7):
                special_cells.append((r, c, g[r][c]))

    if not special_cells:
        return g

    sr, sc, color = special_cells[0]

    # Find connected components of all non-zero cells using 8-neighborhood.
    visited = [[False] * w for _ in range(h)]
    comps = []

    for r in range(h):
        for c in range(w):
            if g[r][c] != 0 and not visited[r][c]:
                stack = [(r, c)]
                visited[r][c] = True
                cells = []
                while stack:
                    x, y = stack.pop()
                    cells.append((x, y))
                    for dx in (-1, 0, 1):
                        for dy in (-1, 0, 1):
                            if dx == 0 and dy == 0:
                                continue
                            nx, ny = x + dx, y + dy
                            if 0 <= nx < h and 0 <= ny < w and g[nx][ny] != 0 and not visited[nx][ny]:
                                visited[nx][ny] = True
                                stack.append((nx, ny))
                comps.append(cells)

    def bbox(cells):
        rs = [r for r, _ in cells]
        cs = [c for _, c in cells]
        return min(rs), min(cs), max(rs), max(cs)

    source_comp = None
    source_box = None
    for comp in comps:
        if any(r == sr and c == sc for r, c in comp):
            source_comp = comp
            source_box = bbox(comp)
            break

    if source_comp is None:
        return g

    sh0, sw0, sh1, sw1 = source_box
    source_size = (sh1 - sh0, sw1 - sw0)

    # Prefer another component with same bounding-box size.
    candidates = []
    for comp in comps:
        if comp is source_comp:
            continue
        b = bbox(comp)
        size = (b[2] - b[0], b[3] - b[1])
        if size == source_size:
            candidates.append((b, comp))

    # If no same-sized component exists, choose the largest other component.
    if candidates:
        target_box, _ = candidates[0]
    else:
        others = [(bbox(comp), comp) for comp in comps if comp is not source_comp]
        if not others:
            # only restore the source special cell(s)
            for r in range(h):
                for c in range(w):
                    if g[r][c] == color:
                        g[r][c] = 5
            return g
        target_box, _ = max(others, key=lambda x: (x[0][2] - x[0][0] + 1) * (x[0][3] - x[0][1] + 1))

    # Restore special-colored cells in the source object to 5.
    for r in range(h):
        for c in range(w):
            if g[r][c] == color:
                g[r][c] = 5

    # Draw a full rectangle outline in the special color on the target bbox.
    r0, c0, r1, c1 = target_box
    for c in range(c0, c1 + 1):
        g[r0][c] = color
        g[r1][c] = color
    for r in range(r0, r1 + 1):
        g[r][c0] = color
        g[r][c1] = color

    return g
