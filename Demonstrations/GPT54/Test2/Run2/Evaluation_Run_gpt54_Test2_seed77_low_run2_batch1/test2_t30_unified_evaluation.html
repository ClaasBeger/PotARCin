2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0])

    # Find separator rows (all 6s)
    sep_rows = [i for i, row in enumerate(grid) if all(v == 6 for v in row)]
    if not sep_rows:
        return [row[:] for row in grid]

    # Split into blocks
    blocks = []
    starts = [0] + [r + 1 for r in sep_rows]
    ends = sep_rows + [h]
    for s, e in zip(starts, ends):
        blocks.append([row[:] for row in grid[s:e]])

    bh = len(blocks[0])
    color = None
    for b in blocks:
        for row in b:
            for v in row:
                if v not in (0, 6):
                    color = v
                    break
            if color is not None:
                break
        if color is not None:
            break
    if color is None:
        return [row[:] for row in grid]

    def coords(block):
        return {(r, c) for r in range(bh) for c in range(w) if block[r][c] == color}

    sets = [coords(b) for b in blocks]
    nonzero_idx = [i for i, s in enumerate(sets) if s]

    # Heuristic 1: nested growth (good for Example 3 / test)
    nested = True
    for a, b in zip(nonzero_idx, nonzero_idx[1:]):
        if not (sets[a] <= sets[b]):
            nested = False
            break

    if nested and nonzero_idx:
        out_sets = [set(s) for s in sets]

        # Fill leading empty blocks by shrinking the first non-empty block.
        first = nonzero_idx[0]
        if first > 0:
            s = set(out_sets[first])

            # If shape is made of horizontal symmetric pairs growing downward,
            # reconstruct earlier stages by keeping only top rows.
            rows = {}
            for r, c in s:
                rows.setdefault(r, []).append(c)
            row_keys = sorted(rows)

            symmetric_by_rows = True
            center2 = None
            spans = []
            for r in row_keys:
                cs = sorted(rows[r])
                if center2 is None:
                    center2 = cs[0] * 2 if len(cs) == 1 else cs[0] + cs[-1]
                if cs[0] + cs[-1] != center2:
                    symmetric_by_rows = False
                    break
                spans.append((r, len(cs)))

            if symmetric_by_rows and len(row_keys) >= 2:
                # Build previous stages by removing bottom rows, with the earliest
                # stage reduced to only the top-center cell when needed.
                prev_blocks = []
                current_rows = row_keys[:]
                for _ in range(first):
                    if len(current_rows) > 1:
                        current_rows = current_rows[:-1]
                    else:
                        current_rows = current_rows[:]
                    prev = set()
                    for rr in current_rows:
                        cs = sorted(rows[rr])
                        if len(current_rows) == 1 and len(cs) > 1:
                            mid = (cs[0] + cs[-1]) // 2
                            prev.add((rr, mid))
                        else:
                            for c in cs:
                                prev.add((rr, c))
                    prev_blocks.append(prev)
                for i in range(first):
                    out_sets[first - 1 - i] = prev_blocks[i]
            else:
                # Generic fallback: repeatedly remove "latest" cells by row-major reverse
                ordered = sorted(s, key=lambda x: (x[0], x[1]))
                step = max(1, len(ordered) // (first + 1))
                cur = ordered[:]
                for i in range(first - 1, -1, -1):
                    cur = cur[:-step] if len(cur) > step else cur[:1]
                    out_sets[i] = set(cur)

        # Fill internal/trailing empty blocks by simple interpolation/extrapolation
        for i in range(len(blocks)):
            if out_sets[i]:
                continue
            # nearest previous non-empty
            p = i - 1
            while p >= 0 and not out_sets[p]:
                p -= 1
            n = i + 1
            while n < len(blocks) and not out_sets[n]:
                n += 1

            if p >= 0 and n < len(blocks) and out_sets[p] <= out_sets[n]:
                add = sorted(out_sets[n] - out_sets[p], key=lambda x: (x[0], x[1]))
                gap = n - p
                take = len(add) * (i - p) // gap
                out_sets[i] = set(out_sets[p]) | set(add[:take])
            elif p >= 0:
                out_sets[i] = set(out_sets[p])
            elif n < len(blocks):
                out_sets[i] = set(out_sets[n])

        # Rebuild grid
        out = []
        for bi, s in enumerate(out_sets):
            block = [[0] * w for _ in range(bh)]
            for r, c in s:
                block[r][c] = color
            out.extend(block)
            if bi != len(out_sets) - 1:
                out.append([6] * w)
        return out

    # Heuristic 2: equal-size moving points on perimeter (Example 2-like)
    sizes = [len(s) for s in sets if s]
    if sizes and all(sz == sizes[0] for sz in sizes):
        out_sets = [set(s) for s in sets]

        def perimeter_positions(hh, ww):
            pos = []
            for c in range(ww):
                pos.append((0, c))
            for r in range(1, hh):
                pos.append((r, ww - 1))
            for c in range(ww - 2, -1, -1):
                pos.append((hh - 1, c))
            for r in range(hh - 2, 0, -1):
                pos.append((r, 0))
            return pos

        per = perimeter_positions(bh, w)
        pidx = {p: i for i, p in enumerate(per)}

        known = [i for i, s in enumerate(out_sets) if s]
        if len(known) >= 2:
            k0, k1 = known[0], known[1]
            a = sorted(out_sets[k0])
            b = sorted(out_sets[k1])
            if len(a) == len(b) and all(x in pidx for x in a + b):
                shifts = []
                m = len(per)
                for pa, pb in zip(a, b):
                    shifts.append((pidx[pb] - pidx[pa]) % m)
                if len(set(shifts)) == 1:
                    step = shifts[0]
                    for i in range(len(blocks)):
                        if out_sets[i]:
                            continue
                        base_i = max(j for j in known if j < i) if any(j < i for j in known) else known[0]
                        delta = i - base_i
                        newset = set()
                        for p in out_sets[base_i]:
                            newset.add(per[(pidx[p] + step * delta) % m])
                        out_sets[i] = newset

                    out = []
                    for bi, s in enumerate(out_sets):
                        block = [[0] * w for _ in range(bh)]
                        for r, c in s:
                            block[r][c] = color
                        out.extend(block)
                        if bi != len(out_sets) - 1:
                            out.append([6] * w)
                    return out

    # Heuristic 3: diamond/shells around center (Example 1-like)
    out_sets = [set(s) for s in sets]
    cr, cc = bh // 2, w // 2

    def shell(k):
        return {
            (r, c)
            for r in range(bh)
            for c in range(w)
            if abs(r - cr) + abs(c - cc) == k
        }

    radii = {}
    valid = True
    for i, s in enumerate(sets):
        if not s:
            continue
        matched = None
        for k in range(bh + w):
            if s == shell(k):
                matched = k
                break
        if matched is None:
            valid = False
            break
        radii[i] = matched

    if valid and radii:
        known = sorted(radii)
        # fill missing by linear sequence in index->radius if possible
        for i in range(len(blocks)):
            if i in radii:
                continue
            # choose nearest progression based on neighbors
            p = max([k for k in known if k < i], default=None)
            n = min([k for k in known if k > i], default=None)
            if p is not None and n is not None:
                if radii[n] - radii[p] == n - p:
                    k = radii[p] + (i - p)
                else:
                    k = radii[p]
            elif p is not None:
                # continue the progression
                prev_known = [k for k in known if k < p]
                if prev_known:
                    step = radii[p] - radii[prev_known[-1]]
                    k = radii[p] + step * (i - p)
                else:
                    k = radii[p]
            else:
                k = radii[n]
            out_sets[i] = shell(k)

        out = []
        for bi, s in enumerate(out_sets):
            block = [[0] * w for _ in range(bh)]
            for r, c in s:
                block[r][c] = color
            out.extend(block)
            if bi != len(out_sets) - 1:
                out.append([6] * w)
        return out

    return [row[:] for row in grid]
