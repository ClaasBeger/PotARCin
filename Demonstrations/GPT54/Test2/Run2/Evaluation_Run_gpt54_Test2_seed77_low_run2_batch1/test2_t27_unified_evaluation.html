2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import deque
    from functools import lru_cache

    R, C = len(grid), len(grid[0])

    # ---------- find connected components ----------
    seen = [[False] * C for _ in range(R)]
    comps = []

    for r in range(R):
        for c in range(C):
            if grid[r][c] != 0 and not seen[r][c]:
                color = grid[r][c]
                q = deque([(r, c)])
                seen[r][c] = True
                cells = []
                while q:
                    x, y = q.popleft()
                    cells.append((x, y))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < R and 0 <= ny < C and not seen[nx][ny] and grid[nx][ny] == color:
                            seen[nx][ny] = True
                            q.append((nx, ny))
                minr = min(r for r, _ in cells)
                minc = min(c for _, c in cells)
                norm = tuple(sorted((r - minr, c - minc) for r, c in cells))
                h = max(r for r, _ in norm) + 1
                w = max(c for _, c in norm) + 1
                comps.append({
                    "color": color,
                    "cells": cells,
                    "norm": norm,
                    "minr": minr,
                    "minc": minc,
                    "h": h,
                    "w": w,
                    "size": len(cells),
                })

    n = len(comps)
    if n == 0:
        return grid

    # ---------- shape transforms ----------
    def normalize(cells):
        minr = min(r for r, _ in cells)
        minc = min(c for _, c in cells)
        out = tuple(sorted((r - minr, c - minc) for r, c in cells))
        h = max(r for r, _ in out) + 1
        w = max(c for _, c in out) + 1
        return out, h, w

    def transforms(norm):
        cells = list(norm)
        out = {}
        for _ in range(4):
            variants = [
                cells,
                [(r, -c) for r, c in cells],
            ]
            newvars = []
            for v in variants:
                nv, h, w = normalize(v)
                out[nv] = (nv, h, w)
                newvars.append(v)
            cells = [(c, -r) for r, c in cells]
        return list(out.values())

    all_transforms = [transforms(comp["norm"]) for comp in comps]

    # ---------- tiling search for a subset ----------
    @lru_cache(None)
    def subset_indices(mask):
        return [i for i in range(n) if (mask >> i) & 1]

    def rect_dims(area):
        ds = []
        for h in range(1, area + 1):
            if area % h == 0:
                w = area // h
                ds.append((h, w))
        ds.sort(key=lambda x: abs(x[0] - x[1]))
        return ds

    def find_tiling(mask, anchor=None):
        idxs = subset_indices(mask)
        area = sum(comps[i]["size"] for i in idxs)

        for H, W in rect_dims(area):
            board_size = H * W

            allowed = {}
            for i in idxs:
                vals = []
                for norm, h, w in all_transforms[i]:
                    if anchor == i and norm != comps[i]["norm"]:
                        continue
                    if h <= H and w <= W:
                        vals.append((norm, h, w))
                if not vals:
                    break
                allowed[i] = vals
            else:
                full_mask = (1 << board_size) - 1

                placements_by_piece = {}
                for i in idxs:
                    plist = []
                    for norm, h, w in allowed[i]:
                        cells = list(norm)
                        for r0 in range(H - h + 1):
                            for c0 in range(W - w + 1):
                                bits = 0
                                placed = []
                                ok = True
                                for r, c in cells:
                                    rr, cc = r0 + r, c0 + c
                                    if not (0 <= rr < H and 0 <= cc < W):
                                        ok = False
                                        break
                                    p = rr * W + cc
                                    bits |= 1 << p
                                    placed.append((rr, cc))
                                if ok:
                                    plist.append({
                                        "bits": bits,
                                        "cells": tuple(sorted(placed)),
                                        "norm": norm,
                                        "top": r0,
                                        "left": c0,
                                    })
                    if not plist:
                        break
                    placements_by_piece[i] = plist
                else:
                    # If anchor fixed, try all anchor placements first
                    anchor_list = placements_by_piece[anchor] if anchor is not None else [None]

                    for anchor_place in anchor_list:
                        used = 0
                        chosen = {}
                        remaining = set(idxs)
                        if anchor is not None:
                            used = anchor_place["bits"]
                            chosen[anchor] = anchor_place
                            remaining.remove(anchor)

                        def backtrack(used, remaining, chosen):
                            if not remaining:
                                return used == full_mask, chosen

                            # first empty cell
                            first = 0
                            while (used >> first) & 1:
                                first += 1

                            # choose piece with fewest placements covering first
                            best_i = None
                            best_candidates = None
                            for i in remaining:
                                cands = []
                                for p in placements_by_piece[i]:
                                    if (p["bits"] & used) == 0 and ((p["bits"] >> first) & 1):
                                        cands.append(p)
                                if not cands:
                                    return False, None
                                if best_candidates is None or len(cands) < len(best_candidates):
                                    best_i = i
                                    best_candidates = cands

                            rem2 = set(remaining)
                            rem2.remove(best_i)
                            for p in best_candidates:
                                chosen[best_i] = p
                                ok, sol = backtrack(used | p["bits"], rem2, chosen)
                                if ok:
                                    return True, sol
                                del chosen[best_i]
                            return False, None

                        ok, sol = backtrack(used, remaining, dict(chosen))
                        if ok:
                            return H, W, sol
        return None

    # ---------- partition components into tileable groups ----------
    tileable = {}
    for mask in range(1, 1 << n):
        if bin(mask).count("1") >= 2:
            res = find_tiling(mask, None)
            if res is not None:
                tileable[mask] = True

    @lru_cache(None)
    def best_partition(mask):
        if mask == 0:
            return []
        bits = [i for i in range(n) if (mask >> i) & 1]
        first = bits[0]
        best = None

        # try groups containing first
        sub = mask
        while sub:
            if ((sub >> first) & 1) and sub in tileable:
                rest = mask ^ sub
                tail = best_partition(rest)
                if tail is not None:
                    cand = [sub] + tail
                    if best is None:
                        best = cand
                    else:
                        # prefer fewer leftover singletons implicitly, then smaller number of groups
                        cov_best = sum(bin(x).count("1") for x in best)
                        cov_cand = sum(bin(x).count("1") for x in cand)
                        if cov_cand > cov_best or (cov_cand == cov_best and len(cand) < len(best)):
                            best = cand
            sub = (sub - 1) & mask

        if best is not None:
            return best

        # no tileable group using first
        rest = best_partition(mask ^ (1 << first))
        if rest is None:
            return None
        return rest

    groups = best_partition((1 << n) - 1)
    grouped_mask = 0
    for g in groups:
        grouped_mask |= g

    out = [row[:] for row in grid]

    # ---------- reconstruct each group around each member ----------
    for mask in groups:
        idxs = subset_indices(mask)
        for anchor in idxs:
            res = find_tiling(mask, anchor)
            if res is None:
                continue
            H, W, sol = res
            anchor_place = sol[anchor]

            # rectangle top-left in output so anchor placement aligns with original anchor
            top0 = comps[anchor]["minr"] - anchor_place["top"]
            left0 = comps[anchor]["minc"] - anchor_place["left"]

            for i in idxs:
                color = comps[i]["color"]
                for rr, cc in sol[i]["cells"]:
                    gr, gc = top0 + rr, left0 + cc
                    if 0 <= gr < R and 0 <= gc < C:
                        out[gr][gc] = color

    return out
