2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import defaultdict, deque

    h, w = len(grid), len(grid[0])

    # Find marker columns from color 4
    markers = [(r, c) for r in range(h) for c in range(w) if grid[r][c] == 4]
    marker_cols = sorted(set(c for _, c in markers))
    if len(marker_cols) != 2:
        return grid
    left_marker_col, right_marker_col = marker_cols

    # Connected components for each nonzero/non-4 color
    visited = [[False] * w for _ in range(h)]
    comps_by_color = defaultdict(list)

    for r in range(h):
        for c in range(w):
            color = grid[r][c]
            if color in (0, 4) or visited[r][c]:
                continue
            q = deque([(r, c)])
            visited[r][c] = True
            cells = []
            while q:
                x, y = q.popleft()
                cells.append((x, y))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = x + dx, y + dy
                    if 0 <= nx < h and 0 <= ny < w and not visited[nx][ny] and grid[nx][ny] == color:
                        visited[nx][ny] = True
                        q.append((nx, ny))
            comps_by_color[color].append(cells)

    def is_vertical_line(cells):
        cols = {c for _, c in cells}
        if len(cols) != 1:
            return False
        rows = sorted(r for r, _ in cells)
        return rows == list(range(rows[0], rows[-1] + 1))

    items = []
    for color, comps in comps_by_color.items():
        line_comp = None
        shape_comp = None
        for comp in comps:
            if is_vertical_line(comp):
                line_comp = comp
            else:
                shape_comp = comp
        if line_comp is None or shape_comp is None:
            return grid

        line_col = line_comp[0][1]
        line_len = len(line_comp)

        min_r = min(r for r, c in shape_comp)
        max_r = max(r for r, c in shape_comp)
        min_c = min(c for r, c in shape_comp)
        max_c = max(c for r, c in shape_comp)

        shape_h = max_r - min_r + 1
        shape_w = max_c - min_c + 1
        shape = [[0] * shape_w for _ in range(shape_h)]
        for r, c in shape_comp:
            shape[r - min_r][c - min_c] = color

        side = 'left' if abs(line_col - left_marker_col) <= abs(line_col - right_marker_col) else 'right'
        items.append({
            "color": color,
            "line_len": line_len,
            "line_col": line_col,
            "shape": shape,
            "shape_h": shape_h,
            "shape_w": shape_w,
            "side": side
        })

    # Order by marker side / line column
    items.sort(key=lambda x: x["line_col"])
    if len(items) != 2:
        return grid

    internal_h = max(item["line_len"] for item in items)

    rendered = []
    for idx, item in enumerate(items):
        color = item["color"]
        shape = item["shape"]
        sh, sw = item["shape_h"], item["shape_w"]

        block = [[0] * (sw + 1) for _ in range(internal_h)]
        if item["side"] == 'left':
            for r in range(internal_h):
                block[r][0] = color
            for r in range(sh):
                for c in range(sw):
                    if shape[r][c] != 0:
                        block[r][c + 1] = color
        else:
            for r in range(internal_h):
                block[r][sw] = color
            for r in range(sh):
                for c in range(sw):
                    if shape[r][c] != 0:
                        block[r][c] = color
        rendered.append(block)

    left_block, right_block = rendered
    out_w = len(left_block[0]) + len(right_block[0])
    out = [[0] * out_w for _ in range(internal_h + 2)]

    out[0][0] = 4
    out[0][-1] = 4
    out[-1][0] = 4
    out[-1][-1] = 4

    lw = len(left_block[0])
    rw = len(right_block[0])

    for r in range(internal_h):
        for c in range(lw):
            out[r + 1][c] = left_block[r][c]
        for c in range(rw):
            out[r + 1][lw + c] = right_block[r][c]

    return out
