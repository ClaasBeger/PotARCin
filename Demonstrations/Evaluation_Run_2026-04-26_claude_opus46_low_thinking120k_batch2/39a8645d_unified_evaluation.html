def solve(grid):
    from collections import Counter
    
    rows, cols = len(grid), len(grid[0])
    
    # Find all connected components of non-zero cells

    def flood(r, c, val):
        stack = [(r,c)]
        cells = []
        while stack:
            cr, cc = stack.pop()
            if cr<0 or cr>=rows or cc<0 or cc>=cols: continue
            if visited[cr][cc] or grid[cr][cc] != val: continue
            visited[cr][cc] = True
            cells.append((cr,cc))
