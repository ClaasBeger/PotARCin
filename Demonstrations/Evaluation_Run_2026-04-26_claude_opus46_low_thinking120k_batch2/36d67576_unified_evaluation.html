2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import deque
    rows = len(grid)
    cols = len(grid[0])
    
    visited = [[False]*cols for _ in range(rows)]
    components = []
    
    def bfs(r, c):
        q = deque([(r,c)])
        visited[r][c] = True
        cells = [(r,c)]
        while q:
            cr, cc = q.popleft()
            for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                nr, nc = cr+dr, cc+dc
                if 0<=nr<rows and 0<=nc<cols and not visited[nr][nc] and grid[nr][nc]==4:
                    visited[nr][nc] = True
                    q.append((nr,nc))
                    cells.append((nr,nc))
        return cells
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 4 and not visited[r][c]:
                components.append(bfs(r, c))
    
    def get_decorations(comp):
        comp_set = set(comp)
        decos = {}
        checked = set()
        for r, c in comp:
            for dr in range(-3, 4):
                for dc in range(-3, 4):
                    nr, nc = r+dr, c+dc
                    if (nr,nc) in checked or (nr,nc) in comp_set:
                        continue
                    checked.add((nr,nc))
                    if 0<=nr<rows and 0<=nc<cols and grid[nr][nc] not in (0, 4):
                        decos[(nr,nc)] = grid[nr][nc]
        return decos
    
    template = None
    incompletes = []
    
    for comp in components:
        decos = get_decorations(comp)
        values = set(decos.values())
        if 1 in values and 3 in values:
            template = (comp, decos)
        else:
            incompletes.append((comp, decos))
    
    def normalize(cells):
        min_r = min(r for r,c in cells)
        min_c = min(c for r,c in cells)
        return set((r-min_r, c-min_c) for r,c in cells), min_r, min_c
    
    def get_transforms(cells_with_vals):
        transforms = [
            lambda r,c: (r, c),
            lambda r,c: (c, -r),
            lambda r,c: (-r, -c),
            lambda r,c: (-c, r),
            lambda r,c: (r, -c),
            lambda r,c: (-r, c),
            lambda r,c: (c, r),
            lambda r,c: (-c, -r),
        ]
        results = []
        for t in transforms:
            transformed = [(t(r,c), v) for (r,c), v in cells_with_vals]
            min_r = min(p[0] for p,v in transformed)
            min_c = min(p[1] for p,v in transformed)
            results.append([((p[0]-min_r, p[1]-min_c), v) for p, v in transformed])
        return results
    
    t_comp, t_decos = template
    t_norm, t_mr, t_mc = normalize(t_comp)
    
    all_cells = [((r-t_mr, c-t_mc), 4) for r,c in t_comp]
    for (r,c), v in t_decos.items():
        all_cells.append(((r-t_mr, c-t_mc), v))
    
    result = [row[:] for row in grid]
    
    for comp, decos in incompletes:
        i_norm, i_mr, i_mc = normalize(comp)
        for transformed in get_transforms(all_cells):
            t_fours = set(p for p, v in transformed if v == 4)
            if t_fours == i_norm:
                for (r,c), v in transformed:
                    if v != 4:
                        nr, nc = r + i_mr, c + i_mc
                        if 0<=nr<rows and 0<=nc<cols and result[nr][nc] == 0:
                            result[nr][nc] = v
                break
    
    return result
