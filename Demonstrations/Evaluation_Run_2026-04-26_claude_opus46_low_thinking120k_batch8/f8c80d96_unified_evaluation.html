Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    R = len(grid)
    C = len(grid[0])
    
    color = 0
    for r in range(R):
        for c in range(C):
            if grid[r][c] != 0:
                color = grid[r][c]
                break
        if color: break
    if not color:
        return [row[:] for row in grid]
    
    h_segs = []
    for r in range(R):
        c = 0
        while c < C:
            if grid[r][c] == color:
                start = c
                while c < C and grid[r][c] == color:
                    c += 1
                if c - start >= 2:
                    h_segs.append((r, start, c - 1))
            else:
                c += 1
    
    v_segs = []
    for c in range(C):
        r = 0
        while r < R:
            if grid[r][c] == color:
                start = r
                while r < R and grid[r][c] == color:
                    r += 1
                if r - start >= 2:
                    v_segs.append((c, start, r - 1))
            else:
                r += 1
    
    import math
    h_rows = sorted(set(s[0] for s in h_segs))
    v_cols = sorted(set(s[0] for s in v_segs))
    
    step = 0
    for lst in [h_rows, v_cols]:
        if len(lst) >= 2:
            for i in range(len(lst) - 1):
                d = lst[i + 1] - lst[i]
                step = math.gcd(step, d) if step else d
    if step == 0:
        step = max(R, C)
    
    ext_h_rows = sorted(r for r in range(R) if h_rows and r % step == h_rows[0] % step)
    ext_v_cols = sorted(c for c in range(C) if v_cols and c % step == v_cols[0] % step)
    
    def extrap(keys, vals, key, lo, hi):
        kv = dict(zip(keys, vals))
        if key in kv:
            return kv[key]
        if not keys:
            return None
        if len(keys) == 1:
            return max(lo, min(hi, vals[0]))
        if key > keys[-1]:
            k1, k2 = keys[-2], keys[-1]
        elif key < keys[0]:
            k1, k2 = keys[0], keys[1]
        else:
            k1, k2 = keys[0], keys[1]
            for i in range(len(keys) - 1):
                if keys[i] <= key <= keys[i + 1]:
                    k1, k2 = keys[i], keys[i + 1]
                    break
        v1, v2 = kv[k1], kv[k2]
        slope = (v2 - v1) / (k2 - k1)
        v = v1 + slope * (key - k1)
        return max(lo, min(hi, round(v)))
    
    h_data = {}
    for r, s, e in h_segs:
        if r not in h_data or (e - s) > (h_data[r][1] - h_data[r][0]):
            h_data[r] = (s, e)
    
    v_data = {}

        s = extrap(hk, hs, r, 0, C - 1)
        e = extrap(hk, he, r, 0, C - 1)
        if s is not None and e is not None and s <= e:
            for c in range(s, e + 1):
                colored.add((r, c))

    return result
