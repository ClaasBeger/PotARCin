2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import Counter

    rows, cols = len(grid), len(grid[0])

    row_bgs = [Counter(grid[r]).most_common(1)[0][0] for r in range(rows)]
    col_bgs = [Counter(grid[r][c] for r in range(rows)).most_common(1)[0][0] for c in range(cols)]

    split_type = None
    for r in range(1, rows):
        if row_bgs[r] != row_bgs[0]:
            split_type = 'h'; split_pos = r; break
    if split_type is None:
        for c in range(1, cols):
            if col_bgs[c] != col_bgs[0]:
                split_type = 'v'; split_pos = c; break

    if split_type == 'h':
        region_a = (range(split_pos), range(cols), row_bgs[0])
        region_b = (range(split_pos, rows), range(cols), row_bgs[split_pos])
    else:
        region_a = (range(rows), range(split_pos), col_bgs[0])
        region_b = (range(rows), range(split_pos, cols), col_bgs[split_pos])

    def get_cells(region):
        rr, cr, bg = region
        return {(r, c): grid[r][c] for r in rr for c in cr if grid[r][c] != bg}

    def find_components(cells):
        visited = set(); components = []
        for pos in cells:
            if pos in visited: continue
            stack = [pos]; comp = set()
            while stack:
                p = stack.pop()
                if p in visited or p not in cells: continue
                visited.add(p); comp.add(p)
                r, c = p
                for dr in [-1, 0, 1]:
                    for dc in [-1, 0, 1]:
                        if dr or dc: stack.append((r + dr, c + dc))
            components.append(comp)
        return components

    cells_a, cells_b = get_cells(region_a), get_cells(region_b)
    comps_a, comps_b = find_components(cells_a), find_components(cells_b)

    max_a = max((len(c) for c in comps_a), default=0)
    max_b = max((len(c) for c in comps_b), default=0)

    if max_a > max_b:
        t_region, m_region = region_a, region_b
        t_comps, m_cells = comps_a, cells_b
    else:
        t_region, m_region = region_b, region_a
        t_comps, m_cells = comps_b, cells_a

    m_colors = set(m_cells.values())
    templates = []
    for comp in t_comps:
        min_r, min_c = min(p[0] for p in comp), min(p[1] for p in comp)
        max_r, max_c = max(p[0] for p in comp), max(p[1] for p in comp)
        data = [[grid[r][c] for c in range(min_c, max_c + 1)] for r in range(min_r, max_r + 1)]
        special = [(r - min_r, c - min_c) for r in range(min_r, max_r + 1)
                   for c in range(min_c, max_c + 1) if grid[r][c] in m_colors]
        templates.append({'data': data, 'origin': (min_r, min_c), 'special': special})

    mr_start, mc_start = m_region[0][0], m_region[1][0]
    out_rows, out_cols = len(m_region[0]), len(m_region[1])
    m_bg = m_region[2]
    output = [[m_bg] * out_cols for _ in range(out_rows)]

    templates.sort(key=lambda t: -len(t['special']))
    used = set()

    for t in templates:
        if not t['special']: continue
        sp = t['special']
        or_, oc = t['origin']
        for m_pos in sorted(m_cells.keys()):
            if m_pos in used: continue
            dr = m_pos[0] - (or_ + sp[0][0])
            dc = m_pos[1] - (oc + sp[0][1])
            matched = set(); ok = True
            for s in sp:
                target = (or_ + s[0] + dr, oc + s[1] + dc)
                if target not in m_cells or target in used:
                    ok = False; break
                matched.add(target)
            if ok:
                used |= matched
                origin_r, origin_c = or_ + dr, oc + dc
                for tr in range(len(t['data'])):
                    for tc in range(len(t['data'][0])):
                        out_r = origin_r + tr - mr_start
                        out_c = origin_c + tc - mc_start
                        if 0 <= out_r < out_rows and 0 <= out_c < out_cols:
                            output[out_r][out_c] = t['data'][tr][tc]
                break

    return output
