2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import math

    # --------------------------------------------------------------------- #
    # 1. read input, collect colored components
    # --------------------------------------------------------------------- #
    rows = len(grid)
    cols = len(grid[0]) if rows else 0
    color_cells = {}
    for r in range(rows):
        for c in range(cols):
            v = grid[r][c]
            if v:
                color_cells.setdefault(v, []).append((r, c))

    # total number of coloured cells
    total = sum(len(p) for p in color_cells.values())
    N = int(math.isqrt(total))
    if N * N != total:
        # should never happen for the given puzzles
        raise ValueError("Total coloured cells not a perfect square")

    # --------------------------------------------------------------------- #
    # 2. helper : all rotations of a shape (including original)
    # --------------------------------------------------------------------- #
    def rotations(coords):
        """coords – list of (x,y) with arbitrary offset.
           returns list of (rotated list, height, width) for 0,90,180,270°"""
        if not coords:
            return []
        # normalise to minimal bounding box
        xs = [p[0] for p in coords]
        ys = [p[1] for p in coords]
        min_x, max_x = min(xs), max(xs)
        min_y, max_y = min(ys), max(ys)
        base = [(x - min_x, y - min_y) for x, y in coords]
        h0 = max_x - min_x + 1
        w0 = max_y - min_y + 1

        res = [(base, h0, w0)]  # 0°

        # 90° clockwise
        r1 = [(y, w0 - 1 - x) for (x, y) in base]
        h1, w1 = w0, h0
        res.append((r1, h1, w1))

        # 180°
        r2 = [(h1 - 1 - y, x) for (x, y) in r1]
        h2, w2 = h1, w1
        res.append((r2, h2, w2))

        # 270°
        r3 = [(w2 - 1 - x, y) for (x, y) in r2]
        h3, w3 = w2, h2
        res.append((r3, h3, w3))

        return res

    # --------------------------------------------------------------------- #
    # 3. try to fill a block for a given anchor
    # --------------------------------------------------------------------- #
    # order in which we try the remaining colours (descending looks similar to examples)
    other_colors_order = lambda lst: sorted(lst, reverse=True)

    # back‑tracking placer
    def place_other_shapes(anchor_color, anchor_min_r, anchor_min_c,
                           a_h, a_w, other_cols):
        # block top‑left (anchor is bottom‑left aligned)
        top = anchor_min_r - (N - a_h)
        left = anchor_min_c  # j0 = 0

        # occupancy matrix for the block
        occ = [[False] * N for _ in range(N)]
        # place anchor (it stays where it is)
        for (r, c) in color_cells[anchor_color]:
            br = r - top
            bc = c - left
            occ[br][bc] = True

        # list of colours to place
        to_place = other_cols[:]
        # try descending order
        to_place.sort(reverse=True)

        # ----------------------------------------------------------------- #
        # recursive search
        # ----------------------------------------------------------------- #
        # we will store the final placements in a dictionary
        #   block_pos -> colour
        placements = {}   # (br,bc) -> colour

        def backtrack(idx):
            if idx == len(to_place):
                return True
            col = to_place[idx]
            shape_pts = color_cells[col]
            for rot_pts, rh, rw in rotations(shape_pts):
                for dy in range(N - rh + 1):
                    for dx in range(N - rw + 1):
                        # check all cells
                        ok = True
                        for (x, y) in rot_pts:
                            bx, by = x + dx, y + dy
                            if occ[bx][by]:
                                ok = False
                                break
                        if not ok:
                            continue
                        # place here
                        for (x, y) in rot_pts:
                            bx, by = x + dx, y + dy
                            occ[bx][by] = True
                            placements[(bx, by)] = col
                        if backtrack(idx + 1):
                            return True
                        # backtrack
                        for (x, y) in rot_pts:
                            bx, by = x + dx, y + dy
                            occ[bx][by] = False
                            del placements[(bx, by)]
            return False

        if not backtrack(0):
            return None   # cannot place – should not happen for the puzzles

        # translate block coordinates back to global grid
        for (br, bc), col in placements.items():
            gr = top + br
            gc = left + bc
            # write into output grid (zeros are overwritten)
            out_grid[gr][gc] = col

    # --------------------------------------------------------------------- #
    # 4. build output: start with an empty grid
    # --------------------------------------------------------------------- #
    out_grid = [[0] * cols for _ in range(rows)]

    # --------------------------------------------------------------------- #
    # 5. treat every colour as an anchor, create its block
    # --------------------------------------------------------------------- #
    for col, pts in color_cells.items():
        # bounding box of the anchor
        min_r = min(r for r, c in pts)
        max_r = max(r for r, c in pts)
        min_c = min(c for r, c in pts)
        max_c = max(c for r, c in pts)
        a_h = max_r - min_r + 1
        a_w = max_c - min_c + 1

        # block top‑left (bottom‑left alignment)
        top = min_r - (N - a_h)
        left = min_c  # j0 = 0

        # sanity check – block must stay inside the board
        if top < 0 or left < 0 or top + N > rows or left + N > cols:
            # shouldn't happen for the given data
            continue

        # write the anchor (unchanged)
        for (r, c) in pts:
            out_grid[r][c] = col

        # other colours
        others = [c for c in color_cells if c != col]
        # try to fill the block
        place_other_shapes(col, min_r, min_c, a_h, a_w, others)

    return out_grid
