2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    """
    Transforms the input grid according to the discovered rule.
    """
    from collections import Counter

    n = len(grid)
    if n == 0:
        return grid
    m = len(grid[0])

    # ------------------------------------------------------------
    # 1. Compute frequency of each colour and bounding boxes
    # ------------------------------------------------------------
    freq = Counter(cell for row in grid for cell in row)

    # bounding boxes for every non‑zero colour
    boxes = {}  # colour -> {'min_r', 'max_r', 'min_c', 'max_c'}
    for r in range(n):
        for c in range(m):
            val = grid[r][c]
            if val == 0:
                continue
            if val not in boxes:
                boxes[val] = {'min_r': r, 'max_r': r, 'min_c': c, 'max_c': c}
            else:
                b = boxes[val]
                if r < b['min_r']: b['min_r'] = r
                if r > b['max_r']: b['max_r'] = r
                if c < b['min_c']: b['min_c'] = c
                if c > b['max_c']: b['max_c'] = c

    # ------------------------------------------------------------
    # 2. Identify the "odd" cell (the single misplaced cell)
    # ------------------------------------------------------------
    # colours that occur exactly once (ignoring the background 0)
    unique = [c for c, cnt in freq.items() if cnt == 1 and c != 0]

    odd_colour = None
    odd_pos = None

    for col in unique:
        # position of this colour
        pos = None
        for r in range(n):
            for c in range(m):
                if grid[r][c] == col:
                    pos = (r, c)
                    break
            if pos is not None:
                break

        # smallest region that contains this cell (different colour)
        best_region = None
        best_area = float('inf')
        for colour, b in boxes.items():
            if colour == col:
                continue
            if b['min_r'] <= pos[0] <= b['max_r'] and b['min_c'] <= pos[1] <= b['max_c']:
                area = (b['max_r'] - b['min_r'] + 1) * (b['max_c'] - b['min_c'] + 1)
                if area < best_area:
                    best_area = area
                    best_region = colour

        # centre of that enclosing region
        b = boxes[best_region]
        centre = ((b['min_r'] + b['max_r']) // 2, (b['min_c'] + b['max_c']) // 2)

        # if the cell is not at the centre of its enclosing region → it's the odd one
        if pos != centre:
            odd_colour = col
            odd_pos = pos
            break

    # If for some reason no odd cell was found, fall back to the first unique colour
    if odd_colour is None:
        col = unique[0]
        for r in range(n):
            for c in range(m):
                if grid[r][c] == col:
                    odd_colour = col
                    odd_pos = (r, c)
                    break
            if odd_colour is not None:
                break

    # ------------------------------------------------------------
    # 3. Determine the colour that should replace the odd cell
    # ------------------------------------------------------------
    # smallest region that contains the odd cell (excluding the odd colour itself)
    replace_colour = None
    best_area = float('inf')
    for colour, b in boxes.items():
        if colour == odd_colour:
            continue
        if b['min_r'] <= odd_pos[0] <= b['max_r'] and b['min_c'] <= odd_pos[1] <= b['max_c']:
            area = (b['max_r'] - b['min_r'] + 1) * (b['max_c'] - b['min_c'] + 1)
            if area < best_area:
                best_area = area
                replace_colour = colour

    # Apply the replacement
    grid[odd_pos[0]][odd_pos[1]] = replace_colour

    # ------------------------------------------------------------
    # 4. Compute the depth (number of nested layers) of the region
    #    that contained the odd cell.
    # ------------------------------------------------------------
    # sort colours by bounding‑box area (largest first) – exclude the odd colour
    sorted_cols = sorted(
        [(c, (boxes[c]['max_r'] - boxes[c]['min_r'] + 1) *
             (boxes[c]['max_c'] - boxes[c]['min_c'] + 1)) for c in boxes if c != odd_colour],
        key=lambda x: x[1], reverse=True
    )
    depth = 1
    for i, (c, _) in enumerate(sorted_cols):
        if c == replace_colour:
            depth = i + 1
            break

    # ------------------------------------------------------------
    # 5. Draw the vertical line of 9's at the bottom‑right corner
    # ------------------------------------------------------------
    start_row = n - depth
    for r in range(start_row, n):
        grid[r][m - 1] = 9

    return grid
