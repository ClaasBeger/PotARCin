2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

from collections import Counter
from copy import deepcopy

def solve(grid):
    n = len(grid)
    m = len(grid[0]) if n else 0

    # ------------------------------------------------------------------
    # 1. find the most frequent non‑zero value (the axis value)
    # ------------------------------------------------------------------
    flat = [v for row in grid for v in row]
    cnt = Counter(flat)
    cnt.pop(0, None)                     # ignore zeros
    if not cnt:
        return grid                       # empty grid – nothing to do
    axis_val = max(cnt, key=cnt.get)      # most frequent value

    # ------------------------------------------------------------------
    # 2. gather possible axis lines
    # ------------------------------------------------------------------
    candidates = []   # (kind, index, count, pure)

    # central rows
    if n % 2 == 1:
        central_rows = [n // 2]
    else:
        central_rows = [n // 2 - 1, n // 2]

    for r in central_rows:
        col_vals = [grid[r][c] for c in range(m)]
        count = col_vals.count(axis_val)
        pure = all(v == 0 or v == axis_val for v in col_vals)
        candidates.append(('row', r, count, pure))

    # central columns
    if m % 2 == 1:
        central_cols = [m // 2]
    else:
        central_cols = [m // 2 - 1, m // 2]

    for c in central_cols:
        row_vals = [grid[r][c] for r in range(n)]
        count = row_vals.count(axis_val)
        pure = all(v == 0 or v == axis_val for v in row_vals)
        candidates.append(('col', c, count, pure))

    # main diagonal (only for square grids)
    if n == m:
        diag_count = sum(1 for i in range(n) if grid[i][i] == axis_val)
        pure = all(grid[i][i] == 0 or grid[i][i] == axis_val for i in range(n))
        candidates.append(('diag', -1, diag_count, pure))

        # anti diagonal
        anti_count = sum(1 for i in range(n) if grid[i][n - 1 - i] == axis_val)
        pure = all(grid[i][n - 1 - i] == 0 or grid[i][n - 1 - i] == axis_val for i in range(n))
        candidates.append(('anti', -1, anti_count, pure))

    # ------------------------------------------------------------------
    # 3. choose the axis line
    # ------------------------------------------------------------------
    # separate pure and non‑pure candidates
    non_pure = [c for c in candidates if not c[3]]
    if non_pure:
        pool = non_pure
    else:
        pool = candidates

    # priority order for tie breaking
    priority = {'row': 0, 'col': 1, 'diag': 2, 'anti': 3}
    pool.sort(key=lambda x: (-x[2], priority[x[0]]))
    axis_kind, axis_idx, _, _ = pool[0]

    # ------------------------------------------------------------------
    # 4. helper: reflection according to the selected axis
    # ------------------------------------------------------------------
    def reflect(r, c):
        if axis_kind == 'row':
            nr = 2 * axis_idx - r
            nc = c
        elif axis_kind == 'col':
            nr = r
            nc = 2 * axis_idx - c
        elif axis_kind == 'diag':
            nr = c
            nc = r
        elif axis_kind == 'anti':
            nr = n - 1 - c
            nc = n - 1 - r
        else:
            nr, nc = r, c
        return nr, nc

    # ------------------------------------------------------------------
    # 5. move each non‑axis number
    # ------------------------------------------------------------------
    new_grid = deepcopy(grid)

    for r in range(n):
        for c in range(m):
            val = grid[r][c]
            if val == 0 or val == axis_val:
                continue

            r2, c2 = reflect(r, c)
            # out of bounds ?
            if not (0 <= r2 < n and 0 <= c2 < m):
                continue
            # target not empty ?
            if new_grid[r2][c2] != 0:
                continue

            final_r, final_c = r2, c2

            # ----- check adjacency to axis value -----
            def has_axis_neighbour(pr, pc):
                for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    nr, nc = pr + dr, pc + dc
                    if 0 <= nr < n and 0 <= nc < m:
                        if new_grid[nr][nc] == axis_val:
                            return True
                return False

            if has_axis_neighbour(final_r, final_c):
                # decide direction towards the axis
                if axis_kind == 'row':
                    # vertical shift
                    step = -1 if final_r > axis_idx else 1 if final_r < axis_idx else 0
                    if step != 0:
                        cr, cc = final_r, final_c
                        while True:
                            nr, nc = cr + step, cc
                            if not (0 <= nr < n):
                                break
                            if new_grid[nr][nc] == 0:
                                cr, cc = nr, nc
                                break
                            cr, nc = nr, nc   # keep moving even if occupied (skip)
                        final_r, final_c = cr, cc

                elif axis_kind == 'col':
                    # horizontal shift
                    step = -1 if final_c > axis_idx else 1 if final_c < axis_idx else 0
                    if step != 0:
                        cr, cc = final_r, final_c
                        while True:
                            nr, nc = cr, cc + step
                            if not (0 <= nc < m):
                                break
                            if new_grid[nr][nc] == 0:
                                cr, cc = nr, nc
                                break
                            cr, cc = nr, nc
                        final_r, final_c = cr, cc

                else:   # diagonal or anti‑diagonal – use central row/col
                    axis_row = n // 2
                    axis_col = m // 2
                    dr = final_r - axis_row
                    dc = final_c - axis_col
                    # shift along the smaller distance
                    if abs(dc) <= abs(dr):
                        step = -1 if dc > 0 else 1 if dc < 0 else 0
                        if step != 0:
                            cr, cc = final_r, final_c
                            while True:
                                nr, nc = cr, cc + step
                                if not (0 <= nc < m):
                                    break
                                if new_grid[nr][nc] == 0:
                                    cr, cc = nr, nc
                                    break
                                cr, cc = nr, nc
                            final_r, final_c = cr, cc
                    else:
                        step = -1 if dr > 0 else 1 if dr < 0 else 0
                        if step != 0:
                            cr, cc = final_r, final_c
                            while True:
                                nr, nc = cr + step, cc
                                if not (0 <= nr < n):
                                    break
                                if new_grid[nr][nc] == 0:
                                    cr, cc = nr, nc
                                    break
                                cr, cc = nr, nc
                            final_r, final_c = cr, cc

            # perform the move
            new_grid[final_r][final_c] = val
            new_grid[r][c] = 0

    return new_grid
