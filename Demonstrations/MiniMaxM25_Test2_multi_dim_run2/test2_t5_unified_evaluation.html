2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import collections

    H = len(grid)
    W = len(grid[0])

    # ---------- find barrier (line of 7s) ----------
    # longest run of 7s in rows
    max_row_run = 0
    row_idx = -1
    for i in range(H):
        cnt = 0
        best = 0
        for j in range(W):
            if grid[i][j] == 7:
                cnt += 1
            else:
                best = max(best, cnt)
                cnt = 0
        best = max(best, cnt)
        if best > max_row_run:
            max_row_run = best
            row_idx = i

    # longest run of 7s in columns
    max_col_run = 0
    col_idx = -1
    for j in range(W):
        cnt = 0
        best = 0
        for i in range(H):
            if grid[i][j] == 7:
                cnt += 1
            else:
                best = max(best, cnt)
                cnt = 0
        best = max(best, cnt)
        if best > max_col_run:
            max_col_run = best
            col_idx = j

    # decide orientation
    if max_row_run >= max_col_run:
        orientation = 'horizontal'
        barrier_i = row_idx
        barrier_line = [(barrier_i, j) for j in range(W)]
    else:
        orientation = 'vertical'
        barrier_j = col_idx
        barrier_line = [(i, barrier_j) for i in range(H)]

    # ----- find hole (zero) on the barrier -----
    hole_pos = None
    for (i, j) in barrier_line:
        if grid[i][j] == 0:
            hole_pos = j if orientation == 'horizontal' else i
            break
    hole_exists = hole_pos is not None

    # ----- region definitions -----
    if orientation == 'horizontal':
        top_rows = list(range(0, barrier_i))
        bottom_rows = list(range(barrier_i + 1, H))
        cols = list(range(W))
        region_top = (top_rows, cols)
        region_bottom = (bottom_rows, cols)
    else:
        left_cols = list(range(0, barrier_j))
        right_cols = list(range(barrier_j + 1, W))
        rows = list(range(H))
        region_left = (rows, left_cols)
        region_right = (rows, right_cols)

    # ----- find the largest 8 component (anchor) -----
    visited = [[False] * W for _ in range(H)]
    anchor_cells = []
    max_sz = 0
    for i in range(H):
        for j in range(W):
            if grid[i][j] == 8 and not visited[i][j]:
                q = collections.deque()
                q.append((i, j))
                visited[i][j] = True
                comp = []
                while q:
                    x, y = q.popleft()
                    comp.append((x, y))
                    for dx, dy in ((1,0),(-1,0),(0,1),(0,-1)):
                        nx, ny = x+dx, y+dy
                        if 0 <= nx < H and 0 <= ny < W and not visited[nx][ny] and grid[nx][ny] == 8:
                            visited[nx][ny] = True
                            q.append((nx, ny))
                if len(comp) > max_sz:
                    max_sz = len(comp)
                    anchor_cells = comp
    anchor_set = set(anchor_cells)

    total_eight = sum(1 for i in range(H) for j in range(W) if grid[i][j] == 8)
    floating_cnt = total_eight - len(anchor_cells)

    # ----- compute region data -----
    if orientation == 'horizontal':
        reg_top, reg_bottom = region_top, region_bottom
    else:
        reg_left, reg_right = region_left, region_right

    # helper to count anchor cells inside a region
    def anchor_in_region(region_rows, region_cols):
        cnt = 0
        rows_set = set(region_rows)
        cols_set = set(region_cols)
        for (i, j) in anchor_cells:
            if i in rows_set and j in cols_set:
                cnt += 1
        return cnt

    if orientation == 'horizontal':
        anchor_top = anchor_in_region(*reg_top)
        anchor_bottom = anchor_in_region(*reg_bottom)
        cap_top = len(reg_top[0]) * len(reg_top[1])
        cap_bottom = len(reg_bottom[0]) * len(reg_bottom[1])
    else:
        anchor_left = anchor_in_region(*reg_left)
        anchor_right = anchor_in_region(*reg_right)
        cap_left = len(reg_left[0]) * len(reg_left[1])
        cap_right = len(reg_right[0]) * len(reg_right[1])

    # ----- allocate floating cells -----
    # decide which region is anchor region
    if orientation == 'horizontal':
        anchor_in_top = anchor_top > 0
        if anchor_in_top:
            anchor_region = 'top'
            other_region = 'bottom'
            anchor_rows, anchor_cols = reg_top
            other_rows, other_cols = reg_bottom
            anchor_cap = cap_top
            other_cap = cap_bottom
            anchor_own = anchor_top
            other_own = anchor_bottom
        else:
            anchor_region = 'bottom'
            other_region = 'top'
            anchor_rows, anchor_cols = reg_bottom
            other_rows, other_cols = reg_top
            anchor_cap = cap_bottom
            other_cap = cap_top
            anchor_own = anchor_bottom
            other_own = anchor_top
    else:
        anchor_in_left = anchor_left > 0
        if anchor_in_left:
            anchor_region = 'left'
            other_region = 'right'
            anchor_rows, anchor_cols = reg_left
            other_rows, other_cols = reg_right
            anchor_cap = cap_left
            other_cap = cap_right
            anchor_own = anchor_left
            other_own = anchor_right
        else:
            anchor_region = 'right'
            other_region = 'left'
            anchor_rows, anchor_cols = reg_right
            other_rows, other_cols = reg_left
            anchor_cap = cap_right
            other_cap = cap_left
            anchor_own = anchor_right
            other_own = anchor_left

    free_anchor = anchor_cap - anchor_own
    free_other = other_cap - other_own

    # allocate according to hole existence
    if hole_exists:
        # first fill anchor region
        alloc_anchor = min(free_anchor, floating_cnt)
        remaining = floating_cnt - alloc_anchor
        # barrier positions that are free (zero) after clearing non‑anchor 8s
        # will be computed later after we cleared the grid
        alloc_barrier = 0
        alloc_other = 0
        # placeholder – we'll compute barrier later
    else:
        # first fill other region
        alloc_other = min(free_other, floating_cnt)
        remaining = floating_cnt - alloc_other
        alloc_barrier = 0
        alloc_anchor = 0

    # After tentative allocation we still need to handle barrier cells
    # We'll recompute properly after we have the cleared grid

    # ---- clear grid, keep 7 and anchor ----
    out = [row[:] for row in grid]
    for i in range(H):
        for j in range(W):
            if out[i][j] == 8:
                out[i][j] = 0
    for (i, j) in anchor_cells:
        out[i][j] = 8

    # ---- compute free barrier positions (zero, not 7) ----
    barrier_free = []
    for (i, j) in barrier_line:
        if out[i][j] == 0:
            barrier_free.append((i, j))

    # ---- redo allocation with correct barrier capacity ----
    if hole_exists:
        alloc_anchor = min(free_anchor, floating_cnt)
        remaining = floating_cnt - alloc_anchor
        alloc_barrier = min(remaining, len(barrier_free))
        remaining -= alloc_barrier
        alloc_other = min(free_other, remaining)
        # any leftover stays zero (should not happen)
    else:
        alloc_other = min(free_other, floating_cnt)
        remaining = floating_cnt - alloc_other
        alloc_barrier = min(remaining, len(barrier_free))
        remaining -= alloc_barrier
        alloc_anchor = min(free_anchor, remaining)

    # ---- fill anchor region ----
    # prepare anchor positions inside the region
    def anchor_in_region_set(region_rows, region_cols):
        s = set()
        rows_set = set(region_rows)
        cols_set = set(region_cols)
        for (i, j) in anchor_cells:
            if i in rows_set and j in cols_set:
                s.add((i, j))
        return s

    def fill_region(region_rows, region_cols, anchor_set_in_region, extra_cnt, hole_col):
        height = len(region_rows)
        width = len(region_cols)
        # initial grid: 0 everywhere
        region_grid = [[0] * width for _ in range(height)]
        # place anchor cells
        for (gi, gj) in anchor_set_in_region:
            li = region_rows.index(gi)
            lj = region_cols.index(gj)
            region_grid[li][lj] = 8
        total = len(anchor_set_in_region) + extra_cnt
        full_rows = total // width
        remainder = total % width
        # fill full rows from bottom, but only where cell is 0
        for r in range(full_rows):
            row_idx = height - 1 - r
            for c in range(width):
                if region_grid[row_idx][c] == 0:
                    region_grid[row_idx][c] = 8
        # remainder row
        if remainder:
            row_idx = height - 1 - full_rows
            start = full_rows
            if hole_col is not None:
                if hole_col in region_cols:
                    hole_local = region_cols.index(hole_col)
                    if not (start <= hole_local <= start + remainder - 1):
                        start = (width - remainder) // 2
                else:
                    start = (width - remainder) // 2
            else:
                start = (width - remainder) // 2
            if start + remainder > width:
                start = width - remainder
            for c in range(remainder):
                if region_grid[row_idx][start + c] == 0:
                    region_grid[row_idx][start + c] = 8
        return region_grid

    if orientation == 'horizontal':
        if anchor_region == 'top':
            anchor_set_in_top = anchor_in_region_set(*reg_top)
            region_grid_anchor = fill_region(*reg_top, anchor_set_in_top, alloc_anchor, hole_pos)
            # apply
            for li, gi in enumerate(reg_top[0]):
                for lj, gj in enumerate(reg_top[1]):
                    if region_grid_anchor[li][lj] == 8:
                        out[gi][gj] = 8
            # other region
            if alloc_other:
                anchor_set_in_bottom = anchor_in_region_set(*reg_bottom)
                region_grid_other = fill_region(*reg_bottom, anchor_set_in_bottom, alloc_other, hole_pos)
                for li, gi in enumerate(reg_bottom[0]):
                    for lj, gj in enumerate(reg_bottom[1]):
                        if region_grid_other[li][lj] == 8:
                            out[gi][gj] = 8
        else:
            anchor_set_in_bottom = anchor_in_region_set(*reg_bottom)
            region_grid_anchor = fill_region(*reg_bottom, anchor_set_in_bottom, alloc_anchor, hole_pos)
            for li, gi in enumerate(reg_bottom[0]):
                for lj, gj in enumerate(reg_bottom[1]):
                    if region_grid_anchor[li][lj] == 8:
                        out[gi][gj] = 8
            if alloc_other:
                anchor_set_in_top = anchor_in_region_set(*reg_top)
                region_grid_other = fill_region(*reg_top, anchor_set_in_top, alloc_other, hole_pos)
                for li, gi in enumerate(reg_top[0]):
                    for lj, gj in enumerate(reg_top[1]):
                        if region_grid_other[li][lj] == 8:
                            out[gi][gj] = 8
    else:  # vertical
        if anchor_region == 'left':
            anchor_set_in_left = anchor_in_region_set(*reg_left)
            region_grid_anchor = fill_region(*reg_left, anchor_set_in_left, alloc_anchor, None)  # no hole column for vertical
            for li, gi in enumerate(reg_left[0]):
                for lj, gj in enumerate(reg_left[1]):
                    if region_grid_anchor[li][lj] == 8:
                        out[gi][gj] = 8
            if alloc_other:
                anchor_set_in_right = anchor_in_region_set(*reg_right)
                region_grid_other = fill_region(*reg_right, anchor_set_in_right, alloc_other, None)
                for li, gi in enumerate(reg_right[0]):
                    for lj, gj in enumerate(reg_right[1]):
                        if region_grid_other[li][lj] == 8:
                            out[gi][gj] = 8
        else:
            anchor_set_in_right = anchor_in_region_set(*reg_right)
            region_grid_anchor = fill_region(*reg_right, anchor_set_in_right, alloc_anchor, None)
            for li, gi in enumerate(reg_right[0]):
                for lj, gj in enumerate(reg_right[1]):
                    if region_grid_anchor[li][lj] == 8:
                        out[gi][gj] = 8
            if alloc_other:
                anchor_set_in_left = anchor_in_region_set(*reg_left)
                region_grid_other = fill_region(*reg_left, anchor_set_in_left, alloc_other, None)
                for li, gi in enumerate(reg_left[0]):
                    for lj, gj in enumerate(reg_left[1]):
                        if region_grid_other[li][lj] == 8:
                            out[gi][gj] = 8

    # ---- place cells on barrier holes ----
    for k in range(alloc_barrier):
        i, j = barrier_free[k]
        out[i][j] = 8

    return out
