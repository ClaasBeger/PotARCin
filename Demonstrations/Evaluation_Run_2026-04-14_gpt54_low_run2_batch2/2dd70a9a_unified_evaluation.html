2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0])

    out = [row[:] for row in grid]

    pos2 = [(r, c) for r in range(h) for c in range(w) if grid[r][c] == 2]
    pos3 = [(r, c) for r in range(h) for c in range(w) if grid[r][c] == 3]

    if not pos2 or not pos3:
        return out

    def orientation(cells):
        rs = {r for r, _ in cells}
        cs = {c for _, c in cells}
        if len(rs) == 1:
            return "h"
        if len(cs) == 1:
            return "v"
        return None

    ori3 = orientation(pos3)
    ori2 = orientation(pos2)
    ori = ori3 if ori3 == ori2 else ori3

    def center(cells):
        sr = sum(r for r, _ in cells) / len(cells)
        sc = sum(c for _, c in cells) / len(cells)
        return sr, sc

    def clear_vertical(col, r1, r2):
        a, b = sorted((r1, r2))
        for r in range(a, b + 1):
            if grid[r][col] not in (0, 2, 3):
                return False
        return True

    def clear_horizontal(row, c1, c2):
        a, b = sorted((c1, c2))
        for c in range(a, b + 1):
            if grid[row][c] not in (0, 2, 3):
                return False
        return True

    if ori == "v":
        c3 = pos3[0][1]
        c2 = pos2[0][1]
        r3s = sorted(r for r, _ in pos3)
        r2s = sorted(r for r, _ in pos2)
        ctr3, _ = center(pos3)
        ctr2, _ = center(pos2)

        if ctr2 < ctr3:
            start3 = min(r3s)
            dir3 = -1
            start2 = max(r2s)
            dir2 = 1
        else:
            start3 = max(r3s)
            dir3 = 1
            start2 = min(r2s)
            dir2 = -1

        reach3 = []
        r = start3 + dir3
        while 0 <= r < h and grid[r][c3] == 0:
            reach3.append(r)
            r += dir3

        reach2 = []
        r = start2 + dir2
        while 0 <= r < h and grid[r][c2] == 0:
            reach2.append(r)
            r += dir2

        common = []
        s2 = set(reach2)
        for r in reach3:
            if r in s2 and clear_horizontal(r, c3, c2):
                common.append(r)

        if not common:
            return out

        row = min(common)

        a, b = sorted((start3, row))
        for r in range(a, b + 1):
            if out[r][c3] == 0:
                out[r][c3] = 3

        a, b = sorted((start2, row))
        for r in range(a, b + 1):
            if out[r][c2] == 0:
                out[r][c2] = 3

        a, b = sorted((c3, c2))
        for c in range(a, b + 1):
            if out[row][c] == 0:
                out[row][c] = 3

        return out

    if ori == "h":
        r3 = pos3[0][0]
        r2 = pos2[0][0]
        c3s = sorted(c for _, c in pos3)
        c2s = sorted(c for _, c in pos2)

        candidates = []

        # right-side candidates
        right3 = []
        c = max(c3s) + 1
        while c < w and grid[r3][c] == 0:
            right3.append(c)
            c += 1

        right2 = []
        c = max(c2s) + 1
        while c < w and grid[r2][c] == 0:
            right2.append(c)
            c += 1

        common_right = sorted(set(right3) & set(right2))
        for c in common_right:
            if clear_vertical(c, r3, r2):
                candidates.append(c)

        # left-side candidates
        left3 = []
        c = min(c3s) - 1
        while c >= 0 and grid[r3][c] == 0:
            left3.append(c)
            c -= 1

        left2 = []
        c = min(c2s) - 1
        while c >= 0 and grid[r2][c] == 0:
            left2.append(c)
            c -= 1

        common_left = sorted(set(left3) & set(left2))
        for c in common_left:
            if clear_vertical(c, r3, r2):
                candidates.append(c)

        if not candidates:
            return out

        col = max(candidates)

        if col > max(c3s):
            a, b = sorted((max(c3s), col))
        else:
            a, b = sorted((min(c3s), col))
        for c in range(a, b + 1):
            if out[r3][c] == 0:
                out[r3][c] = 3

        if col > max(c2s):
            a, b = sorted((max(c2s), col))
        else:
            a, b = sorted((min(c2s), col))
        for c in range(a, b + 1):
            if out[r2][c] == 0:
                out[r2][c] = 3

        a, b = sorted((r3, r2))
        for r in range(a, b + 1):
            if out[r][col] == 0:
                out[r][col] = 3

        return out

    return out
