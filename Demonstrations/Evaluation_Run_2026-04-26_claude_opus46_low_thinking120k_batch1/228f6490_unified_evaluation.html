2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import deque
    rows, cols = len(grid), len(grid[0])
    result = [row[:] for row in grid]
    
    visited_5 = [[False]*cols for _ in range(rows)]
    rectangles = []
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 5 and not visited_5[r][c]:
                q = deque([(r,c)])
                visited_5[r][c] = True
                cells = [(r,c)]
                while q:
                    cr, cc = q.popleft()
                    for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                        nr, nc = cr+dr, cc+dc
                        if 0<=nr<rows and 0<=nc<cols and not visited_5[nr][nc] and grid[nr][nc]==5:
                            visited_5[nr][nc] = True
                            q.append((nr,nc))
                            cells.append((nr,nc))
                min_r = min(x[0] for x in cells)
                max_r = max(x[0] for x in cells)
                min_c = min(x[1] for x in cells)
                max_c = max(x[1] for x in cells)
                holes = []
                for hr in range(min_r, max_r+1):
                    for hc in range(min_c, max_c+1):
                        if grid[hr][hc] == 0:
                            holes.append((hr, hc))
                if holes:
                    rectangles.append((min_r, max_r, min_c, max_c, holes))
    
    in_rect = [[False]*cols for _ in range(rows)]
    for min_r, max_r, min_c, max_c, holes in rectangles:
        for r2 in range(min_r, max_r+1):
            for c2 in range(min_c, max_c+1):
                in_rect[r2][c2] = True
    
    visited_c = [[False]*cols for _ in range(rows)]
    patterns = []
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] not in (0, 5) and not visited_c[r][c] and not in_rect[r][c]:
                color = grid[r][c]
                q = deque([(r,c)])
                visited_c[r][c] = True
                cells = [(r,c)]
                while q:
                    cr, cc = q.popleft()
                    for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                        nr, nc = cr+dr, cc+dc
                        if 0<=nr<rows and 0<=nc<cols and not visited_c[nr][nc] and grid[nr][nc]==color and not in_rect[nr][nc]:
                            visited_c[nr][nc] = True
                            q.append((nr,nc))
                            cells.append((nr,nc))
                patterns.append((color, cells))
    
    def normalize(cells):
        min_r = min(x[0] for x in cells)
        min_c = min(x[1] for x in cells)
        return frozenset((r-min_r, c-min_c) for r,c in cells)
    
    used_patterns = set()
    for rect_idx, (min_r, max_r, min_c, max_c, holes) in enumerate(rectangles):
        hole_norm = normalize(holes)
        for pat_idx, (color, cells) in enumerate(patterns):
            if pat_idx in used_patterns:
                continue
            pat_norm = normalize(cells)
            if pat_norm == hole_norm:
                hole_min_r = min(x[0] for x in holes)
                hole_min_c = min(x[1] for x in holes)
                pat_min_r = min(x[0] for x in cells)
                pat_min_c = min(x[1] for x in cells)
                for pr, pc in cells:
                    tr = pr - pat_min_r + hole_min_r
                    tc = pc - pat_min_c + hole_min_c
                    result[tr][tc] = color
                for pr, pc in cells:
                    result[pr][pc] = 0
                used_patterns.add(pat_idx)
                break
    
    return result
