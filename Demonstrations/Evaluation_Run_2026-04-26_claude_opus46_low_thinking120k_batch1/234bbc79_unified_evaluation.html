2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import deque
    
    rows = len(grid)
    cols = len(grid[0])
    
    visited = [[False]*cols for _ in range(rows)]
    shapes = []
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] not in (0, 5) and not visited[r][c]:
                color = grid[r][c]
                cells = set()
                queue = deque([(r, c)])
                visited[r][c] = True
                while queue:
                    cr, cc = queue.popleft()
                    cells.add((cr, cc))
                    for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                        nr, nc = cr+dr, cc+dc
                        if 0 <= nr < rows and 0 <= nc < cols and not visited[nr][nc] and grid[nr][nc] == color:
                            visited[nr][nc] = True
                            queue.append((nr, nc))
                shapes.append((color, cells))
    
    five_to_shape = {}
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 5:
                for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                    nr, nc = r+dr, c+dc
                    if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] not in (0, 5):
                        for i, (color, cells) in enumerate(shapes):
                            if (nr, nc) in cells:
                                five_to_shape[(r, c)] = i
                                break
                        break
    
    extended = []
    shape_fives = [[] for _ in range(len(shapes))]
    for i, (color, cells) in enumerate(shapes):
        ext = set(cells)
        for pos, si in five_to_shape.items():
            if si == i:
                ext.add(pos)
                shape_fives[i].append(pos)
        extended.append((color, ext))
    
    order = sorted(range(len(shapes)), key=lambda i: min(c for _, c in shapes[i][1]))
    
    pairs = []
    used = set()
    for k in range(len(order) - 1):
        a, b = order[k], order[k+1]
        fa = [f for f in shape_fives[a] if f not in used]
        fb = [f for f in shape_fives[b] if f not in used]
        best = min(((x, y) for x in fa for y in fb), key=lambda p: abs(p[0][0]-p[1][0]) + abs(p[0][1]-p[1][1]))
        pairs.append((best[0], best[1], a, b))
        used.add(best[0])
        used.add(best[1])
    
    moves = {order[0]: (0, 0)}
    for fa, fb, sa, sb in pairs:
        ma = moves[sa]
        fa_new = (fa[0] + ma[0], fa[1] + ma[1])
        target = (fa_new[0], fa_new[1] + 1)
        moves[sb] = (target[0] - fb[0], target[1] - fb[1])
    
    all_cells = []
    for i, (color, ext) in enumerate(extended):
        dr, dc = moves[i]
        for r, c in ext:
            all_cells.append((r + dr, c + dc, color))
    
    min_c = min(c for _, c, _ in all_cells)
    max_c = max(c for _, c, _ in all_cells)
    
    result = [[0] * (max_c - min_c + 1) for _ in range(rows)]
    for r, c, color in all_cells:
        result[r][c - min_c] = color
    
    return result
