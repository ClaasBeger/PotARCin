def solve(grid):
    rows, cols = len(grid), len(grid[0])
    # Find columns with 5s and count their height
    col_heights = {}
    for c in range(cols):
        h = sum(1 for r in range(rows) if grid[r][c] == 5)
        if h > 0:
            col_heights[c] = h
    
    if not col_heights:
        return grid
    
    max_h = max(col_heights.values())
    min_h = min(col_heights.values())
    
    result = [[0]*cols for _ in range(rows)]
    
    for c, h in col_heights.items():
        if h == max_h:
            val = 1
        elif h == min_h:
            val = 2
        else:
            val = 0
        for r in range(rows):
            if grid[r][c] == 5:
                result[r][c] = val
    
    return result
