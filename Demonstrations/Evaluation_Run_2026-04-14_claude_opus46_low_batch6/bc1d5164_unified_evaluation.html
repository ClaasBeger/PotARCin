Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    left = [[grid[r][c] for c in range(3)] for r in range(rows)]
    right = [[grid[r][c] for c in range(4,7)] for r in range(rows)]
    out = [[0]*3 for _ in range(3)]
    for r in range(rows):
        mr = 0 if r<2 else (2 if r>=3 else 1)
        for c in range(3):
            out[mr][c] = max(out[mr][c], left[r][c], right[r][c])
    return out
