2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

from itertools import combinations

def solve(grid):
    h = len(grid)
    w = len(grid[0]) if h else 0
    out = [row[:] for row in grid]

    def neighbors(r, c):
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < h and 0 <= nc < w:
                yield nr, nc

    def components_of_value(val):
        vis = [[False] * w for _ in range(h)]
        comps = []
        for r in range(h):
            for c in range(w):
                if grid[r][c] == val and not vis[r][c]:
                    stack = [(r, c)]
                    vis[r][c] = True
                    cells = []
                    while stack:
                        x, y = stack.pop()
                        cells.append((x, y))
                        for nx, ny in neighbors(x, y):
                            if grid[nx][ny] == val and not vis[nx][ny]:
                                vis[nx][ny] = True
                                stack.append((nx, ny))
                    comps.append(cells)
        return comps

    def mixed_nonzero_components():
        vis = [[False] * w for _ in range(h)]
        comps = []
        for r in range(h):
            for c in range(w):
                if grid[r][c] != 0 and not vis[r][c]:
                    stack = [(r, c)]
                    vis[r][c] = True
                    cells = []
                    vals = set()
                    while stack:
                        x, y = stack.pop()
                        cells.append((x, y))
                        vals.add(grid[x][y])
                        for nx, ny in neighbors(x, y):
                            if grid[nx][ny] != 0 and not vis[nx][ny]:
                                vis[nx][ny] = True
                                stack.append((nx, ny))
                    comps.append((cells, vals))
        return comps

    def bbox(cells):
        rs = [r for r, _ in cells]
        cs = [c for _, c in cells]
        return min(rs), min(cs), max(rs), max(cs)

    def draw_block(r0, c0, hh, ww, color=1):
        for r in range(r0, r0 + hh):
            for c in range(c0, c0 + ww):
                if 0 <= r < h and 0 <= c < w and out[r][c] == 0:
                    out[r][c] = color

    # Find exemplar: connected nonzero component containing both 1 and 2
    exemplar = None
    for cells, vals in mixed_nonzero_components():
        if 1 in vals and 2 in vals:
            exemplar = cells
            break
    if exemplar is None:
        return out

    ex_set = set(exemplar)
    src_ones = [(r, c) for r, c in exemplar if grid[r][c] == 1]

    all_twos = components_of_value(2)
    src_two_comps = [comp for comp in all_twos if any(cell in ex_set for cell in comp)]
    tgt_two_comps = [comp for comp in all_twos if not any(cell in ex_set for cell in comp)]

    # Case 1: single 2-anchor in exemplar -> attach pattern to each target 2-object
    if len(src_two_comps) == 1:
        src2 = src_two_comps[0]
        sr0, sc0, sr1, sc1 = bbox(src2)

        for comp in tgt_two_comps:
            tr0, tc0, tr1, tc1 = bbox(comp)
            bh = tr1 - tr0 + 1
            bw = tc1 - tc0 + 1
            sh = max(1, bh - 1)
            sw = max(1, bw - 1)

            for r, c in src_ones:
                dr = r - sr0
                dc = c - sc0

                # vertical placement
                if r < sr0:  # above source 2
                    rr = tr0 - bh + (dr + 1) * sh
                elif r > sr1:  # below source 2
                    rr = tr1 + 1 + (r - sr1 - 1) * sh
                else:  # overlapping source 2 rows
                    rr = tr0 + (r - sr0) * sh

                # horizontal placement
                if c < sc0:  # left of source 2
                    cc = tc0 - bw + (dc + 1) * sw
                elif c > sc1:  # right of source 2
                    cc = tc1 + 1 + (c - sc1 - 1) * sw
                else:  # overlapping source 2 cols
                    cc = tc0 + (c - sc0) * sw

                draw_block(rr, cc, bh, bw, 1)

        return out

    # Case 2: multiple 2-components define a scalable template
    src_boxes = [bbox(comp) for comp in src_two_comps]
    src_boxes.sort()
    sminr = min(b[0] for b in src_boxes)
    sminc = min(b[1] for b in src_boxes)
    src_rel = sorted((b[0] - sminr, b[1] - sminc) for b in src_boxes)

    # Group target 2-components by size
    size_groups = {}
    for comp in tgt_two_comps:
        b = bbox(comp)
        key = (b[2] - b[0] + 1, b[3] - b[1] + 1)
        size_groups.setdefault(key, []).append(comp)

    for (bh, bw), comps in size_groups.items():
        m = len(src_two_comps)
        if len(comps) < m:
            continue
        comps_boxes = [bbox(comp) for comp in comps]

        used_subsets = []
        for idxs in combinations(range(len(comps_boxes)), m):
            boxes = [comps_boxes[i] for i in idxs]
            boxes.sort()
            tminr = min(b[0] for b in boxes)
            tminc = min(b[1] for b in boxes)
            rel = sorted((b[0] - tminr, b[1] - tminc) for b in boxes)

            ok = True
            for (sr, sc), (tr, tc) in zip(src_rel, rel):
                if tr != sr * bh or tc != sc * bw:
                    ok = False
                    break
            if not ok:
                continue

            if idxs in used_subsets:
                continue
            used_subsets.append(idxs)

            for r, c in src_ones:
                rr = tminr + (r - sminr) * bh
                cc = tminc + (c - sminc) * bw
                draw_block(rr, cc, bh, bw, 1)

    return out
