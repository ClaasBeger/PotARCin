2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0]) if h else 0

    orig = [row[:] for row in grid]
    out = [[0] * w for _ in range(h)]

    # Default: all 5s become 2
    for r in range(h):
        for c in range(w):
            if orig[r][c] == 5:
                out[r][c] = 2

    # Candidate cells: part of at least one 2x2 block of 5s
    cand = [[False] * w for _ in range(h)]
    for r in range(h - 1):
        for c in range(w - 1):
            if orig[r][c] == 5 and orig[r + 1][c] == 5 and orig[r][c + 1] == 5 and orig[r + 1][c + 1] == 5:
                cand[r][c] = True
                cand[r + 1][c] = True
                cand[r][c + 1] = True
                cand[r + 1][c + 1] = True

    remaining = [row[:] for row in cand]

    def largest_rect(mask):
        best = None
        best_area = 0
        for r1 in range(h):
            cols_ok = [True] * w
            for r2 in range(r1, h):
                for c in range(w):
                    cols_ok[c] = cols_ok[c] and mask[r2][c]
                c = 0
                while c < w:
                    if not cols_ok[c]:
                        c += 1
                        continue
                    s = c
                    while c < w and cols_ok[c]:
                        c += 1
                    e = c - 1
                    hh = r2 - r1 + 1
                    ww = e - s + 1
                    if hh >= 2 and ww >= 2:
                        area = hh * ww
                        if area > best_area:
                            best_area = area
                            best = (r1, r2, s, e)
        return best

    def peel_rect(rect):
        r1, r2, c1, c2 = rect
        active = {(r, c) for r in range(r1, r2 + 1) for c in range(c1, c2 + 1)}

        changed = True
        while changed and active:
            changed = False

            # peel full columns with vertical protrusions
            cols = sorted({c for _, c in active})
            rows = sorted({r for r, _ in active})

            for c in cols:
                col_cells = sorted(r for r, cc in active if cc == c)
                if not col_cells:
                    continue
                if col_cells != list(range(min(col_cells), max(col_cells) + 1)):
                    continue
                rr1, rr2 = min(col_cells), max(col_cells)
                if rr2 - rr1 + 1 < 2:
                    continue
                protrude = False
                if rr1 > 0 and orig[rr1 - 1][c] == 5 and (rr1 - 1, c) not in active:
                    protrude = True
                if rr2 + 1 < h and orig[rr2 + 1][c] == 5 and (rr2 + 1, c) not in active:
                    protrude = True
                if protrude:
                    for r in col_cells:
                        active.discard((r, c))
                    changed = True

            # peel full rows with horizontal protrusions
            rows = sorted({r for r, _ in active})
            for r in rows:
                row_cells = sorted(c for rr, c in active if rr == r)
                if not row_cells:
                    continue
                if row_cells != list(range(min(row_cells), max(row_cells) + 1)):
                    continue
                cc1, cc2 = min(row_cells), max(row_cells)
                if cc2 - cc1 + 1 < 2:
                    continue
                protrude = False
                if cc1 > 0 and orig[r][cc1 - 1] == 5 and (r, cc1 - 1) not in active:
                    protrude = True
                if cc2 + 1 < w and orig[r][cc2 + 1] == 5 and (r, cc2 + 1) not in active:
                    protrude = True
                if protrude:
                    for c in row_cells:
                        active.discard((r, c))
                    changed = True

        return active

    while True:
        rect = largest_rect(remaining)
        if rect is None:
            break

        r1, r2, c1, c2 = rect
        for r in range(r1, r2 + 1):
            for c in range(c1, c2 + 1):
                remaining[r][c] = False

        kept = peel_rect(rect)
        for r, c in kept:
            out[r][c] = 8

    return out
