2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    R = len(grid)
    C_cols = len(grid[0])
    
    # Find all non-zero pixels
    non_zeros = set()
    for r in range(R):
        for c in range(C_cols):
            if grid[r][c] != 0:
                non_zeros.add((r, c))
                
    # Group non-zeros into 8-connected components
    visited = set()
    components = []
    for r, c in non_zeros:
        if (r, c) not in visited:
            comp = []
            q = [(r, c)]
            visited.add((r, c))
            while q:
                curr_r, curr_c = q.pop(0)
                comp.append((curr_r, curr_c))
                for dr in [-1, 0, 1]:
                    for dc in [-1, 0, 1]:
                        if dr == 0 and dc == 0: continue
                        nr, nc = curr_r + dr, curr_c + dc
                        if (nr, nc) in non_zeros and (nr, nc) not in visited:
                            visited.add((nr, nc))
                            q.append((nr, nc))
            components.append(comp)
            
    if not components:
        return grid
        
    # Main box is the component with the largest number of pixels
    main_comp = max(components, key=len)
    
    # Bounding box of main component
    min_r = min(p[0] for p in main_comp)
    max_r = max(p[0] for p in main_comp)
    min_c = min(p[1] for p in main_comp)
    max_c = max(p[1] for p in main_comp)
    
    out_grid = []
    for r in range(min_r, max_r + 1):
        row = []
        for c in range(min_c, max_c + 1):
            row.append(grid[r][c])
        out_grid.append(row)
        
    out_R = len(out_grid)
    out_C = len(out_grid[0])
    
    # Process small shapes
    for comp in components:
        if comp == main_comp:
            continue
            
        c_min_r = min(p[0] for p in comp)
        c_max_r = max(p[0] for p in comp)
        c_min_c = min(p[1] for p in comp)
        c_max_c = max(p[1] for p in comp)
        
        H = c_max_r - c_min_r + 1
        W = c_max_c - c_min_c + 1
        
        # Find the color of this shape (non-2 and non-0)
        colors = set(grid[r][c] for r, c in comp if grid[r][c] != 2)
        if not colors:
            continue
        color = list(colors)[0]
        
        # Create mask for this shape
        mask = []
        for r in range(c_min_r, c_max_r + 1):
            row = []
            for c in range(c_min_c, c_max_c + 1):
                row.append(grid[r][c] == color)
            mask.append(row)
            
        # Slide window over out_grid to find matching 0s
        for r in range(out_R - H + 1):
            for c in range(out_C - W + 1):
                match = True
                for i in range(H):
                    for j in range(W):
                        is_zero = (out_grid[r+i][c+j] == 0)
                        if is_zero != mask[i][j]:
                            match = False
                            break
                    if not match:
                        break
                
                if match:
                    for i in range(H):
                        for j in range(W):
                            if mask[i][j]:
                                out_grid[r+i][c+j] = color

    # Fill any remaining 0s with 2s just in case, though they should be filled
    for r in range(out_R):
        for c in range(out_C):
            if out_grid[r][c] == 0:
                out_grid[r][c] = 2

    return out_grid
