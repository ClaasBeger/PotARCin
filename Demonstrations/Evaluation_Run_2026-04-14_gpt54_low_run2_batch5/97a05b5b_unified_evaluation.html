2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    H, W = len(grid), len(grid[0])

    def neighbors(r, c):
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < H and 0 <= nc < W:
                yield nr, nc

    def get_components_nonzero():
        seen = [[False] * W for _ in range(H)]
        comps = []
        for r in range(H):
            for c in range(W):
                if grid[r][c] != 0 and not seen[r][c]:
                    stack = [(r, c)]
                    seen[r][c] = True
                    cells = []
                    while stack:
                        cr, cc = stack.pop()
                        cells.append((cr, cc))
                        for nr, nc in neighbors(cr, cc):
                            if grid[nr][nc] != 0 and not seen[nr][nc]:
                                seen[nr][nc] = True
                                stack.append((nr, nc))
                    comps.append(cells)
        return comps

    def bbox(cells):
        rs = [r for r, _ in cells]
        cs = [c for _, c in cells]
        return min(rs), max(rs), min(cs), max(cs)

    def extract_matrix(cells):
        r0, r1, c0, c1 = bbox(cells)
        mat = [[0] * (c1 - c0 + 1) for _ in range(r1 - r0 + 1)]
        for r, c in cells:
            mat[r - r0][c - c0] = grid[r][c]
        return mat, (r0, r1, c0, c1)

    def rot90(mat):
        return [list(row) for row in zip(*mat[::-1])]

    def flip_h(mat):
        return [row[::-1] for row in mat]

    def transforms(mat):
        out = []
        cur = mat
        for _ in range(4):
            out.append(cur)
            out.append(flip_h(cur))
            cur = rot90(cur)
        uniq = []
        seen = set()
        for m in out:
            key = tuple(tuple(row) for row in m)
            if key not in seen:
                seen.add(key)
                uniq.append(m)
        return uniq

    def twos_shape(mat):
        coords = set()
        for r in range(len(mat)):
            for c in range(len(mat[0])):
                if mat[r][c] == 2:
                    coords.add((r, c))
        return coords

    # Find the main large component of 2s
    comps = get_components_nonzero()
    main = None
    main_size = -1
    for comp in comps:
        colors = {grid[r][c] for r, c in comp}
        if colors == {2} and len(comp) > main_size:
            main = comp
            main_size = len(comp)

    r0, r1, c0, c1 = bbox(main)
    out_h, out_w = r1 - r0 + 1, c1 - c0 + 1

    # Start with solid 2 rectangle
    out = [[2 for _ in range(out_w)] for _ in range(out_h)]

    # Find 0-hole components inside the main bbox
    seen0 = [[False] * out_w for _ in range(out_h)]
    holes = []
    for rr in range(out_h):
        for cc in range(out_w):
            gr, gc = r0 + rr, c0 + cc
            if grid[gr][gc] == 0 and not seen0[rr][cc]:
                stack = [(rr, cc)]
                seen0[rr][cc] = True
                cells = []
                while stack:
                    cr, cc2 = stack.pop()
                    cells.append((cr, cc2))
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = cr + dr, cc2 + dc
                        if 0 <= nr < out_h and 0 <= nc < out_w:
                            if grid[r0 + nr][c0 + nc] == 0 and not seen0[nr][nc]:
                                seen0[nr][nc] = True
                                stack.append((nr, nc))
                holes.append(cells)

    # Pattern components are all non-main nonzero components containing a non-2 color
    pattern_mats = []
    main_set = set(main)
    for comp in comps:
        if set(comp) == main_set:
            continue
        colors = {grid[r][c] for r, c in comp}
        if any(v != 2 for v in colors):
            mat, _ = extract_matrix(comp)
            pattern_mats.append(mat)

    used = [False] * len(pattern_mats)

    for hole in holes:
        hrs = [r for r, _ in hole]
        hcs = [c for _, c in hole]
        hr0, hr1, hc0, hc1 = min(hrs), max(hrs), min(hcs), max(hcs)
        h_h, h_w = hr1 - hr0 + 1, hc1 - hc0 + 1
        hole_shape = {(r - hr0, c - hc0) for r, c in hole}

        placed = False
        for i, pm in enumerate(pattern_mats):
            if used[i]:
                continue
            for tm in transforms(pm):
                if len(tm) == h_h and len(tm[0]) == h_w and twos_shape(tm) == hole_shape:
                    for rr in range(h_h):
                        for cc in range(h_w):
                            if tm[rr][cc] != 0:
                                out[hr0 + rr][hc0 + cc] = tm[rr][cc]
                    used[i] = True
                    placed = True
                    break
            if placed:
                break

    return out
