2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    n = len(grid)
    m = len(grid[0])

    # Determine which corner contains the seed pattern
    nz = [(r, c) for r in range(n) for c in range(m) if grid[r][c] != 0]
    if not nz:
        return [row[:] for row in grid]

    min_r = min(r for r, c in nz)
    max_r = max(r for r, c in nz)
    min_c = min(c for r, c in nz)
    max_c = max(c for r, c in nz)

    top = min_r <= (n - 1 - max_r)
    left = min_c <= (m - 1 - max_c)

    def to_tl(r, c):
        if not top:
            r = n - 1 - r
        if not left:
            c = m - 1 - c
        return r, c

    def from_tl(r, c):
        if not top:
            r = n - 1 - r
        if not left:
            c = m - 1 - c
        return r, c

    norm = [[0] * m for _ in range(n)]
    for r in range(n):
        for c in range(m):
            if grid[r][c] != 0:
                rr, cc = to_tl(r, c)
                norm[rr][cc] = grid[r][c]

    norm_nz = [(r, c, norm[r][c]) for r in range(n) for c in range(m) if norm[r][c] != 0]

    # Outer corner = closest nonzero to top-left
    ra, ca, a = min(norm_nz, key=lambda x: (x[0] + x[1], x[0], x[1]))

    # Inner corner = closest nonzero strictly down-right from outer corner
    dr_candidates = [(r, c, v) for r, c, v in norm_nz if r > ra and c > ca]
    rd, cd, d = min(dr_candidates, key=lambda x: ((x[0] - ra) + (x[1] - ca), x[0], x[1]))

    # Edge color = repeated color nearest the corner (ignore outer/inner corner colors first)
    color_positions = {}
    for r, c, v in norm_nz:
        color_positions.setdefault(v, []).append((r, c))

    edge_color = None
    best_score = None
    for v, pts in color_positions.items():
        if len(pts) >= 2:
            pts_sorted = sorted(pts, key=lambda p: (p[0] + p[1], p[0], p[1]))
            score = pts_sorted[0][0] + pts_sorted[0][1]
            if best_score is None or score < best_score:
                best_score = score
                edge_color = v

    # Pick the two nearest edge-color cells around the corner motif
    edge_pts_all = sorted(color_positions.get(edge_color, []), key=lambda p: (p[0] + p[1], p[0], p[1]))
    edge_pts = []
    for p in edge_pts_all:
        if p != (ra, ca) and p != (rd, cd):
            edge_pts.append(p)
        if len(edge_pts) == 2:
            break

    # Fallbacks if needed
    if len(edge_pts) < 2:
        others = [(r, c) for r, c, v in norm_nz if v == edge_color]
        edge_pts = sorted(others, key=lambda p: (p[0] + p[1], p[0], p[1]))[:2]

    # Determine edge row/col and starting points
    er1, ec1 = edge_pts[0]
    er2, ec2 = edge_pts[1]

    hrow = min(er1, er2)
    vcol = min(ec1, ec2)
    hstart = max(ec1, ec2)
    vstart = max(er1, er2)

    out = [[0] * m for _ in range(n)]

    def set_sym(r, c, v):
        for rr in {r, n - 1 - r}:
            for cc in {c, m - 1 - c}:
                out[rr][cc] = v

    # Place outer and inner corners symmetrically
    set_sym(ra, ca, a)
    set_sym(rd, cd, d)

    # Draw horizontal edges
    for c in range(hstart, m - hstart, 2):
        set_sym(hrow, c, edge_color)

    # Draw vertical edges
    for r in range(vstart, n - vstart, 2):
        set_sym(r, vcol, edge_color)

    # Mirror all remaining interior objects
    motif_cells = {
        (ra, ca),
        (rd, cd),
        (er1, ec1),
        (er2, ec2),
    }

    for r, c, v in norm_nz:
        if (r, c) in motif_cells:
            continue
        set_sym(r, c, v)

    # Map back to original orientation
    final = [[0] * m for _ in range(n)]
    for r in range(n):
        for c in range(m):
            rr, cc = from_tl(r, c)
            final[rr][cc] = out[r][c]

    return final
