2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    H = len(grid)
    W = len(grid[0]) if H else 0

    # Find corner markers (color 4) and crop to their bounding box.
    fours = [(r, c) for r in range(H) for c in range(W) if grid[r][c] == 4]
    if not fours:
        return [row[:] for row in grid]

    rmin = min(r for r, _ in fours)
    rmax = max(r for r, _ in fours)
    cmin = min(c for _, c in fours)
    cmax = max(c for _, c in fours)

    out_h = rmax - rmin + 1
    out_w = cmax - cmin + 1
    out = [[0 for _ in range(out_w)] for _ in range(out_h)]

    # Preserve the 4s inside the cropped area.
    for r in range(rmin, rmax + 1):
        for c in range(cmin, cmax + 1):
            if grid[r][c] == 4:
                out[r - rmin][c - cmin] = 4

    # Find large rectangular components inside the crop (excluding 0 and 4).
    visited = [[False] * W for _ in range(H)]
    rects = []

    def bfs(sr, sc, color):
        q = [(sr, sc)]
        visited[sr][sc] = True
        cells = [(sr, sc)]
        qi = 0
        while qi < len(q):
            r, c = q[qi]
            qi += 1
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = r + dr, c + dc
                if (
                    rmin <= nr <= rmax and cmin <= nc <= cmax and
                    not visited[nr][nc] and grid[nr][nc] == color
                ):
                    visited[nr][nc] = True
                    q.append((nr, nc))
                    cells.append((nr, nc))
        return cells

    for r in range(rmin, rmax + 1):
        for c in range(cmin, cmax + 1):
            val = grid[r][c]
            if val not in (0, 4) and not visited[r][c]:
                cells = bfs(r, c, val)
                rs = [x for x, _ in cells]
                cs = [y for _, y in cells]
                rr0, rr1 = min(rs), max(rs)
                cc0, cc1 = min(cs), max(cs)
                h = rr1 - rr0 + 1
                w = cc1 - cc0 + 1
                # Keep solid rectangles only.
                if len(cells) == h * w:
                    rects.append((val, rr0 - rmin, cc0 - cmin, h, w))

    if not rects:
        return out

    # All examples use a common block size.
    block_h = min(h for _, _, _, h, _ in rects)
    block_w = min(w for _, _, _, _, w in rects)

    # Placement origin is the minimum top-left among the large blocks.
    origin_r = min(r for _, r, _, _, _ in rects)
    origin_c = min(c for _, _, c, _, _ in rects)

    # Collect the small pattern outside the crop (excluding 0 and 4).
    pattern_cells = []
    for r in range(H):
        for c in range(W):
            if rmin <= r <= rmax and cmin <= c <= cmax:
                continue
            val = grid[r][c]
            if val not in (0, 4):
                pattern_cells.append((r, c, val))

    if not pattern_cells:
        return out

    prmin = min(r for r, _, _ in pattern_cells)
    prmax = max(r for r, _, _ in pattern_cells)
    pcmin = min(c for _, c, _ in pattern_cells)
    pcmax = max(c for _, c, _ in pattern_cells)

    pat_h = prmax - prmin + 1
    pat_w = pcmax - pcmin + 1
    pattern = [[0 for _ in range(pat_w)] for _ in range(pat_h)]
    for r, c, val in pattern_cells:
        pattern[r - prmin][c - pcmin] = val

    # Enlarge the small pattern into the output using the detected block size.
    for pr in range(pat_h):
        for pc in range(pat_w):
            val = pattern[pr][pc]
            if val == 0:
                continue
            top = origin_r + pr * block_h
            left = origin_c + pc * block_w
            for dr in range(block_h):
                for dc in range(block_w):
                    rr = top + dr
                    cc = left + dc
                    if 0 <= rr < out_h and 0 <= cc < out_w:
                        out[rr][cc] = val

    return out
