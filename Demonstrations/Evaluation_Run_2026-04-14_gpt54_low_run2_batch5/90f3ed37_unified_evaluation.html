2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0]) if h else 0
    out = [row[:] for row in grid]

    def is_blank_row(r):
        return all(v == 0 for v in grid[r])

    # Split into vertically separated nonblank objects
    objects = []
    r = 0
    while r < h:
        if is_blank_row(r):
            r += 1
            continue
        s = r
        rows = []
        while r < h and not is_blank_row(r):
            rows.append(grid[r][:])
            r += 1
        objects.append((s, rows))

    if not objects:
        return out

    # Exemplar = densest object
    exemplar_start, exemplar_rows = max(objects, key=lambda x: sum(v != 0 for row in x[1] for v in row))
    H = len(exemplar_rows)

    # Class 1: single-row horizontal completion
    if H == 1:
        for s, rows in objects:
            row = rows[0]
            nz = [c for c, v in enumerate(row) if v == 8]
            if not nz:
                continue
            m = max(nz)
            for c in range(m + 1, w):
                if out[s][c] == 0:
                    out[s][c] = 1
        return out

    # Class 2: 2-row checkerboard completion
    if H == 2:
        ex_sets = []
        ok_checker = True
        for rr in exemplar_rows:
            cols = [c for c, v in enumerate(rr) if v == 8]
            if not cols:
                ok_checker = False
                break
            parity = cols[0] % 2
            if any(c % 2 != parity for c in cols):
                ok_checker = False
                break
            if any(cols[i + 1] - cols[i] != 2 for i in range(len(cols) - 1)):
                ok_checker = False
                break
            ex_sets.append((parity, max(cols)))
        if ok_checker and ex_sets[0][0] != ex_sets[1][0]:
            for s, rows in objects:
                for i in range(2):
                    cols = [c for c, v in enumerate(rows[i]) if v == 8] if i < len(rows) else []
                    parity = ex_sets[i][0]
                    last = max(cols) if cols else parity - 2
                    for c in range(last + 2, w, 2):
                        rr = s + i
                        if 0 <= rr < h and out[rr][c] == 0:
                            out[rr][c] = 1
            return out

    # Class 3: threshold-based contiguous pattern over multiple rows
    ops = ["<", "<=", "=", ">=", ">"]

    def relation_set(op, c):
        if op == "<":
            return set(range(0, max(0, c)))
        if op == "<=":
            return set(range(0, min(w - 1, c) + 1)) if c >= 0 else set()
        if op == "=":
            return {c} if 0 <= c < w else set()
        if op == ">=":
            return set(range(max(0, c), w)) if c < w else set()
        if op == ">":
            return set(range(min(w, c + 1), w))
        return set()

    # Infer exemplar row relations to a cut c
    exemplar_pattern = None
    for c in range(-1, w + 1):
        row_ops = []
        good = True
        for rr in exemplar_rows:
            s8 = {i for i, v in enumerate(rr) if v == 8}
            matched = None
            for op in ops:
                if s8 == relation_set(op, c):
                    matched = op
                    break
            if matched is None:
                good = False
                break
            row_ops.append(matched)
        if good:
            exemplar_pattern = row_ops
            break

    # Fallback: if no threshold pattern found, return unchanged
    if exemplar_pattern is None:
        return out

    for s, rows in objects:
        best_c = None
        best_score = None

        for c in range(-1, w + 1):
            valid = True
            add = 0
            exact_nonempty = 0

            for i in range(H):
                observed = {j for j, v in enumerate(rows[i]) if v == 8} if i < len(rows) else set()
                target = relation_set(exemplar_pattern[i], c)
                if not observed.issubset(target):
                    valid = False
                    break
                add += len(target) - len(observed)
                if observed and observed == target:
                    exact_nonempty += 1

            if valid:
                score = (add, -exact_nonempty)
                if best_score is None or score < best_score:
                    best_score = score
                    best_c = c

        if best_c is None:
            continue

        for i in range(H):
            rr = s + i
            if rr >= h:
                break
            target = relation_set(exemplar_pattern[i], best_c)
            for c in target:
                if out[rr][c] == 0:
                    out[rr][c] = 1

    return out
