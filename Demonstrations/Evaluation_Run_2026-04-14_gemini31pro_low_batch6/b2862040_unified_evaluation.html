2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    # Find background 9s (those connected to the grid boundary)
    background = set()
    q = []
    for r in range(rows):
        for c in range(cols):
            if (r == 0 or r == rows - 1 or c == 0 or c == cols - 1) and grid[r][c] == 9:
                background.add((r, c))
                q.append((r, c))
                
    while q:
        r, c = q.pop(0)
        for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
            nr, nc = r + dr, c + dc
            if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == 9 and (nr, nc) not in background:
                background.add((nr, nc))
                q.append((nr, nc))
                
    enclosed_9s = set()
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 9 and (r, c) not in background:
                enclosed_9s.add((r, c))
                
    # Find 8-connected components of 1s
    visited_1s = set()
    components = []
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 1 and (r, c) not in visited_1s:
                comp = []
                q_comp = [(r, c)]
                visited_1s.add((r, c))
                while q_comp:
                    curr_r, curr_c = q_comp.pop(0)
                    comp.append((curr_r, curr_c))
                    for dr in [-1, 0, 1]:
                        for dc in [-1, 0, 1]:
                            if dr == 0 and dc == 0:
                                continue
                            nr, nc = curr_r + dr, curr_c + dc
                            if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == 1 and (nr, nc) not in visited_1s:
                                visited_1s.add((nr, nc))
                                q_comp.append((nr, nc))
                components.append(comp)
                
    output = [row[:] for row in grid]
    
    # Change components to 8 if they enclose any 9s
    for comp in components:
        is_enclosing = False
        for r, c in comp:
            for dr, dc in [(-1, 0), (1, 0), (0, -1), (0, 1)]:
                nr, nc = r + dr, c + dc
                if (nr, nc) in enclosed_9s:
                    is_enclosing = True
                    break
            if is_enclosing:
                break
        
        if is_enclosing:
            for r, c in comp:
                output[r][c] = 8
                
    return output
