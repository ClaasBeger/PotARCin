2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    rows, cols = len(grid), len(grid[0])
    out = [row[:] for row in grid]
    
    # Find rectangular blocks of non-zero values
    visited = [[False]*cols for _ in range(rows)]
    
    # Find connected components of non-zero cells
    def flood(r, c):
        stack = [(r,c)]
        cells = []
        while stack:
            cr, cc = stack.pop()
            if 0<=cr<rows and 0<=cc<cols and not visited[cr][cc] and grid[cr][cc]!=0:
                visited[cr][cc]=True
                cells.append((cr,cc))
                for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                    stack.append((cr+dr,cc+dc))
        return cells
    
    components = []
    for r in range(rows):
        for c in range(cols):
            if grid[r][c]!=0 and not visited[r][c]:
                cells = flood(r,c)
                components.append(cells)
    
    for cells in components:
        rs = [r for r,c in cells]
        cs = [c for r,c in cells]
        r1,r2,c1,c2 = min(rs),max(rs),min(cs),max(cs)
        
        # Identify border and inner colors
        border_color = grid[r1][c1]
        # Inner region
        ir1,ir2,ic1,ic2 = r1+1,r2-1,c1+1,c2-1
        inner_color = grid[ir1][ic1]
        
        iw = ic2-ic1+1  # inner width
        ih = ir2-ir1+1  # inner height
        
        # New outer ring of border_color, expanded by inner dimensions
        # The pattern: swap colors and grow
        # Original: border surrounds inner
        # Output: inner_color surrounds border (original), with new border outside
        
        nr1 = r1-ih
        nr2 = r2+ih
        nc1 = c1-iw
        nc2 = c2+iw
        
        # Clear original area first isn't needed since we overwrite
        
        # Draw new border ring (border_color)
        for r in range(nr1, nr2+1):
            for c in range(nc1, nc2+1):
                if 0<=r<rows and 0<=c<cols:
                    # Is it in the outer ring?
                    if r<r1 or r>r2 or c<c1 or c>c2:
                        out[r][c] = border_color
        
        # Original rectangle: swap colors
        for r in range(r1, r2+1):
            for c in range(c1, c2+1):
                if grid[r][c] == border_color:
                    out[r][c] = inner_color
                else:
                    out[r][c] = border_color
    
    return out
