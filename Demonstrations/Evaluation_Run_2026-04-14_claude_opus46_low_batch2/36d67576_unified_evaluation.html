2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    rows = len(grid)
    cols = len(grid[0])
    result = copy.deepcopy(grid)
    
    # Find all connected components of 4s (the "L" shapes)
    visited = [[False]*cols for _ in range(rows)]
    structures = []
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 4 and not visited[r][c]:
                # BFS to find connected 4s
                queue = [(r,c)]
                visited[r][c] = True
                cells = []
                while queue:
                    cr,cc = queue.pop(0)
                    cells.append((cr,cc))
                    for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                        nr,nc = cr+dr, cc+dc
                        if 0<=nr<rows and 0<=nc<cols and not visited[nr][nc] and grid[nr][nc]==4:
                            visited[nr][nc]=True
                            queue.append((nr,nc))
                structures.append(cells)
    
    # For each structure, find adjacent non-zero, non-4 values (the pattern)
    # A "complete" structure has 2 and other values; incomplete has only 2 or nothing
    
    def get_neighbors(cells):
        cell_set = set(cells)
        neighbors = {}
        for r,c in cells:
            for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                nr,nc = r+dr, c+dc
                if (nr,nc) not in cell_set and 0<=nr<rows and 0<=nc<cols and grid[nr][nc]!=0:
                    neighbors[(nr,nc)] = grid[nr][nc]
        return neighbors
    
    # Find template (structure with most non-4 neighbors including values beyond just 2)
    struct_info = []
    for cells in structures:
        nbrs = get_neighbors(cells)
        vals = set(nbrs.values())
        struct_info.append((cells, nbrs, vals))
    
    # Template: has 1, 2, 3 neighbors. Incomplete: missing some.
    templates = [s for s in struct_info if len(s[2] - {2}) >= 2]
    incompletes = [s for s in struct_info if len(s[2] - {2}) < 2]
    
    if not templates:
        templates = [max(struct_info, key=lambda s: len(s[1]))]
        incompletes = [s for s in struct_info if s is not templates[0]]
    
    template = templates[0]
    t_cells, t_nbrs, _ = template
    
    # For each incomplete structure, find rotation/reflection that maps template 4-cells to it
    # then apply the neighbor pattern
    from itertools import product
    
    def normalize(cells):
        mr = min(r for r,c in cells)
        mc = min(c for r,c in cells)
        return frozenset((r-mr,c-mc) for r,c in cells)
    
    def transform(cells, t):
        # t: 0-7 for rotations and reflections
        result = []
        for r,c in cells:
            for _ in range(t % 4):
                r,c = c,-r
            if t >= 4:
                r,c = r,-c
            result.append((r,c))
        return result
    
    t_base = [(r,c) for r,c in t_cells]
    t_mr = min(r for r,c in t_base)
    t_mc = min(c for r,c in t_base)
    t_rel = [(r-t_mr,c-t_mc) for r,c in t_base]
    
    t_nbr_rel = {(r-t_mr,c-t_mc): v for (r,c),v in t_nbrs.items()}
    
    for inc_cells, inc_nbrs, _ in incompletes:
        i_mr = min(r for r,c in inc_cells)
        i_mc = min(c for r,c in inc_cells)
        i_rel = [(r-i_mr,c-i_mc) for r,c in inc_cells]
        i_norm = frozenset(i_rel)
        
        for t in range(8):
            tr = transform(t_rel, t)
            mr2 = min(r for r,c in tr)
            mc2 = min(c for r,c in tr)
            tr_norm = frozenset((r-mr2,c-mc2) for r,c in tr)
            if tr_norm == i_norm:
                # Apply same transform to neighbors
                for (nr,nc), v in t_nbr_rel.items():
                    tn = transform([(nr,nc)], t)[0]
                    tn = (tn[0]-mr2+i_mr, tn[1]-mc2+i_mc)
                    if 0<=tn[0]<rows and 0<=tn[1]<cols and result[tn[0]][tn[1]]==0:
                        result[tn[0]][tn[1]] = v
                break
    
    return result
