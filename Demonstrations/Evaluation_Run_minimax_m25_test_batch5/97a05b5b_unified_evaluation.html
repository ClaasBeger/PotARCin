2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import Counter
    from copy import deepcopy

    # 1. background colour = most frequent value
    flat = [c for row in grid for c in row]
    bg = Counter(flat).most_common(1)[0][0]

    # 2. extract shapes
    shapes = []  # list of dicts: color, mask, area, rotated, symmetric
    rows = len(grid)
    cols = len(grid[0])
    # find all non‑background cells
    for r in range(rows):
        for c in range(cols):
            col = grid[r][c]
            if col != bg:
                # see if we already have this colour
                existing = next((s for s in shapes if s['color'] == col), None)
                if existing is None:
                    # start a new shape
                    shape = {'color': col, 'cells': [(r,c)], 'area':0}
                    shapes.append(shape)
                else:
                    existing['cells'].append((r,c))

    # compute masks and other data
    for sh in shapes:
        cells = sh['cells']
        sh['area'] = len(cells)
        min_r = min(r for r,c in cells)
        max_r = max(r for r,c in cells)
        min_c = min(c for r,c in cells)
        max_c = max(c for r,c in cells)
        h = max_r - min_r + 1
        w = max_c - min_c + 1
        mask = [[0]*w for _ in range(h)]
        for (r,c) in cells:
            mask[r-min_r][c-min_c] = 1
        sh['mask'] = mask
        sh['height'] = h
        sh['width'] = w

        # rotate 90° clockwise
        rh = w
        rw = h
        rot = [[0]*rh for _ in range(rw)]
        for i in range(h):
            for j in range(w):
                rot[j][h-1-i] = mask[i][j]
        sh['rotated'] = rot

        # check symmetry (rotated equals original)
        sym = (rot == mask)
        sh['symmetric'] = sym

    # 3. decide which shape goes to centre (if any)
    centre_shape = None
    other_shapes = []
    symmetric = [s for s in shapes if s['symmetric']]
    if symmetric:
        # take the largest symmetric shape
        centre_shape = max(symmetric, key=lambda s: (s['area'], s['color']))
        other_shapes = [s for s in shapes if s is not centre_shape]
    else:
        other_shapes = shapes

    # 4. sort remaining shapes by descending area, then descending colour
    other_shapes.sort(key=lambda s: (-s['area'], -s['color']))

    # 5. assign to corners
    # cell positions in a 3×3 grid
    # 0 = top‑left, 1 = top‑right, 2 = bottom‑left, 3 = bottom‑right
    corners = ['TL','TR','BL','BR']
    # special case for two shapes: place them opposite
    placed = []
    n = len(other_shapes)
    if n == 0:
        pass
    elif n == 1:
        placed.append(('TL', other_shapes[0]))
    elif n == 2:
        placed.append(('TL', other_shapes[0]))
        placed.append(('BR', other_shapes[1]))
    elif n == 3:
        placed.append(('TL', other_shapes[0]))
        placed.append(('TR', other_shapes[1]))
        placed.append(('BL', other_shapes[2]))
    else:  # 4 or more – use first four
        placed.append(('TL', other_shapes[0]))
        placed.append(('TR', other_shapes[1]))
        placed.append(('BL', other_shapes[2]))
        placed.append(('BR', other_shapes[3]))

    # map names to (row, col) in the 3×3 cell grid
    pos = {
        'TL': (0,0),
        'TR': (0,2),
        'BL': (2,0),
        'BR': (2,2),
        'C' : (1,1)
    }

    # 6. compute row heights and column widths (empty cells get size 1)
    row_heights = [1,1,1]
    col_widths  = [1,1,1]

    # centre shape
    if centre_shape:
        r,c = pos['C']
        row_heights[r] = max(row_heights[r], centre_shape['height'])
        col_widths[c] = max(col_widths[c], centre_shape['width'])

    # corner shapes
    for corner, sh in placed:
        r,c = pos[corner]
        row_heights[r] = max(row_heights[r], sh['height'])
        col_widths[c]  = max(col_widths[c],  sh['width'])

    # 7. place shapes
    # total size (including one‑cell border at top and left)
    tot_h = 1 + sum(row_heights)
    tot_w = 1 + sum(col_widths)

    # initialise grid with background colour
    out = [[bg]*tot_w for _ in range(tot_h)]

    # helper: cumulative sums
    row_start = [0]*3
    col_start = [0]*3
    row_start[0] = 1
    col_start[0] = 1
    for i in range(1,3):
        row_start[i] = row_start[i-1] + row_heights[i-1]
        col_start[i] = col_start[i-1] + col_widths[i-1]

    # write centre shape
    if centre_shape:
        r,c = pos['C']
        rs = row_start[r]
        cs = col_start[c]
        mat = centre_shape['rotated']
        for i in range(centre_shape['height']):
            for j in range(centre_shape['width']):
                if mat[i][j]:
                    out[rs+i][cs+j] = centre_shape['color']

    # write corner shapes
    for corner, sh in placed:
        r,c = pos[corner]
        rs = row_start[r]
        cs = col_start[c]
        mat = sh['rotated']
        for i in range(sh['height']):
            for j in range(sh['width']):
                if mat[i][j]:
                    out[rs+i][cs+j] = sh['color']

    # 8. trim trailing rows/columns that contain only background
    # find last row with a non‑background cell
    last_r = tot_h-1
    while last_r >= 0 and all(out[last_r][c]==bg for c in range(tot_w)):
        last_r -= 1
    last_c = tot_w-1
    while last_c >= 0 and all(out[r][last_c]==bg for r in range(tot_h)):
        last_c -= 1
    # slice to include only used part (border row/col kept if needed)
    final = [row[:last_c+1] for row in out[:last_r+1]]

    return final
