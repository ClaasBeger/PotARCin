Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    n = len(grid)
    # Find separator color and positions
    sep_color = grid[0][5]
    
    # Find horizontal and vertical separator lines
    h_seps = [i for i in range(n) if all(grid[i][j] == sep_color for j in range(n))]
    v_seps = [j for j in range(n) if all(grid[i][j] == sep_color for i in range(n))]
    
    # Define cell boundaries
    def get_regions(seps, size):
        bounds = [-1] + seps + [size]
        regions = []
        for i in range(len(bounds)-1):
            r0, r1 = bounds[i]+1, bounds[i+1]
            if r0 < r1:
                regions.append((r0, r1))
        return regions
    
    row_regions = get_regions(h_seps, n)
    col_regions = get_regions(v_seps, n)
    
    # Extract pattern from each cell (non-zero, non-sep values)
    other_color = None
    for i in range(n):
        for j in range(n):
            if grid[i][j] not in (0, sep_color):
                other_color = grid[i][j]
                break
        if other_color: break
    
    # For each cell, find the pattern of other_color pixels
    # Find the "most complete" pattern (largest count) - that's the template
    # Then for each cell, place sep_color where template has values but cell doesn't
    
    # Collect all patterns relative to their cell
    cells = {}
    for ri, (r0, r1) in enumerate(row_regions):
        for ci, (c0, c1) in enumerate(col_regions):
            pattern = set()
            for i in range(r0, r1):
                for j in range(c0, c1):
                    if grid[i][j] == other_color:
                        pattern.add((i - r0, j - c0))
            cells[(ri, ci)] = (r0, r1, c0, c1, pattern)
    
    # Find the template (cell with most other_color pixels)
    max_cell = max(cells.values(), key=lambda x: len(x[4]))
    template = max_cell[4]
    
    # Build output
    out = [row[:] for row in grid]
    
    for (ri, ci), (r0, r1, c0, c1, pattern) in cells.items():
        if len(pattern) == 0:
            continue
        # Fill template positions with sep_color where cell doesn't have other_color
        for (dr, dc) in template:
            i, j = r0 + dr, c0 + dc
            if i < r1 and j < c1:
                if (dr, dc) not in pattern:
                    out[i][j] = sep_color
    
    # Also for empty cells: if any cell in same row-region or col-region has content,
    # we need to check... Actually looking at examples more carefully:
    # Every cell gets the template filled with sep_color, except the original pixels stay.
    
    # Re-examine: empty cells also get filled if they share row/col with non-empty cells?
    # In example 1: cell (0,1) was empty, got template in sep_color. Cell(0,2) had partial, got mix.
    # It seems ALL cells with 0 pattern also get the template in sep_color.
    # Wait, cell (2,0) in ex1 row2,col0 was empty and got filled with sep_color template.
    
    # So: every cell gets template in sep_color, overlaid with its own other_color pixels

            i, j = r0 + dr, c0 + dc
            if i < r1 and j < c1:
                if (dr, dc) not in pattern:
                    out[i][j] = sep_color
    
    return out
