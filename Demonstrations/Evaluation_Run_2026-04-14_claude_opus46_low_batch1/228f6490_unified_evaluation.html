2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import numpy as np
    g = np.array(grid)
    rows, cols = g.shape
    
    # Find rectangles of 5s
    visited = np.zeros_like(g, dtype=bool)
    rects = []
    
    for r in range(rows):
        for c in range(cols):
            if g[r,c] == 5 and not visited[r,c]:
                # BFS to find connected 5s
                from collections import deque
                q = deque([(r,c)])
                visited[r,c] = True
                cells = [(r,c)]
                while q:
                    cr,cc = q.popleft()
                    for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                        nr,nc = cr+dr, cc+dc
                        if 0<=nr<rows and 0<=nc<cols and not visited[nr,nc] and g[nr,nc] == 5:
                            visited[nr,nc] = True
                            q.append((nr,nc))
                            cells.append((nr,nc))
                min_r = min(x[0] for x in cells)
                max_r = max(x[0] for x in cells)
                min_c = min(x[1] for x in cells)
                max_c = max(x[1] for x in cells)
                rects.append((min_r, min_c, max_r, max_c))
    
    # For each rect, find the hole (0s inside) and find matching shape outside
    result = g.copy()
    
    for (r1,c1,r2,c2) in rects:
        # Find hole cells (0s inside the rect)
        hole = []
        for r in range(r1,r2+1):
            for c in range(c1,c2+1):
                if g[r,c] == 0:
                    hole.append((r,c))
        if not hole:
            continue
        
        # Hole shape relative positions
        hr0 = min(h[0] for h in hole)
        hc0 = min(h[1] for h in hole)
        hole_rel = set((r-hr0, c-hc0) for r,c in hole)
        hole_h = max(r for r,c in hole_rel)+1
        hole_w = max(c for r,c in hole_rel)+1
        
        # Find non-zero, non-5 connected components outside all rects
        all_rect_cells = set()
        for (a1,b1,a2,b2) in rects:
            for r in range(a1,a2+1):
                for c in range(b1,b2+1):
                    all_rect_cells.add((r,c))
        
        # Find colored blobs outside rects
        visited2 = np.zeros_like(g, dtype=bool)
        for (a1,b1,a2,b2) in rects:
            for r in range(a1,a2+1):
                for c in range(b1,b2+1):
                    visited2[r,c] = True
        
        for r in range(rows):
            for c in range(cols):
                if visited2[r,c] or g[r,c] == 0 or g[r,c] == 5:
                    continue
                color = g[r,c]
                q = deque([(r,c)])
                visited2[r,c] = True
                cells = [(r,c)]
                while q:
                    cr,cc = q.popleft()
                    for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                        nr,nc = cr+dr, cc+dc
                        if 0<=nr<rows and 0<=nc<cols and not visited2[nr,nc] and g[nr,nc]==color and (nr,nc) not in all_rect_cells:
                            visited2[nr,nc] = True
                            q.append((nr,nc))
                            cells.append((nr,nc))
                
                br0 = min(x[0] for x in cells)
                bc0 = min(x[1] for x in cells)
                blob_rel = set((x[0]-br0, x[1]-bc0) for x in cells)
                
                if blob_rel == hole_rel:
                    # Fill hole with this color, clear blob
                    for hr,hc in hole:
                        result[hr,hc] = color
                    for cr2,cc2 in cells:
                        result[cr2,cc2] = 0
                    break
    
    return result.tolist()
