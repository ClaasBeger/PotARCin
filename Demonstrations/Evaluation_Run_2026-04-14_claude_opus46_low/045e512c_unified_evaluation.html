2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import numpy as np
    g = np.array(grid)
    rows, cols = g.shape
    
    # Find all non-zero cells and group by color
    from collections import defaultdict
    color_cells = defaultdict(list)
    for r in range(rows):
        for c in range(cols):
            if g[r][c] != 0:
                color_cells[g[r][c]].append((r, c))
    
    # Main shape is the color with most cells
    main_color = max(color_cells, key=lambda k: len(color_cells[k]))
    main_cells = color_cells[main_color]
    mr1, mc1 = min(r for r,c in main_cells), min(c for r,c in main_cells)
    mr2, mc2 = max(r for r,c in main_cells), max(c for r,c in main_cells)
    
    # Extract main shape pattern (relative, binary)
    sh = mr2 - mr1 + 1
    sw = mc2 - mc1 + 1
    pattern = np.zeros((sh, sw), dtype=int)
    for r, c in main_cells:
        pattern[r - mr1][c - mc1] = 1
    
    main_center_r = (mr1 + mr2) / 2
    main_center_c = (mc1 + mc2) / 2
    
    result = g.copy()
    
    for color, cells in color_cells.items():
        if color == main_color:
            continue
        count = len(cells)
        cr = sum(r for r, c in cells) / count
        cc = sum(c for r, c in cells) / count
        
        dr = cr - main_center_r
        dc = cc - main_center_c
        
        # Determine direction: normalize to unit steps in shape+gap increments
        step_r = 0
        step_c = 0
        if abs(dr) > abs(dc):
            step_r = (sh + 1) * (1 if dr > 0 else -1)
        elif abs(dc) > abs(dr):
            step_c = (sw + 1) * (1 if dc > 0 else -1)
        else:
            step_r = (sh + 1) * (1 if dr > 0 else -1)
            step_c = (sw + 1) * (1 if dc > 0 else -1)
        
        for i in range(1, count + 1):
            for pr in range(sh):
                for pc in range(sw):
                    if pattern[pr][pc]:
                        nr = mr1 + pr + step_r * i
                        nc = mc1 + pc + step_c * i
                        if 0 <= nr < rows and 0 <= nc < cols:
                            result[nr][nc] = color
    
    return result.tolist()
