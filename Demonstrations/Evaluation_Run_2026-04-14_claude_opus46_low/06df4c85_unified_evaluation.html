2. Definition

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    rows = len(grid)
    cols = len(grid[0])
    
    # Find separator color (most frequent)
    sep = grid[2][0]  # The separator row
    
    # Cell size is 2, separator width is 1, period is 3
    # Cell positions: cell (r,c) starts at pixel (r*3, c*3) and is 2x2
    
    n_cell_rows = (rows + 1) // 3
    n_cell_cols = (cols + 1) // 3
    
    # Read cell values
    cell_vals = {}
    for cr in range(n_cell_rows):
        for cc in range(n_cell_cols):
            pr, pc = cr * 3, cc * 3
            val = grid[pr][pc]
            if val != 0 and val != sep:
                cell_vals[(cr, cc)] = val
    
    # Group by color
    from collections import defaultdict
    color_cells = defaultdict(list)
    for (cr, cc), val in cell_vals.items():
        color_cells[val].append((cr, cc))
    
    result = copy.deepcopy(grid)
    
    # For each color, find pairs in same row or column and fill between
    for color, cells in color_cells.items():
        # Same row pairs
        row_groups = defaultdict(list)
        for cr, cc in cells:
            row_groups[cr].append(cc)
        for cr, ccs in row_groups.items():
            if len(ccs) >= 2:
                min_cc = min(ccs)
                max_cc = max(ccs)
                for cc in range(min_cc, max_cc + 1):
                    pr, pc = cr * 3, cc * 3
                    for dr in range(2):
                        for dc in range(2):
                            if pr + dr < rows and pc + dc < cols:
                                if result[pr + dr][pc + dc] == 0:
                                    result[pr + dr][pc + dc] = color
        
        # Same column pairs
        col_groups = defaultdict(list)
        for cr, cc in cells:
            col_groups[cc].append(cr)
        for cc, crs in col_groups.items():
            if len(crs) >= 2:
                min_cr = min(crs)
                max_cr = max(crs)
                for cr in range(min_cr, max_cr + 1):
                    pr, pc = cr * 3, cc * 3
                    for dr in range(2):
                        for dc in range(2):
                            if pr + dr < rows and pc + dc < cols:
                                if result[pr + dr][pc + dc] == 0:
                                    result[pr + dr][pc + dc] = color
    
    return result
