Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import numpy as np
    
    grid = [row[:] for row in grid]
    rows = len(grid)
    cols = len(grid[0])
    
    # Find the 8-object (2x2 block) and 2-object positions
    eight_cells = []
    two_cells = []
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 8:
                eight_cells.append((r, c))
            elif grid[r][c] == 2:
                two_cells.append((r, c))
    
    if not eight_cells or not two_cells:
        return grid
    
    # Bounding box of 8-object
    eight_min_r = min(r for r, c in eight_cells)
    eight_max_r = max(r for r, c in eight_cells)
    eight_min_c = min(c for r, c in eight_cells)
    eight_max_c = max(c for r, c in eight_cells)
    
    # Bounding box of 2-object
    two_min_r = min(r for r, c in two_cells)
    two_max_r = max(r for r, c in two_cells)
    two_min_c = min(c for r, c in two_cells)
    two_max_c = max(c for r, c in two_cells)
    
    # Center of 8-object
    eight_center_r = (eight_min_r + eight_max_r) / 2
    eight_center_c = (eight_min_c + eight_max_c) / 2
    
    # Center of 2-object
    two_center_r = (two_min_r + two_max_r) / 2
    two_center_c = (two_min_c + two_max_c) / 2
    
    # Direction from 2-object to 8-object
    dr = eight_center_r - two_center_r
    dc = eight_center_c - two_center_c
    
    # We need to move the 2-object toward the 8-object until it's adjacent
    # Try shifting the 2-object by increments in the direction of the 8-object
    
    # Extract the 2-shape relative to its bounding box top-left
    two_shape = set()
    for r, c in two_cells:
        two_shape.add((r - two_min_r, c - two_min_c))
    
    # Determine the direction to move (normalize to unit steps)
    # We'll try moving in the primary direction
    def sign(x):
        if x > 0: return 1
        if x < 0: return -1
        return 0
    
    step_r = sign(dr)
    step_c = sign(dc)
    
    # Try moving the 2-object step by step until it's adjacent to the 8-object
    # (i.e., one more step would cause overlap)
    
    best_offset_r = 0
    best_offset_c = 0
    
    for step in range(1, max(rows, cols) + 1):
        offset_r = step * step_r
        offset_c = step * step_c
        
        # Check if placing 2-object at this offset causes overlap with 8
        new_two_cells = set()
        for r, c in two_cells:
            new_two_cells.add((r + offset_r, c + offset_c))
        
        overlap = new_two_cells & set(eight_cells)
        
        # Check if any cell goes out of bounds
        out_of_bounds = any(r < 0 or r >= rows or c < 0 or c >= cols for r, c in new_two_cells)
        
        if overlap or out_of_bounds:
            break
        
        best_offset_r = offset_r
        best_offset_c = offset_c
    
    # Clear old 2-object positions
    result = [row[:] for row in grid]
    for r, c in two_cells:
        result[r][c] = 0
    
    # Place 2-object at new position
    for r, c in two_cells:
        nr, nc = r + best_offset_r, c + best_offset_c
        result[nr][nc] = 2
    
    return result
