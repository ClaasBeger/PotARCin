2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import deque

    rows = len(grid)
    cols = len(grid[0]) if rows else 0

    # locate all 4 cells
    four_pos = [(r, c) for r in range(rows) for c in range(cols) if grid[r][c] == 4]
    if not four_pos:
        return grid

    # global centroid of 4 cells
    avg_r = sum(r for r, c in four_pos) / len(four_pos)
    avg_c = sum(c for r, c in four_pos) / len(four_pos)

    # ---------- connected components of 4 ----------
    visited = set()
    components = []
    for r, c in four_pos:
        if (r, c) in visited:
            continue
        q = deque([(r, c)])
        comp = set()
        while q:
            x, y = q.popleft()
            if (x, y) in visited:
                continue
            visited.add((x, y))
            comp.add((x, y))
            for dx, dy in [(-1,0),(1,0),(0,-1),(0,1)]:
                nx, ny = x+dx, y+dy
                if 0 <= nx < rows and 0 <= ny < cols and grid[nx][ny] == 4 and (nx, ny) not in visited:
                    q.append((nx, ny))
    components.append(comp)

    # global set of 4 cells
    global_4 = set(four_pos)

    # ---------- find corner cells (degree 2 with both horizontal and vertical neighbours) ----------
    def is_corner(cell, comp):
        r, c = cell
        hor = any((r, c+1) in comp or (r, c-1) in comp)
        ver = any((r+1, c) in comp or (r-1, c) in comp)
        return hor and ver

    corners = {cell for comp in components for cell in comp if is_corner(cell, comp)}

    # ---------- helper to extract straight segments ----------
    def extract_segments(comp):
        # horizontal segments
        hor_seg = {}
        for r, c in comp:
            hor_seg.setdefault(r, []).append(c)
        segments = []
        for r, clist in hor_seg.items():
            clist.sort()
            # split into runs of consecutive columns
            start = clist[0]
            prev = start
            for c in clist[1:]:
                if c == prev + 1:
                    prev = c
                else:
                    # finish current run
                    seg_cells = [(r, col) for col in range(start, prev+1)]
                    segments.append((seg_cells, 'horizontal', r, start))
                    start = c
                    prev = c
            seg_cells = [(r, col) for col in range(start, prev+1)]
            segments.append((seg_cells, 'horizontal', r, start))

        # vertical segments
        ver_seg = {}
        for r, c in comp:
            ver_seg.setdefault(c, []).append(r)
        for c, rlist in ver_seg.items():
            rlist.sort()
            start = rlist[0]
            prev = start
            for r in rlist[1:]:
                if r == prev + 1:
                    prev = r
                else:
                    seg_cells = [(row, c) for row in range(start, prev+1)]
                    segments.append((seg_cells, 'vertical', c, start))
                    start = r
                    prev = r
            seg_cells = [(row, c) for row in range(start, prev+1)]
            segments.append((seg_cells, 'vertical', c, start))
        return segments

    # ---------- process each component ----------
    for comp in components:
        segments = extract_segments(comp)
        # map from cell to whether it is a corner
        for seg_cells, orientation, fixed, start in segments:
            # set of coordinates in this segment
            seg_set = set(seg_cells)
            L = len(seg_cells)

            # interior direction
            if orientation == 'horizontal':
                # interior is down if centroid below the row
                interior_down = avg_r > fixed
                dir_v = 1 if interior_down else -1
                # define interior and exterior neighbour functions
                def interior_neigh(r, c):
                    return (r + dir_v, c)
                def exterior_neigh(r, c):
                    return (r - dir_v, c)
            else:  # vertical
                interior_right = avg_c > fixed
                dir_h = 1 if interior_right else -1
                def interior_neigh(r, c):
                    return (r, c + dir_h)
                def exterior_neigh(r, c):
                    return (r, c - dir_h)

            # Determine whether segment has a corner endpoint
            corner_endpoint = None
            if seg_cells[0] in corners:
                corner_endpoint = 'first'
            elif seg_cells[-1] in corners:
                corner_endpoint = 'last'

            # iterate over offsets from -(L-1) to (L-1)
            for offset in range(-(L-1), L):
                # coordinate on the line
                if orientation == 'horizontal':
                    cur_r = fixed
                    cur_c = start + offset
                else:
                    cur_r = start + offset
                    cur_c = fixed
                # ensure inside grid
                if not (0 <= cur_r < rows and 0 <= cur_c < cols):
                    continue

                # decide which neighbour to use
                # special handling for corner endpoint
                if corner_endpoint == 'first' and offset == 0:
                    # corner cell: place 1 if possible
                    target = interior_neigh(cur_r, cur_c)
                    if 0 <= target[0] < rows and 0 <= target[1] < cols and grid[target[0]][target[1]] == 0:
                        # also check adjacency to other 4s not in this segment
                        blocked = False
                        for d in [(-1,0),(1,0),(0,-1),(0,1)]:
                            nr, nc = target[0]+d[0], target[1]+d[1]
                            if (nr, nc) in global_4 and (nr, nc) not in seg_set:
                                blocked = True
                                break
                        if not blocked:
                            grid[target[0]][target[1]] = 1
                            continue
                    # fallback to exterior side
                    target = exterior_neigh(cur_r, cur_c)
                elif corner_endpoint == 'last' and offset == L-1:
                    # far end of branch: place 3
                    target = interior_neigh(cur_r, cur_c)
                    if 0 <= target[0] < rows and 0 <= target[1] < cols and grid[target[0]][target[1]] == 0:
                        blocked = False
                        for d in [(-1,0),(1,0),(0,-1),(0,1)]:
                            nr, nc = target[0]+d[0], target[1]+d[1]
                            if (nr, nc) in global_4 and (nr, nc) not in seg_set:
                                blocked = True
                                break
                        if not blocked:
                            grid[target[0]][target[1]] = 3
                            continue
                    # cannot place, skip
                    continue
                else:
                    # normal parity rule
                    if offset % 2 == 0:
                        # even offset -> 3 on interior side
                        target = interior_neigh(cur_r, cur_c)
                        color = 3
                    else:
                        # odd offset -> 1 on exterior side
                        target = exterior_neigh(cur_r, cur_c)
                        color = 1
                    # check bounds and emptiness
                    if 0 <= target[0] < rows and 0 <= target[1] < cols and grid[target[0]][target[1]] == 0:
                        # check adjacency to other 4s not in this segment
                        blocked = False
                        for d in [(-1,0),(1,0),(0,-1),(0,1)]:
                            nr, nc = target[0]+d[0], target[1]+d[1]
                            if (nr, nc) in global_4 and (nr, nc) not in seg_set:
                                blocked = True
                                break
                        if not blocked:
                            grid[target[0]][target[1]] = color

    return grid
