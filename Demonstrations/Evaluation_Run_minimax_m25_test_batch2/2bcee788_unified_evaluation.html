2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0]) if rows else 0

    # collect distinct non‑zero colours
    colours = set()
    for r in range(rows):
        for v in grid[r]:
            if v != 0:
                colours.add(v)

    # if the grid is empty (all zeros) just fill with 3
    if not colours:
        return [[3] * cols for _ in range(rows)]

    # count frequencies to find the dominant colour
    from collections import Counter
    cnt = Counter(v for row in grid for v in row if v != 0)
    primary = max(cnt, key=cnt.get)

    # gather positions of primary and of all other non‑zero cells (secondary)
    primary_pos = []
    secondary_pos = []
    for r in range(rows):
        for c in range(cols):
            val = grid[r][c]
            if val == primary:
                primary_pos.append((r, c))
            elif val != 0:
                secondary_pos.append((r, c))

    # bounding boxes for primary and secondary
    def bbox(pos):
        if not pos:
            return None, None, None, None
        min_r = min(r for r, _ in pos)
        max_r = max(r for r, _ in pos)
        min_c = min(c for _, c in pos)
        max_c = max(c for _, c in pos)
        return min_r, max_r, min_c, max_c

    pr = bbox(primary_pos)
    sr = bbox(secondary_pos)

    # decide orientation and axis of symmetry
    axis_type = None
    axis_val = None
    if secondary_pos:
        min_r_p, max_r_p, min_c_p, max_c_p = pr
        min_r_s, max_r_s, min_c_s, max_c_s = sr

        # secondary to the left of primary
        if max_c_s < min_c_p:
            axis_type = 'vertical'
            axis_val = (min_c_p + max_c_s) / 2.0
        # secondary to the right of primary
        elif min_c_s > max_c_p:
            axis_type = 'vertical'
            axis_val = (max_c_p + min_c_s) / 2.0
        # secondary above primary
        elif max_r_s < min_r_p:
            axis_type = 'horizontal'
            axis_val = (min_r_p + max_r_s) / 2.0
        # secondary below primary
        elif min_r_s > max_r_p:
            axis_type = 'horizontal'
            axis_val = (max_r_p + min_r_s) / 2.0

    # build the set of cells that will become the primary colour
    filled = set()
    for r, c in primary_pos:
        filled.add((r, c))
    for r, c in secondary_pos:
        filled.add((r, c))

    # reflect the shape across the axis, if an axis was identified
    if axis_type == 'vertical':
        ax = axis_val
        for r, c in list(filled):
            mc = int(2 * ax - c)
            if 0 <= mc < cols:
                filled.add((r, mc))
    elif axis_type == 'horizontal':
        ay = axis_val
        for r, c in list(filled):
            mr = int(2 * ay - r)
            if 0 <= mr < rows:
                filled.add((mr, c))

    # create output grid: background colour is 3 everywhere
    out = [[3] * cols for _ in range(rows)]
    for r, c in filled:
        out[r][c] = primary

    return out
