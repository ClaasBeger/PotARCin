def solve(grid):

    sys.setrecursionlimit(10000)
    rows = len(grid)
    cols = len(grid[0]) if rows else 0
    
    # find source (color that appears exactly twice and is not 2)
    from collections import Counter
    cnt = Counter()
    for r in range(rows):
        for c in range(cols):
            cnt[grid[r][c]] += 1
    
    target_color = 2
    source_color = None
    for col, k in cnt.items():
        if k == 2 and col != target_color:
            source_color = col
            break
    if source_color is None:
        # fallback: assume 3
        source_color = 3
    
    # positions of source and target
    source_pos = [(r, c) for r in range(rows) for c in range(cols) if grid[r][c] == source_color]

    target_set = set(target_pos)
    
    # adjacency to target (including diagonal)
    def is_adjacent_to_target(r, c):
        for dr in (-1, 0, 1):
            for dc in (-1, 0, 1):
                if dr == 0 and dc == 0:
                    continue

            path.append((r, c))
