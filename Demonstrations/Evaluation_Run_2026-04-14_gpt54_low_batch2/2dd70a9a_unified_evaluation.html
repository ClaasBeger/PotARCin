Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import deque

    h = len(grid)
    w = len(grid[0])

    def find_component(color):
        cells = [(r, c) for r in range(h) for c in range(w) if grid[r][c] == color]
        return cells

    cells2 = find_component(2)
    cells3 = find_component(3)

    if not cells2 or not cells3:
        return [row[:] for row in grid]

    def orientation(cells):
        rows = {r for r, _ in cells}
        cols = {c for _, c in cells}
        if len(rows) == 1:
            return "h"
        return "v"

    def endpoints_and_dirs(cells, orient):
        if orient == "h":
            r = cells[0][0]
            minc = min(c for _, c in cells)
            maxc = max(c for _, c in cells)
            return [
                ((r, minc - 1), (0, -1)),
                ((r, maxc + 1), (0, 1)),
            ]
        else:
            c = cells[0][1]
            minr = min(r for r, _ in cells)
            maxr = max(r for r, _ in cells)
            return [
                ((minr - 1, c), (-1, 0)),
                ((maxr + 1, c), (1, 0)),
            ]

    def in_bounds(r, c):
        return 0 <= r < h and 0 <= c < w

    def build_ray(start, d):
        r, c = start
        dr, dc = d
        ray = []
        while in_bounds(r, c) and grid[r][c] == 0:
            ray.append((r, c))
            r += dr
            c += dc
        return ray

    def clear_segment(a, b):
        r1, c1 = a
        r2, c2 = b
        if r1 == r2:
            step = 1 if c2 >= c1 else -1
            for c in range(c1, c2 + step, step):
                if (r1, c) != a and (r1, c) != b and grid[r1][c] != 0:
                    return False
            return True
        if c1 == c2:
            step = 1 if r2 >= r1 else -1
            for r in range(r1, r2 + step, step):
                if (r, c1) != a and (r, c1) != b and grid[r][c1] != 0:
                    return False
            return True
        return False

    def segment_cells(a, b):
        r1, c1 = a
        r2, c2 = b
        out = []
        if r1 == r2:
            step = 1 if c2 >= c1 else -1
            for c in range(c1, c2 + step, step):
                out.append((r1, c))
        else:
            step = 1 if r2 >= r1 else -1
            for r in range(r1, r2 + step, step):
                out.append((r, c1))
        return out

    orient2 = orientation(cells2)
    orient3 = orientation(cells3)

    opts2 = endpoints_and_dirs(cells2, orient2)
    opts3 = endpoints_and_dirs(cells3, orient3)

    best = None

    for start3, dir3 in opts3:

        if not ray3:
            continue
        for start2, dir2 in opts2:
            ray2 = build_ray(start2, dir2)
            if not ray2:
                continue
            for i, a in enumerate(ray3):

                        cand = (score, ray3[:i + 1], ray2[:j + 1], segment_cells(a, b))
                        if best is None or cand[0] > best[0]:
                            best = cand

    if best is None:
        return out

    for r, c in set(part3) | set(part2) | set(connector):

    return out
