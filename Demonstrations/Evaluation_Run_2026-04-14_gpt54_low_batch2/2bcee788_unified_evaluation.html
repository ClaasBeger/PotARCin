2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0]) if h else 0

    # Find the main object's color (non-zero and not 2)
    obj_color = None
    obj_cells = []
    two_cells = []

    for r in range(h):
        for c in range(w):
            v = grid[r][c]
            if v == 2:
                two_cells.append((r, c))
            elif v != 0:
                obj_color = v
                obj_cells.append((r, c))

    if obj_color is None:
        return [[3 for _ in range(w)] for _ in range(h)]

    # Collect all object cells of that color
    if not obj_cells:
        for r in range(h):
            for c in range(w):
                if grid[r][c] == obj_color:
                    obj_cells.append((r, c))

    # Determine orientation of the 2-line
    rows2 = {r for r, c in two_cells}
    cols2 = {c for r, c in two_cells}

    orientation = None  # 'vertical' mirror => reflect columns, 'horizontal' mirror => reflect rows

    if len(two_cells) >= 2:
        if len(cols2) == 1:
            orientation = 'vertical'
        elif len(rows2) == 1:
            orientation = 'horizontal'
    else:
        # Single 2-cell: infer from adjacency to the object
        r2, c2 = two_cells[0]
        for dr, dc, ori in [(-1, 0, 'horizontal'), (1, 0, 'horizontal'),
                            (0, -1, 'vertical'), (0, 1, 'vertical')]:
            rr, cc = r2 + dr, c2 + dc
            if 0 <= rr < h and 0 <= cc < w and grid[rr][cc] == obj_color:
                orientation = ori
                break

    # Fallback if still ambiguous
    if orientation is None:
        if len(cols2) == 1:
            orientation = 'vertical'
        else:
            orientation = 'horizontal'

    out = [[3 for _ in range(w)] for _ in range(h)]

    # Paint original object and 2-cells
    for r, c in obj_cells:
        out[r][c] = obj_color
    for r, c in two_cells:
        out[r][c] = obj_color

    # Reflect the object across the boundary between the object and the 2-line
    if orientation == 'vertical':
        line_c = two_cells[0][1]
        obj_cols = [c for r, c in obj_cells]
        if obj_cols and max(obj_cols) < line_c:
            # object is left of the 2-line; axis at line_c - 0.5
            for r, c in obj_cells:
                mc = 2 * line_c - 1 - c
                if 0 <= mc < w:
                    out[r][mc] = obj_color
        else:
            # object is right of the 2-line; axis at line_c + 0.5
            for r, c in obj_cells:
                mc = 2 * line_c + 1 - c
                if 0 <= mc < w:
                    out[r][mc] = obj_color
    else:
        line_r = two_cells[0][0]
        obj_rows = [r for r, c in obj_cells]
        if obj_rows and max(obj_rows) < line_r:
            # object is above the 2-line; axis at line_r - 0.5
            for r, c in obj_cells:
                mr = 2 * line_r - 1 - r
                if 0 <= mr < h:
                    out[mr][c] = obj_color
        else:
            # object is below the 2-line; axis at line_r + 0.5
            for r, c in obj_cells:
                mr = 2 * line_r + 1 - r
                if 0 <= mr < h:
                    out[mr][c] = obj_color

    return out
