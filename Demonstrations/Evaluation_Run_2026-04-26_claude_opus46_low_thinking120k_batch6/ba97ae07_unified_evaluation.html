2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    rows, cols = len(grid), len(grid[0])
    out = copy.deepcopy(grid)
    
    # Find vertical band columns (same non-zero value in most rows)
    # Find horizontal band rows (same non-zero value in most cols)
    
    # Detect horizontal bands: rows where all/most cells share a non-zero value
    h_rows = {}
    for r in range(rows):
        vals = [grid[r][c] for c in range(cols)]
        non_zero = [v for v in vals if v != 0]
        if non_zero:
            from collections import Counter
            cnt = Counter(non_zero)
            val, freq = cnt.most_common(1)[0]
            if freq > cols // 2:
                h_rows[r] = val
    
    # Detect vertical bands: columns where all/most cells share a non-zero value
    v_cols = {}
    for c in range(cols):
        vals = [grid[r][c] for r in range(rows)]
        non_zero = [v for v in vals if v != 0]
        if non_zero:
            from collections import Counter
            cnt = Counter(non_zero)
            val, freq = cnt.most_common(1)[0]
            if freq > rows // 2:
                v_cols[c] = val
    
    # Count extent: vertical band spans more rows, horizontal spans more cols
    # At intersection, the longer band wins
    v_count = len([r for r in range(rows) if r not in h_rows])  # rows the vertical band owns
    h_count = len([c for c in range(cols) if c not in v_cols])
    
    # Determine which is "longer" by comparing their span relative to grid
    # Vertical band: number of rows it spans vs horizontal band rows
    # Horizontal band: number of cols it spans vs vertical band cols
    
    v_span = rows  # vertical goes full height
    h_span = cols  # horizontal goes full width
    v_band_size = len(v_cols)
    h_band_size = len(h_rows)
    
    # The one with more cells outside intersection dominates
    v_non_intersection = (rows - h_band_size) * v_band_size
    h_non_intersection = (cols - v_band_size) * h_band_size
    
    for r in h_rows:
        for c in v_cols:
            if v_non_intersection >= h_non_intersection:
                out[r][c] = v_cols[c]
            else:
                out[r][c] = h_rows[r]
    
    return out
