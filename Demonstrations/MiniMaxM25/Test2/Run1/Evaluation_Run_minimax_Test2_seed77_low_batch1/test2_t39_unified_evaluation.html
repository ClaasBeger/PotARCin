2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

from collections import defaultdict

def solve(grid):
    # ------------------------------------------------------------
    # 1. find border colour (the most frequent non‑zero value)
    freq = defaultdict(int)
    for row in grid:
        for v in row:
            if v != 0:
                freq[v] += 1
    border = max(freq, key=freq.get)          # e.g. 3, 1, 7 …
    # ------------------------------------------------------------
    # 2. locate all 3×3 blocks (top‑left corners)
    rows = len(grid)
    cols = len(grid[0])
    block_top = {}            # (i,j) -> (r,c) top‑left of block
    block_index = [[-1]*cols for _ in range(rows)]
    for r in range(rows-2):
        for c in range(cols-2):
            if (grid[r][c] == border and grid[r][c+1] == border and grid[r][c+2] == border and
                grid[r+1][c] == border and grid[r+1][c+2] == border and
                grid[r+2][c] == border and grid[r+2][c+1] == border and grid[r+2][c+2] == border):
                # we have a block
                # decide its (i,j) by its position among discovered rows/cols
                # gather unique row starts and column starts
                pass

    # Instead of complicated ordering we simply collect all blocks and then sort them
    raw_blocks = []
    for r in range(rows-2):
        for c in range(cols-2):
            if (grid[r][c] == border and grid[r][c+1] == border and grid[r][c+2] == border and
                grid[r+1][c] == border and grid[r+1][c+2] == border and
                grid[r+2][c] == border and grid[r+2][c+1] == border and grid[r+2][c+2] == border):
                raw_blocks.append((r,c))

    # order by row, then column
    raw_blocks.sort()
    # map row start -> block row index
    rowStarts = sorted({r for r,_ in raw_blocks})
    colStarts = sorted({c for _,c in raw_blocks})
    row_to_i = {r:i for i,r in enumerate(rowStarts)}
    col_to_j = {c:j for j,c in enumerate(colStarts)}
    for r,c in raw_blocks:
        i = row_to_i[r]
        j = col_to_j[c]
        block_top[(i,j)] = (r,c)
        for dr in range(3):
            for dc in range(3):
                block_index[r+dr][c+dc] = (i,j)

    # ------------------------------------------------------------
    # 3. locate the unique 4 and 8
    pos4 = pos8 = None
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 4:
                pos4 = (r,c)
            elif grid[r][c] == 8:
                pos8 = (r,c)

    # find block indices of these cells
    block4 = block_index[pos4[0]][pos4[1]]
    block8 = block_index[pos8[0]][pos8[1]]
    i4, j4 = block4
    i8, j8 = block8

    # ------------------------------------------------------------
    # 4. helper to compute edge cell for a move
    def edge_coord(i,j,move):
        r0,c0 = block_top[(i,j)]
        if move == 'U':
            return r0, c0+1
        if move == 'D':
            return r0+2, c0+1
        if move == 'L':
            return r0+1, c0
        if move == 'R':
            return r0+1, c0+2
        raise ValueError('bad move')

    # ------------------------------------------------------------
    # 5. neighbours of a block
    def neighbours(i,j):
        for di,dj in ((-1,0),(1,0),(0,-1),(0,1)):
            ni, nj = i+di, j+dj
            if (ni,nj) in block_top:
                yield ni,nj

    # ------------------------------------------------------------
    # 6. explore all paths that start at a block in start_set
    #    and finish at any block in target_set.
    #    For each move we count whether the crossed edge contains a 2.
    best_path = None
    best_score = -1

    # two possible orientations
    # orientation A: start = block4 or its neighbours, target = neighbours of block8
    start_set_A = {(i4,j4)} | set(neighbours(i4,j4))
    target_set_A = set(neighbours(i8,j8))
    forbid_A = (i8,j8)          # must not step onto the block that actually holds the 8

    # orientation B: start = neighbours of block8, target = block4 or its neighbours
    start_set_B = set(neighbours(i8,j8))
    target_set_B = {(i4,j4)} | set(neighbours(i4,j4))
    forbid_B = (i4,j4)

    for orientation in ['A','B']:
        if orientation == 'A':
            starts = start_set_A
            targets = target_set_A
            forbid = forbid_A
        else:
            starts = start_set_B
            targets = target_set_B
            forbid = forbid_B

        for si,sj in starts:
            # simple BFS/DFS to find best path to any target
            stack = [(si,sj, [], 0, {(si,sj)})]   # i,j,moves,score,visited
            while stack:
                ci,cj,moves,score,visited = stack.pop()
                if (ci,cj) in targets:
                    if score > best_score or (score == best_score and (best_path is None or moves < best_path)):
                        best_score = score
                        best_path = moves
                    continue
                # continue searching
                for move, (di,dj) in zip(('U','D','L','R'),((-1,0),(1,0),(0,-1),(0,1))):
                    ni, nj = ci+di, cj+dj
                    if (ni,nj) in block_top and (ni,nj) not in visited and (ni,nj) != forbid:
                        er, ec = edge_coord(ci,cj,move)
                        add = 1 if grid[er][ec] == 2 else 0
                        new_vis = visited | {(ni,nj)}
                        stack.append((ni,nj, moves+[move], score+add, new_vis))

    # ------------------------------------------------------------
    # 7. apply the modifications of the best path
    out = [row[:] for row in grid]
    if best_path:
        # we need the block we start from – it is the start set element that gave the best path.
        # The DFS explored all starts, but we didn't keep which start produced the final path.
        # However we can reconstruct the start block by replaying the path from each possible start
        # and checking which yields the same move list and score.
        # Since the grid is small we simply try all starts again.
        for orientation in ['A','B']:
            if orientation == 'A':
                starts = start_set_A
            else:
                starts = start_set_B
            for si,sj in starts:
                # simulate the path
                cur_i, cur_j = si, sj
                cur_score = 0
                for move in best_path:
                    er, ec = edge_coord(cur_i,cur_j,move)
                    if grid[er][ec] == 2:
                        cur_score += 1
                    di,dj = {'U':(-1,0),'D':(1,0),'L':(0,-1),'R':(0,1)}[move]
                    cur_i, cur_j = cur_i+di, cur_j+dj
                if cur_score == best_score:
                    # this is the correct start, apply the edges now
                    cur_i, cur_j = si, sj
                    for move in best_path:
                        er, ec = edge_coord(cur_i,cur_j,move)
                        if grid[er][ec] == 2:
                            out[er][ec] = 5
                        di,dj = {'U':(-1,0),'D':(1,0),'L':(0,-1),'R':(0,1)}[move]
                        cur_i, cur_j = cur_i+di, cur_j+dj
                    break
            else:
                continue
            break

    # ------------------------------------------------------------
    # 8. change the unique numbers: 8 -> 4, 4 -> 0
    out[pos8[0]][pos8[1]] = 4
    out[pos4[0]][pos4[1]] = 0

    return out
