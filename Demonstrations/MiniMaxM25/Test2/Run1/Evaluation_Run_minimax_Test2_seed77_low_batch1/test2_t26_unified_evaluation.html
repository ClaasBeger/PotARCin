2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    """
    Transform the input grid according to the rule:
    - The colour 8 defines an axis of symmetry.
    - All other colours are mirrored across that axis (or across both
      horizontal and vertical axes when the 8's form a cross).
    - Existing cells are kept, missing symmetric cells are added.
    """
    from collections import Counter

    # copy original grid (list of lists)
    n = len(grid)
    m = len(grid[0]) if n else 0
    out = [row[:] for row in grid]

    # collect positions of colour 8
    eights = [(r, c) for r in range(n) for c in range(m) if grid[r][c] == 8]

    if not eights:
        # no axis defined – nothing to do
        return out

    # count 8's per row and per column
    row_cnt = Counter(r for r, _ in eights)
    col_cnt = Counter(c for _, c in eights)

    max_row = max(row_cnt.values(), default=0)
    max_col = max(col_cnt.values(), default=0)

    axes = []          # each element is (kind, param)
    # kind: 'horizontal', 'vertical', 'diag', 'anti'

    # primary orientation: horizontal if any row has at least two 8's
    if max_row >= 2:
        row0 = max(row_cnt, key=row_cnt.get)          # row with most 8's
        cols_in_row = [c for (r, c) in eights if r == row0]
        col0 = (min(cols_in_row) + max(cols_in_row)) // 2
        axes.append(('horizontal', row0))
        axes.append(('vertical', col0))
    # otherwise, vertical if any column has at least two 8's
    elif max_col >= 2:
        col0 = max(col_cnt, key=col_cnt.get)          # column with most 8's
        rows_in_col = [r for (r, c) in eights if c == col0]
        row0 = (min(rows_in_col) + max(rows_in_col)) // 2
        axes.append(('vertical', col0))
        axes.append(('horizontal', row0))
    else:
        # neither row nor column has >=2 8's – look for a diagonal
        diff_cnt = Counter(r - c for (r, c) in eights)
        sum_cnt = Counter(r + c for (r, c) in eights)

        max_diff = max(diff_cnt.values(), default=0)
        max_sum = max(sum_cnt.values(), default=0)

        if max_diff >= 2:
            diff0 = max(diff_cnt, key=diff_cnt.get)
            axes.append(('diag', diff0))
        elif max_sum >= 2:
            sum0 = max(sum_cnt, key=sum_cnt.get)
            axes.append(('anti', sum0))
        else:
            # no recognisable axis – nothing to change
            return out

    # helper: reflect a single point according to a given axis
    def reflect(r, c, kind, param):
        if kind == 'horizontal':
            r2 = 2 * param - r
            c2 = c
        elif kind == 'vertical':
            r2 = r
            c2 = 2 * param - c
        elif kind == 'diag':
            diff0 = param
            r2 = c + diff0
            c2 = r - diff0
        else:               # anti‑diagonal
            sum0 = param
            r2 = sum0 - c
            c2 = sum0 - r
        return r2, c2

    # collect all original non‑zero cells (including 8 – they are not added elsewhere)
    original = [(r, c, grid[r][c]) for r in range(n) for c in range(m) if grid[r][c] != 0]

    # for each original cell, generate its mirrors
    for r, c, val in original:
        # skip colour 8 – we never create new 8's
        if val == 8:
            continue

        # mirrors across each single axis
        for kind, param in axes:
            r2, c2 = reflect(r, c, kind, param)
            if 0 <= r2 < n and 0 <= c2 < m:
                if out[r2][c2] == 0:
                    out[r2][c2] = val

        # if we have two axes, also apply the composition (both axes)
        if len(axes) == 2:

            kind2, param2 = axes[1]
            r12, c12 = reflect(r, c, kind1, param1)
            r2, c2 = reflect(r12, c12, kind2, param2)
            if 0 <= r2 < n and 0 <= c2 < m:
                if out[r2][c2] == 0:
                    out[r2][c2] = val

    return out
