def solve(grid):
    # ----------------------------------------------------------------------
    #  Determine the colour that has the smallest positive frequency
    #  (ignoring the background colour 0).  This is the "target" colour.
    #  All rows and columns that contain this colour are removed.
    #  The remaining sub‑grid is then split in the middle and the two halves
    #  are exchanged (left/right swap).  Finally all‑zero outer rows and
    #  columns are trimmed away (if they appear) and the result is returned.
    # ----------------------------------------------------------------------
    if not grid:
        return []

    # --------------------------------------------------------------
    # 1. frequency of each non‑zero value
    freq = {}
    for row in grid:
        for v in row:
            if v != 0:
                freq[v] = freq.get(v, 0) + 1

    # --------------------------------------------------------------
    # 2. choose the target colour
    #    prefer the smallest count that is larger than 1; if none exists
    #    take the smallest overall count (could be 1).
    target = None
    min_cnt = None
    for v, c in freq.items():
        if min_cnt is None or (c < min_cnt and (c > 1 or min_cnt <= 1)):
            min_cnt = c
            target = v
    # if all counts are 1 we still have a target (the smallest value)
    if target is None:
        target = min(freq, key=freq.get)

    # --------------------------------------------------------------
    # 3. find rows and columns that do NOT contain the target
    rows_to_keep = [i for i, row in enumerate(grid) if target not in row]
    cols_to_keep = [j for j in range(len(grid[0])) if any(grid[i][j] != target for i in range(len(grid)))]

    # Build the reduced grid
    new_grid = [[grid[i][j] for j in cols_to_keep] for i in rows_to_keep]

    # --------------------------------------------------------------
    # 4. left/right half swap (only if number of columns is even)
    if new_grid and len(new_grid[0]) % 2 == 0:
        half = len(new_grid[0]) // 2
        new_grid = [row[half:] + row[:half] for row in new_grid]

    # --------------------------------------------------------------
    # 5. trim all‑zero outer rows / columns (optional, makes the output cleaner)
    # remove leading all‑zero rows
    while new_grid and all(v == 0 for v in new_grid[0]):
        new_grid.pop(0)
    # trailing all‑zero rows
    while new_grid and all(v == 0 for v in new_grid[-1]):
        new_grid.pop()
    # leading all‑zero columns
    if new_grid and new_grid[0]:
        while new_grid and all(row[0] == 0 for row in new_grid):
            for r in range(len(new_grid)):
                new_grid[r] = new_grid[r][1:]

    if new_grid and new_grid[0]:
        while new_grid and all(row[-1] == 0 for row in new_grid):
            for r in range(len(new_grid)):
                new_grid[r] = new_grid[r][:-1]

    return new_grid
