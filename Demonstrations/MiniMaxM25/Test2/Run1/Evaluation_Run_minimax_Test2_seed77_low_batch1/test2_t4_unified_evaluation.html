def solve(grid):
    # Find the outer square of the border colour 2
    rows = len(grid)
    cols = len(grid[0]) if rows else 0
    min_r = rows
    max_r = -1
    min_c = cols
    max_c = -1
    for i in range(rows):
        for j in range(cols):
            if grid[i][j] == 2:
                if i < min_r: min_r = i
                if i > max_r: max_r = i
                if j < min_c: min_c = j
                if j > max_c: max_c = j
    # side length of the outer square
    side = max(max_r - min_r + 1, max_c - min_c + 1)
    N = side - 1                     # size of a block for a single‑cell interior colour
    # interior of the outer square
    interior_r0 = min_r + 1
    interior_c0 = min_c + 1
    interior_sz = side - 2           # original interior size (always 3 in the given tasks)

    # collect distinct interior colours and their positions
    colour_positions = {}
    for i in range(interior_sz):
        for j in range(interior_sz):
            v = grid[interior_r0 + i][interior_c0 + j]
            if v != 0:
                colour_positions.setdefault(v, []).append((i, j))

    # start with an empty grid
    out = [[0] * cols for _ in range(rows)]

    # process colours in the order they appear in the interior (scan row‑major)
    for colour, positions in colour_positions.items():
        cnt = len(positions)
        r_min = min(p[0] for p in positions)
        c_min = min(p[1] for p in positions)

        # decide block size
        if cnt == 1:
            bh = N
            bw = N
        else:
            # for more than one cell we enlarge the block
            factor = min(cnt, 2)
            bh = N * factor
            bw = N * factor

        # top‑left corner of the block
        if cnt == 1:
            row0 = r_min * N
            col0 = c_min * N
        else:
            row0 = r_min * N
            col0 = max(0, (c_min - 1) * N)

        # draw the border of the block
        for i in range(bh):
            for j in range(bw):
                if i == 0 or i == bh - 1 or j == 0 or j == bw - 1:
                    out[row0 + i][col0 + j] = colour

        # interior size
        ih = bh - 2
        iw = bw - 2

        # place the original border colour (2) inside the block
        for (ri, ci) in positions:
            # row position: keep the original row, clamped to the interior height
            ri2 = ri
            if ri2 > ih - 1:
                ri2 = ih - 1
            # column position: mirror horizontally, clamping if necessary

                ci2 = (iw - 1) - ci
            else:
                ci2 = 0
            out[row0 + 1 + ri2][col0 + 1 + ci2] = 2

    return out
