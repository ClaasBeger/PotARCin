2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    # Find the non‑zero colour (the shape colour)
    height = len(grid)
    width = len(grid[0]) if height else 0
    colour = None
    for r in range(height):
        for c in range(width):
            if grid[r][c] != 0:
                colour = grid[r][c]
                break
        if colour is not None:
            break
    if colour is None:                     # nothing to do
        return grid

    # bounding box of the shape
    min_r = height; max_r = -1
    min_c = width;  max_c = -1
    for r in range(height):
        for c in range(width):
            if grid[r][c] == colour:
                if r < min_r: min_r = r
                if r > max_r: max_r = r
                if c < min_c: min_c = c
                if c > max_c: max_c = c
    shape_h = max_r - min_r + 1
    shape_w = max_c - min_c + 1

    # Does the shape touch the right border?
    touches_right = (max_c == width - 1)

    # Does the shape contain a completely filled row inside its bounding box?
    has_full_row = False
    for r in range(min_r, max_r + 1):
        filled = 0
        for c in range(min_c, max_c + 1):
            if grid[r][c] == colour:
                filled += 1
        if filled == shape_w:
            has_full_row = True
            break

    # Find the first completely empty row either below or above the shape
    axis_row = None
    direction = None        # 'below' or 'above'
    for r in range(max_r + 1, height):
        if all(grid[r][c] == 0 for c in range(width)):
            axis_row = r
            direction = 'below'
            break
    if axis_row is None:
        for r in range(min_r - 1, -1, -1):
            if all(grid[r][c] == 0 for c in range(width)):
                axis_row = r
                direction = 'above'
                break

    # If there is no empty row we cannot reflect – return the original grid
    if axis_row is None:
        return grid

    # Determine the region where the original shape can be placed
    if direction == 'below':
        region_start = 0
        region_end   = axis_row - 1
    else:                     # direction == 'above'
        region_start = axis_row + 1
        region_end   = height - 1

    max_top = region_end - shape_h + 1
    if max_top < region_start:
        return grid               # shape does not fit – should not happen

    # Choose the placement that minimises the difference between the gaps
    # above and below the shape, preferring the lower position in case of a tie
    best_t = region_start
    best_diff = None
    for t in range(region_start, max_top + 1):
        gap_top    = t - region_start
        gap_bottom = region_end - (t + shape_h - 1)
        diff = abs(gap_top - gap_bottom)
        if best_diff is None or diff < best_diff or (diff == best_diff and t > best_t):
            best_diff = diff
            best_t = t

    # Shift the original shape to the chosen position
    offset = best_t - min_r
    out = [[0] * width for _ in range(height)]

    # place the (shifted) original shape
    for r in range(min_r, max_r + 1):
        for c in range(min_c, max_c + 1):
            if grid[r][c] == colour:
                out[r + offset][c] = colour

    # vertical reflection across the axis row
    for r in range(min_r, max_r + 1):
        for c in range(min_c, max_c + 1):
            if grid[r][c] == colour:
                r_mir = 2 * axis_row - r
                if 0 <= r_mir < height:
                    out[r_mir][c] = colour
                    # horizontal reflection of the vertically reflected copy
                    if touches_right and not has_full_row:
                        c_mir = width - 1 - c
                        out[r_mir][c_mir] = colour

    return out
