2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

from collections import deque
from typing import List

def solve(grid: List[List[int]]) -> List[List[int]]:
    n = len(grid)
    m = len(grid[0]) if n else 0

    # -------- find connected components for each colour ----------
    visited = [[False] * m for _ in range(n)]
    comps_by_colour = {}

    for i in range(n):
        for j in range(m):
            val = grid[i][j]
            if val == 0 or visited[i][j]:
                continue
            # BFS for this component
            q = deque()
            q.append((i, j))
            visited[i][j] = True
            cells = []
            while q:
                r, c = q.popleft()
                cells.append((r, c))
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < n and 0 <= nc < m:
                        if not visited[nr][nc] and grid[nr][nc] == val:
                            visited[nr][nc] = True
                            q.append((nr, nc))
            comps_by_colour.setdefault(val, []).append(cells)

    # -------- choose the colour that has exactly three components,
    #            preferring the one whose components touch the fewest borders ----------
    target = None
    best_border = None
    for colour, comps in comps_by_colour.items():
        if len(comps) != 3:
            continue
        border_cnt = 0
        for comp in comps:
            if any(r == 0 or r == n - 1 or c == 0 or c == m - 1 for r, c in comp):
                border_cnt += 1
        if best_border is None or border_cnt < best_border:
            best_border = border_cnt
            target = colour
        elif border_cnt == best_border and (target is None or colour < target):
            target = colour

    # if no colour with three components, return a zero grid (should not happen)
    if target is None:
        return [[0] * m for _ in range(n)]

    comps = comps_by_colour[target]

    # -------- compute bounding boxes for each component ----------
    boxes = []
    for comp in comps:
        rows = [r for r, _ in comp]
        cols = [c for _, c in comp]
        boxes.append((min(rows), max(rows), min(cols), max(cols)))  # (min_r, max_r, min_c, max_c)

    # identify leftmost / rightmost / topmost / bottommost components
    left_box = min(boxes, key=lambda b: b[2])
    right_box = max(boxes, key=lambda b: b[3])
    top_box = min(boxes, key=lambda b: b[0])
    bottom_box = max(boxes, key=lambda b: b[0])

    # middle component is the one that is neither top nor bottom
    middle_box = None
    for b in boxes:
        if b != top_box and b != bottom_box:
            middle_box = b
            break

    # -------- compute columns for the two vertical lines ----------
    left_col = (left_box[2] + left_box[3]) // 2
    right_col = (right_box[2] + right_box[3]) // 2
    if left_col > right_col:
        left_col, right_col = right_col, left_col

    # -------- compute row for the horizontal line ----------
    median_row = (top_box[0] + bottom_box[1]) // 2
    horiz_row = min(middle_box[0], median_row)

    # -------- build output grid ----------
    out = [[0] * m for _ in range(n)]

    # vertical lines: start one row below the top component and finish one row above the bottom component
    r_start = top_box[0] + 1
    r_end = bottom_box[0] - 1
    if r_start <= r_end:
        for r in range(r_start, r_end + 1):
            if 0 <= left_col < m:
                out[r][left_col] = 5
            if 0 <= right_col < m:
                out[r][right_col] = 5

    # horizontal line
    if 0 <= horiz_row < n:
        for c in range(left_col, right_col + 1):
            if 0 <= c < m:
                out[horiz_row][c] = 5

    return out
