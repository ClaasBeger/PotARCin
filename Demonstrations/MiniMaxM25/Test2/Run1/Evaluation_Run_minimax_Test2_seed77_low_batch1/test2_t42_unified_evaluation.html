2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0])
    # find block size k (largest divisor giving at least 2 blocks in each direction)
    k = None
    for cand in range(min(h, w), 1, -1):
        if h % cand == 0 and w % cand == 0:
            if h // cand >= 2 and w // cand >= 2:
                k = cand
                break
    if k is None:
        k = 1
    nb_rows = h // k
    nb_cols = w // k

    for br in range(nb_rows):
        for bc in range(nb_cols):
            r0 = br * k
            c0 = bc * k

            # collect distinct values in the block
            vals = set()
            for i in range(k):
                for j in range(k):
                    vals.add(grid[r0 + i][c0 + j])
            if len(vals) != 2:
                # not a two‑colour block – ignore
                continue
            val_a, val_b = tuple(vals)

            # original counts
            orig_cnt_a = sum(1 for i in range(k) for j in range(k) if grid[r0 + i][c0 + j] == val_a)
            orig_cnt_b = k * k - orig_cnt_a

            # ---- helper: compute mismatches and possible changes for a line ----
            def mismatches_and_changes(line_coords):
                """line_coords: list of (global_r, global_c) for the line, length k.
                Returns (d, changes) where d = number of mismatched pairs,
                changes = list of (global_r, global_c, new_val) to fix the line
                (using the later index in each pair)."""
                d = 0
                changes = []
                for i in range(k // 2):
                    r1, c1 = line_coords[i]
                    r2, c2 = line_coords[k - 1 - i]
                    v1 = grid[r1][c1]
                    v2 = grid[r2][c2]
                    if v1 != v2:
                        d += 1
                        # change the later position (the one with larger index) to match the earlier one
                        changes.append((r2, c2, v1))
                return d, changes

            # ----- main diagonal -----
            main_coords = [(r0 + i, c0 + i) for i in range(k)]
            d_main, ch_main = mismatches_and_changes(main_coords)

            # ----- anti diagonal (only for even sized blocks) -----
            d_anti = None
            ch_anti = None
            if k % 2 == 0:
                anti_coords = [(r0 + i, c0 + k - 1 - i) for i in range(k)]
                d_anti, ch_anti = mismatches_and_changes(anti_coords)

            # ----- columns -----
            best_col_d = None
            best_col_idx = -1
            best_col_ch = None
            for col in range(k):
                col_coords = [(r0 + i, c0 + col) for i in range(k)]
                d, ch = mismatches_and_changes(col_coords)
                if best_col_d is None or d < best_col_d or (d == best_col_d and col > best_col_idx):
                    best_col_d = d
                    best_col_idx = col
                    best_col_ch = ch

            # ----- rows -----
            best_row_d = None
            best_row_idx = -1
            best_row_ch = None
            for row in range(k):
                row_coords = [(r0 + row, c0 + j) for j in range(k)]
                d, ch = mismatches_and_changes(row_coords)
                if best_row_d is None or d < best_row_d or (d == best_row_d and row > best_row_idx):
                    best_row_d = d
                    best_row_idx = row
                    best_row_ch = ch

            # collect minimal d
            candidates = [('main', d_main, ch_main)]
            if d_anti is not None:
                candidates.append(('anti', d_anti, ch_anti))
            candidates.append(('col', best_col_d, best_col_ch))
            candidates.append(('row', best_row_d, best_row_ch))

            # ignore anti diagonal for odd sized blocks
            if k % 2 == 1:
                candidates = [c for c in candidates if c[0] != 'anti']

            min_d = min(d for _, d, _ in candidates)

            # odd blocks: apply only when min_d == 1
            if k % 2 == 1 and min_d > 1:
                continue

            # choose line according to priority: main -> anti -> col -> row (for even)
            # for odd: main -> col -> row
            if k % 2 == 0:
                order = ['main', 'anti', 'col', 'row']
            else:
                order = ['main', 'col', 'row']

            selected = None
            selected_changes = None
            for typ in order:
                for name, d, ch in candidates:
                    if name == typ and d == min_d:
                        selected = name
                        selected_changes = ch
                        break
                if selected is not None:
                    break

            # apply the chosen changes
            for (r, c, new_val) in selected_changes:
                grid[r][c] = new_val

            # recompute counts after the first set of changes
            new_cnt_a = sum(1 for i in range(k) for j in range(k) if grid[r0 + i][c0 + j] == val_a)
            new_cnt_b = k * k - new_cnt_a

            # for odd sized blocks: if counts changed, perform a second change to restore them
            if k % 2 == 1 and (new_cnt_a != orig_cnt_a):
                # decide which value increased
                if new_cnt_a > orig_cnt_a:
                    inc_val = val_a
                    dec_val = val_b
                else:
                    inc_val = val_b
                    dec_val = val_a
                # find a cell with inc_val (preferably not already changed)
                swapped = False
                for i in range(k):
                    for j in range(k):
                        if grid[r0 + i][c0 + j] == inc_val:
                            # avoid changing the same cell twice if possible
                            if (r0 + i, c0 + j) not in [(r, c) for (r, c, _) in selected_changes]:
                                grid[r0 + i][c0 + j] = dec_val
                                swapped = True
                                break
                    if swapped:
                        break
                # if for some reason we could not find a different cell, just change the first one we find
                if not swapped:
                    for i in range(k):
                        for j in range(k):
                            if grid[r0 + i][c0 + j] == inc_val:
                                grid[r0 + i][c0 + j] = dec_val
                                break
                        if not swapped:
                            break

    return grid
