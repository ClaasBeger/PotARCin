2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    """
    Transforms the input grid according to the discovered rule.

    The grid contains a vertical line of the value 3 that splits the grid into left
    and right halves. All other non‑zero values form disjoint blocks. Each block is
    shifted across the central line by a distance equal to (central_column_index + 1):
    left blocks are shifted right, right blocks are shifted left. A block moves only
    if all its target cells are empty (or become empty because another block moves
    away). The process is repeated until no further moves are possible, which also
    handles swaps.
    """
    from collections import deque

    # ---------------------------------------------------------------------
    # 1. Find the central column that consists entirely of the value 3
    # ---------------------------------------------------------------------
    nrows = len(grid)
    ncols = len(grid[0])

    central_col = -1
    for c in range(ncols):
        if all(grid[r][c] == 3 for r in range(nrows)):
            central_col = c
            break
    if central_col == -1:
        # No central line – return the input unchanged
        return [row[:] for row in grid]

    shift = central_col + 1            # distance to jump across the line

    # ---------------------------------------------------------------------
    # 2. Identify all connected components (blocks) made of the same colour,
    #    ignoring the central line (value 3) and zeros.
    # ---------------------------------------------------------------------
    visited = [[False] * ncols for _ in range(nrows)]
    components = []                     # each entry: {'color', 'cells', 'target', 'side', 'moved'}

    for r in range(nrows):
        for c in range(ncols):
            if visited[r][c]:
                continue
            val = grid[r][c]
            if val == 0 or val == 3:
                continue

            # start a new component
            q = deque()
            q.append((r, c))
            visited[r][c] = True
            cells = []

            while q:
                cr, cc = q.popleft()
                cells.append((cr, cc))
                for dr, dc in ((-1,0),(1,0),(0,-1),(0,1)):
                    nr, nc = cr+dr, cc+dc
                    if 0 <= nr < nrows and 0 <= nc < ncols:
                        if not visited[nr][nc] and grid[nr][nc] == val:
                            visited[nr][nc] = True
                            q.append((nr, nc))

            # determine whether the component is left or right of the central line
            cols = [c for (_, c) in cells]
            if max(cols) < central_col:
                side = 'left'
            elif min(cols) > central_col:
                side = 'right'
            else:
                # component touches the central line – ignore (should not happen)
                continue

            # compute the target positions after the jump
            if side == 'left':
                target = [(r, c + shift) for (r, c) in cells]
            else:   # side == 'right'
                target = [(r, c - shift) for (r, c) in cells]

            # discard targets that fall outside the grid
            if any(not (0 <= nc < ncols) for (_, nc) in target):
                continue

            components.append({
                'color': val,
                'cells': cells,
                'target': target,
                'side': side,
                'moved': False
            })

    # ---------------------------------------------------------------------
    # 3. Perform the movement iteratively: a block moves if all its target
    #    cells are currently empty. After each move the grid is updated,
    #    which may allow further moves (including swaps).
    # ---------------------------------------------------------------------
    # Working copy of the grid
    cur = [row[:] for row in grid]

    while True:
        moved_any = False
        for comp in components:
            if comp['moved']:
                continue

            # check if all target cells are empty in the current grid
            if all(cur[r][c] == 0 for (r, c) in comp['target']):
                # perform the move
                for (r, c) in comp['cells']:
                    cur[r][c] = 0
                for (r, c) in comp['target']:
                    cur[r][c] = comp['color']
                comp['moved'] = True
                moved_any = True

        if not moved_any:
            break

    # ---------------------------------------------------------------------
    # 4. Return the transformed grid
    # ---------------------------------------------------------------------
    return cur
