2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    n = len(grid)
    m = len(grid[0]) if n else 0

    # ------------------------------------------------------------
    # locate the line of 9s
    nine_pos = [(r, c) for r in range(n) for c in range(m) if grid[r][c] == 9]
    if not nine_pos:
        return grid

    rows = {r for r, _ in nine_pos}
    cols = {c for _, c in nine_pos}
    sums = {r + c for r, c in nine_pos}
    diffs = {r - c for r, c in nine_pos}

    line_type = None
    line_val = None
    if len(rows) == 1:
        line_type = 'horizontal'
        line_val = rows.pop()
    elif len(cols) == 1:
        line_type = 'vertical'
        line_val = cols.pop()
    elif len(sums) == 1:
        line_type = 'diagonal_sum'
        line_val = sums.pop()
    elif len(diffs) == 1:
        line_type = 'diagonal_diff'
        line_val = diffs.pop()
    else:
        # unknown line – do nothing
        return grid

    # ------------------------------------------------------------
    # split cells into the two sides of the line
    # side = -1 for the side where the distance to the line is negative,
    #        +1 for the opposite side
    side_cells = {-1: [], 1: []}
    for r in range(n):
        for c in range(m):
            v = grid[r][c]
            if v == 9 or v == 0:
                continue
            if line_type == 'horizontal':
                side = -1 if r < line_val else 1
            elif line_type == 'vertical':
                side = -1 if c < line_val else 1
            elif line_type == 'diagonal_sum':
                s = r + c
                side = -1 if s < line_val else 1
            else:  # diagonal_diff
                d = r - c
                side = -1 if d < line_val else 1
            side_cells[side].append((r, c, v))

    # ------------------------------------------------------------
    # process each side independently
    for side, cells in side_cells.items():
        if not cells:
            continue

        # bounding box of the whole side (used for region limits)
        min_r = min(r for r, _, _ in cells)
        max_r = max(r for r, _, _ in cells)
        min_c = min(c for _, c, _ in cells)
        max_c = max(c for _, c, _ in cells)

        # ---- find allowed directions (stay on the same side) ----
        dirs = [(-1, 0), (1, 0), (0, -1), (0, 1)]
        allowed = []
        for dx, dy in dirs:
            ok = True
            for r, c, _ in cells:
                nr, nc = r + dx, c + dy
                # check whether the new cell stays on the same side
                if line_type == 'horizontal':
                    if side == -1 and nr >= line_val:
                        ok = False
                        break
                    if side == 1 and nr <= line_val:
                        ok = False
                        break
                elif line_type == 'vertical':
                    if side == -1 and nc >= line_val:
                        ok = False
                        break
                    if side == 1 and nc <= line_val:
                        ok = False
                        break
                elif line_type == 'diagonal_sum':
                    ns = nr + nc
                    if side == -1 and ns >= line_val:
                        ok = False
                        break
                    if side == 1 and ns <= line_val:
                        ok = False
                        break
                else:  # diagonal_diff
                    nd = nr - nc
                    if side == -1 and nd >= line_val:
                        ok = False
                        break
                    if side == 1 and nd <= line_val:
                        ok = False
                        break
            if ok:
                allowed.append((dx, dy))

        # ---- choose the direction that has the largest empty area ----
        def zero_count(dir_pair):
            dx, dy = dir_pair
            # decide whether the direction is orthogonal to the line
            ortho = False
            if line_type == 'horizontal':
                ortho = (dx != 0)
            elif line_type == 'vertical':
                ortho = (dy != 0)
            else:  # any movement changes the distance for a diagonal line
                ortho = True

            if ortho:
                # the "far half" is the part of the side that is farther from the line
                if line_type == 'horizontal':
                    # split by rows
                    if side == -1:  # above the line
                        thr = (min_r + max_r + 1) // 2
                        if dx == -1:  # moving up → target rows 0 … thr-1
                            return sum(1 for r in range(min_r, thr)
                                         for c in range(min_c, max_c + 1)
                                         if grid[r][c] == 0)
                    else:  # side == 1 (below the line)
                        thr = (min_r + max_r) // 2
                        if dx == 1:  # moving down → rows thr+1 … max_r
                            return sum(1 for r in range(thr + 1, max_r + 1)
                                         for c in range(min_c, max_c + 1)
                                         if grid[r][c] == 0)
                elif line_type == 'vertical':
                    if side == -1:  # left of the line
                        thr = (min_c + max_c + 1) // 2
                        if dy == -1:  # moving left → columns 0 … thr-1
                            return sum(1 for r in range(min_r, max_r + 1)
                                         for c in range(min_c, thr)
                                         if grid[r][c] == 0)
                    else:  # right of the line
                        thr = (min_c + max_c) // 2
                        if dy == 1:  # moving right → columns thr+1 … max_c
                            return sum(1 for r in range(min_r, max_r + 1)
                                         for c in range(thr + 1, max_c + 1)
                                         if grid[r][c] == 0)
                else:  # diagonal_sum or diagonal_diff – split by the corresponding coordinate
                    if line_type == 'diagonal_sum':
                        # use the sum as distance to the line
                        sums = [r + c for r, c, _ in cells]
                        min_s, max_s = min(sums), max(sums)
                        if side == -1:
                            thr = (min_s + max_s + 1) // 2
                            if dx == -1 or dy == -1:  # moving up or left (decreases sum)
                                # target area: rows + cols < thr
                                return sum(1 for r, c, _ in cells
                                             if r + c < thr and grid[r][c] == 0)
                        else:
                            thr = (min_s + max_s) // 2
                            if dx == 1 or dy == 1:  # moving down or right (increases sum)
                                return sum(1 for r, c, _ in cells
                                             if r + c > thr and grid[r][c] == 0)
                    else:  # diagonal_diff
                        diffs = [r - c for r, c, _ in cells]
                        min_d, max_d = min(diffs), max(diffs)
                        if side == -1:
                            thr = (min_d + max_d + 1) // 2
                            if (dx == -1 and dy == 1) or (dx == 0 and dy == 1) or (dx == -1 and dy == 0):
                                # directions that decrease the diff
                                return sum(1 for r, c, _ in cells
                                             if r - c < thr and grid[r][c] == 0)
                        else:
                            thr = (min_d + max_d) // 2
                            if (dx == 1 and dy == -1) or (dx == 0 and dy == -1) or (dx == 1 and dy == 0):
                                return sum(1 for r, c, _ in cells
                                             if r - c > thr and grid[r][c] == 0)
                # if we cannot determine a clear far half, treat as zero
                return 0
            else:
                # parallel direction – count zeros outside the region
                if dx == -1:  # up – rows above the region
                    return sum(1 for r in range(min_r)
                                 for c in range(min_c, max_c + 1)
                                 if grid[r][c] == 0)
                if dx == 1:   # down – rows below the region
                    return sum(1 for r in range(max_r + 1, n)
                                 for c in range(min_c, max_c + 1)
                                 if grid[r][c] == 0)
                if dy == -1:  # left – columns left of the region
                    return sum(1 for r in range(min_r, max_r + 1)
                                 for c in range(min_c)
                                 if grid[r][c] == 0)
                if dy == 1:   # right – columns right of the region
                    return sum(1 for r in range(min_r, max_r + 1)
                                 for c in range(max_c + 1, m)
                                 if grid[r][c] == 0)
                return 0

        best_dir = max(allowed, key=lambda d: zero_count(d))
        dx, dy = best_dir

        # ---- decide whether the chosen direction is orthogonal ----
        ortho = False
        if line_type == 'horizontal':
            ortho = (dx != 0)
        elif line_type == 'vertical':
            ortho = (dy != 0)
        else:
            ortho = True

        # ---- split into near/far halves if the movement is orthogonal ----
        if ortho:
            # compute the threshold that separates the half nearer to the line
            if line_type == 'horizontal':
                if side == -1:
                    thr = (min_r + max_r + 1) // 2
                    moving = [(r, c, v) for (r, c, v) in cells if r >= thr]
                else:
                    thr = (min_r + max_r) // 2
                    moving = [(r, c, v) for (r, c, v) in cells if r <= thr]
            elif line_type == 'vertical':
                if side == -1:
                    thr = (min_c + max_c + 1) // 2
                    moving = [(r, c, v) for (r, c, v) in cells if c >= thr]
                else:
                    thr = (min_c + max_c) // 2
                    moving = [(r, c, v) for (r, c, v) in cells if c <= thr]
            elif line_type == 'diagonal_sum':
                sums = [r + c for r, c, _ in cells]
                min_s, max_s = min(sums), max(sums)
                if side == -1:
                    thr = (min_s + max_s + 1) // 2
                    moving = [(r, c, v) for (r, c, v) in cells if (r + c) >= thr]
                else:
                    thr = (min_s + max_s) // 2
                    moving = [(r, c, v) for (r, c, v) in cells if (r + c) <= thr]
            else:  # diagonal_diff
                diffs = [r - c for r, c, _ in cells]
                min_d, max_d = min(diffs), max(diffs)
                if side == -1:
                    thr = (min_d + max_d + 1) // 2
                    moving = [(r, c, v) for (r, c, v) in cells if (r - c) >= thr]
                else:
                    thr = (min_d + max_d) // 2
                    moving = [(r, c, v) for (r, c, v) in cells if (r - c) <= thr]
        else:
            moving = cells

        # ---- remove the moving cells from the grid ----
        for r, c, _ in moving:
            grid[r][c] = 0

        # ---- perform the actual movement ----
        if ortho:
            # sort by distance to the target side – farthest first
            if dx == -1:          # up
                moving.sort(key=lambda x: -x[0])
            elif dx == 1:         # down
                moving.sort(key=lambda x: x[0])
            elif dy == -1:        # left
                moving.sort(key=lambda x: -x[1])
            else:                 # right
                moving.sort(key=lambda x: -x[1])

            # buffer cell that may not be used
            if dx == -1:
                buffer_r = min_r
            elif dx == 1:
                buffer_r = max_r
            elif dy == -1:
                buffer_c = min_c
            else:
                buffer_c = max_c

            for r, c, v in moving:
                # try to move step by step
                while True:
                    nr, nc = r + dx, c + dy
                    # do not cross the line or the buffer cell
                    if (dx != 0 and ((dx == -1 and nr < buffer_r) or
                                      (dx == 1 and nr > buffer_r))):
                        break
                    if (dy != 0 and ((dy == -1 and nc < buffer_c) or
                                      (dy == 1 and nc > buffer_c))):
                        break
                    # do not step onto a non‑zero cell
                    if grid[nr][nc] != 0:
                        break
                    # move into the empty cell
                    grid[nr][nc] = v
                    grid[r][c] = 0
                    r, c = nr, nc
        else:
            # parallel movement – simple alignment (right/left or up/down)
            if dx == 0:  # horizontal movement – treat per row
                # group values by row
                from collections import defaultdict
                rows_dict = defaultdict(list)
                for r, c, v in moving:
                    rows_dict[r].append(v)
                for row, vals in rows_dict.items():
                    if dy == 1:  # right
                        start = max_c - len(vals) + 1
                        for i, v in enumerate(vals):
                            grid[row][start + i] = v
                    else:  # left
                        start = min_c
                        for i, v in enumerate(vals):
                            grid[row][start + i] = v
            else:  # vertical movement – treat per column
                from collections import defaultdict
                cols_dict = defaultdict(list)
                for r, c, v in moving:
                    cols_dict[c].append(v)
                for col, vals in cols_dict.items():
                    if dx == -1:  # up
                        start = min_r
                        for i, v in enumerate(vals):
                            grid[start + i][col] = v
                    else:  # down
                        start = max_r - len(vals) + 1
                        for i, v in enumerate(vals):
                            grid[start + i][col] = v

    return grid
