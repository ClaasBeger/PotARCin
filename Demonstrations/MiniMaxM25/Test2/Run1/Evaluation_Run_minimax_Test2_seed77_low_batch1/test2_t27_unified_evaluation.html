2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import math
    # copy original grid for reference
    orig = [row[:] for row in grid]
    rows = len(grid)
    cols = len(grid[0])
    # find all non‑zero colors and their bounding boxes
    color_info = {}
    for r in range(rows):
        for c in range(cols):
            v = grid[r][c]
            if v != 0:
                if v not in color_info:
                    color_info[v] = {'min_r': r, 'max_r': r,
                                     'min_c': c, 'max_c': c}
                else:
                    info = color_info[v]
                    if r < info['min_r']: info['min_r'] = r
                    if r > info['max_r']: info['max_r'] = r
                    if c < info['min_c']: info['min_c'] = c
                    if c > info['max_c']: info['max_c'] = c
    # build list of shapes
    shapes = []
    for col, info in color_info.items():
        shapes.append((info['min_r'], info['max_r'],
                       info['min_c'], info['max_c'], col))
    # pair leftmost with rightmost
    if len(shapes) >= 2:
        shapes.sort(key=lambda x: x[2])      # sort by min_c
        left = shapes[0]
        right = shapes[-1]
        shapes = shapes[1:-1]
    else:
        left = right = None
    # pair topmost with bottommost among the rest
    if len(shapes) >= 2:
        shapes.sort(key=lambda x: x[0])      # sort by min_r
        top = shapes[0]
        bottom = shapes[-1]
        remaining = shapes[1:-1]
    else:
        top = bottom = None
        remaining = shapes

    # the pairs we will process
    pairs = []
    if left and right:
        pairs.append((left, right))
    if top and bottom:
        pairs.append((top, bottom))
    # any leftover shapes are ignored (they stay unchanged)

    # helper to fill a region
    def fill(src, tgt):
        src_min_r, src_max_r, src_min_c, src_max_c, src_col = src
        tgt_min_r, tgt_max_r, tgt_min_c, tgt_max_c, tgt_col = tgt
        src_h = src_max_r - src_min_r + 1
        src_w = src_max_c - src_min_c + 1
        tgt_h = tgt_max_r - tgt_min_r + 1
        tgt_w = tgt_max_c - tgt_min_c + 1

        row_expand = max(0, src_h - tgt_h)
        col_expand = max(0, src_w - tgt_w)

        new_min_r = tgt_min_r - (row_expand + 1)//2
        new_max_r = tgt_max_r + row_expand//2

        new_min_c = tgt_min_c - col_expand//2
        new_max_c = tgt_max_c + (col_expand + 1)//2

        # extra column shift when expanding rows but not columns
        if col_expand == 0 and row_expand > 0:
            new_min_c += 1
            new_max_c += 1

        for r in range(new_min_r, new_max_r+1):
            for c in range(new_min_c, new_max_c+1):
                if 0 <= r < rows and 0 <= c < cols and orig[r][c] == 0:
                    grid[r][c] = src_col

    # apply both directions for each pair
    for a, b in pairs:
        fill(a, b)   # a onto b
        fill(b, a)   # b onto a

    return grid
