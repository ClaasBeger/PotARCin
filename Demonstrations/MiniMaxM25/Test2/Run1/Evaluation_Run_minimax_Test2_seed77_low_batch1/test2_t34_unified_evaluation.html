2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    # Find the square of filler (most likely 8) that forms a solid block
    nrows = len(grid)
    ncols = len(grid[0]) if nrows else 0
    
    # Count occurrences of each non‑zero value
    from collections import Counter
    cnt = Counter()
    for row in grid:
        for v in row:
            if v != 0:
                cnt[v] += 1
    
    # Candidate filler is the colour that forms a square
    filler = None
    r0 = c0 = side = None
    for col, (val, _) in enumerate(cnt.most_common()):
        # collect positions of this colour
        positions = []
        for i in range(nrows):
            for j in range(ncols):
                if grid[i][j] == val:
                    positions.append((i, j))
        if not positions:
            continue
        min_r = min(p[0] for p in positions)
        max_r = max(p[0] for p in positions)
        min_c = min(p[1] for p in positions)
        max_c = max(p[1] for p in positions)
        side_len = max_r - min_r + 1
        if side_len == max_c - min_c + 1 and side_len > 1:
            # square – check that almost all cells inside are this colour (allow one different cell, the marker)
            good = True
            diff = 0
            for r in range(min_r, max_r+1):
                for c in range(min_c, max_c+1):
                    if grid[r][c] != val:
                        diff += 1
                        if diff > 1:
                            good = False
                            break
                if not good:
                    break
            if good:
                filler = val
                r0, c0, side = min_r, min_c, side_len
                break
    if filler is None:
        # fallback – assume the most frequent non‑zero is filler
        filler = cnt.most_common()[0][0]
        # locate its bounding box
        positions = [(i,j) for i in range(nrows) for j in range(ncols) if grid[i][j]==filler]
        r0 = min(p[0] for p in positions)
        c0 = min(p[1] for p in positions)
        side = max(p[0] for p in positions) - r0 + 1
    
    # Build output grid
    out = [[0]*side for _ in range(side)]
    
    # Process each colour except filler, 0 and the marker 2
    all_vals = sorted(set(cnt.keys()) - {0, filler, 2})
    for val in all_vals:
        # collect coordinates of this colour
        coords = [(i,j) for i in range(nrows) for j in range(ncols) if grid[i][j]==val]
        if not coords:
            continue
        
        # --- try horizontal line crossing the block ---
        # a row inside the block that has the colour both left and right of the block
        hor_hidden = []
        for r in range(r0, r0+side):
            has_left = any(c < c0 for _,c in coords if _ == r)
            has_right = any(c >= c0+side for _,c in coords if _ == r)
            if has_left and has_right:
                # hidden points are the block cells in this row
                for c in range(c0, c0+side):
                    if r0 <= r < r0+side and c0 <= c < c0+side:
                        hor_hidden.append((r,c))
        if hor_hidden:
            # use horizontal line transformation (vertical flip)
            for (r,c) in hor_hidden:
                dr = r - r0
                dc = c - c0
                orow = side-1 - dr
                ocol = dc
                if out[orow][ocol] < val:
                    out[orow][ocol] = val
            continue   # skip other orientations for this colour
        
        # --- try vertical line crossing the block ---
        ver_hidden = []
        for c in range(c0, c0+side):
            has_above = any(r < r0 for r,_ in coords if _ == c)
            has_below = any(r >= r0+side for r,_ in coords if _ == c)
            if has_above or has_below:
                for r in range(r0, r0+side):
                    if r0 <= r < r0+side and c0 <= c < c0+side:
                        ver_hidden.append((r,c))
        if ver_hidden:
            # vertical line -> horizontal flip
            for (r,c) in ver_hidden:
                dr = r - r0
                dc = c - c0
                orow = dr
                ocol = side-1 - dc
                if out[orow][ocol] < val:
                    out[orow][ocol] = val
            continue
        
        # --- diagonal (slope +1) lines (col - row constant) ---
        diffs = {}
        for (r,c) in coords:
            d = c - r
            diffs[d] = True
        for d in diffs:
            # check whether this diagonal appears outside the block
            outside = any((c - r == d) and (r < r0 or r >= r0+side or c < c0 or c >= c0+side) for (r,c) in coords)
            if not outside:
                continue
            # hidden points inside block for this diagonal
            for r in range(r0, r0+side):
                c = r + d
                if c0 <= c < c0+side:
                    # hidden cell
                    dr = r - r0
                    dc = c - c0
                    orow = dr
                    ocol = side-1 - dc
                    if out[orow][ocol] < val:
                        out[orow][ocol] = val
        
        # --- diagonal (slope -1) lines (col + row constant) ---
        sums = {}
        for (r,c) in coords:
            s = c + r
            sums[s] = True
        for s in sums:
            outside = any((c + r == s) and (r < r0 or r >= r0+side or c < c0 or c >= c0+side) for (r,c) in coords)
            if not outside:
                continue
            for r in range(r0, r0+side):
                c = s - r
                if c0 <= c < c0+side:
                    dr = r - r0
                    dc = c - c0
                    orow = dr
                    ocol = side-1 - dc
                    if out[orow][ocol] < val:
                        out[orow][ocol] = val
    
    return out
