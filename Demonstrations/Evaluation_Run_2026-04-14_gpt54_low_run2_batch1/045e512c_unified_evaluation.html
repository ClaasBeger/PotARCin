2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0]) if h else 0

    def neighbors8(r, c):
        for dr in (-1, 0, 1):
            for dc in (-1, 0, 1):
                if dr == 0 and dc == 0:
                    continue
                nr, nc = r + dr, c + dc
                if 0 <= nr < h and 0 <= nc < w:
                    yield nr, nc

    # Find 8-connected monochrome components
    seen = [[False] * w for _ in range(h)]
    comps = []
    for r in range(h):
        for c in range(w):
            color = grid[r][c]
            if color == 0 or seen[r][c]:
                continue
            stack = [(r, c)]
            seen[r][c] = True
            cells = []
            while stack:
                cr, cc = stack.pop()
                cells.append((cr, cc))
                for nr, nc in neighbors8(cr, cc):
                    if not seen[nr][nc] and grid[nr][nc] == color:
                        seen[nr][nc] = True
                        stack.append((nr, nc))
            rs = [x for x, _ in cells]
            cs = [y for _, y in cells]
            r0, r1 = min(rs), max(rs)
            c0, c1 = min(cs), max(cs)
            norm = {(rr - r0, cc - c0) for rr, cc in cells}
            comps.append({
                "color": color,
                "cells": cells,
                "set": set(cells),
                "r0": r0, "r1": r1,
                "c0": c0, "c1": c1,
                "h": r1 - r0 + 1,
                "w": c1 - c0 + 1,
                "norm": norm,
                "size": len(cells),
            })

    if not comps:
        return grid

    # Reference is the largest component
    ref = max(comps, key=lambda comp: comp["size"])
    ref_shape = ref["norm"]
    ref_h = ref["h"]
    ref_w = ref["w"]

    out = [row[:] for row in grid]

    def stamp(top, left, color):
        for dr, dc in ref_shape:
            r = top + dr
            c = left + dc
            if 0 <= r < h and 0 <= c < w:
                out[r][c] = color

    for comp in comps:
        if comp is ref:
            continue

        candidates = []
        comp_cells = comp["cells"]

        # Find placements where this component is a subset of the reference shape
        for rr, cc in ref_shape:
            for ar, ac in comp_cells:
                top = ar - rr
                left = ac - cc
                ok = True
                for xr, xc in comp_cells:
                    if (xr - top, xc - left) not in ref_shape:
                        ok = False
                        break
                if ok:
                    candidates.append((top, left))

        # Deduplicate
        candidates = list(dict.fromkeys(candidates))
        if not candidates:
            continue

        best = None
        best_score = None

        for top, left in candidates:
            dr = top - ref["r0"]
            dc = left - ref["c0"]

            sr = 0 if dr == 0 else (1 if dr > 0 else -1)
            sc = 0 if dc == 0 else (1 if dc > 0 else -1)
            if sr == 0 and sc == 0:
                continue

            # Must be on the expected lattice relative to the reference
            if sr != 0 and abs(dr) % (ref_h + 1) != 0:
                continue
            if sc != 0 and abs(dc) % (ref_w + 1) != 0:
                continue

            kr = abs(dr) // (ref_h + 1) if sr != 0 else 1
            kc = abs(dc) // (ref_w + 1) if sc != 0 else 1
            if sr != 0 and kr < 1:
                continue
            if sc != 0 and kc < 1:
                continue

            score = (abs(dr) + abs(dc), top, left)
            if best is None or score < best_score:
                best = (top, left, sr, sc)
                best_score = score

        if best is None:
            # Fallback: choose nearest valid subset placement by direction
            for top, left in candidates:
                dr = top - ref["r0"]
                dc = left - ref["c0"]
                sr = 0 if dr == 0 else (1 if dr > 0 else -1)
                sc = 0 if dc == 0 else (1 if dc > 0 else -1)
                if sr == 0 and sc == 0:
                    continue
                score = (abs(dr) + abs(dc), top, left)
                if best is None or score < best_score:
                    best = (top, left, sr, sc)
                    best_score = score

        if best is None:
            continue

        top, left, sr, sc = best
        step_r = sr * (ref_h + 1) if sr != 0 else 0
        step_c = sc * (ref_w + 1) if sc != 0 else 0

        cur_top, cur_left = top, left
        while True:
            # Stop if translated bounding box is completely outside
            if cur_top >= h or cur_left >= w or cur_top + ref_h - 1 < 0 or cur_left + ref_w - 1 < 0:
                break
            stamp(cur_top, cur_left, comp["color"])
            cur_top += step_r
            cur_left += step_c

    return out
