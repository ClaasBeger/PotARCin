2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    """
    Transforms the input grid by adding a block of colour 8 in the
    empty region that lies between two rectangular coloured blocks.
    The rule is:
      * If the blocks are stacked vertically (their row intervals are disjoint
        and their column intervals overlap), the gap rows are filled with 8,
        restricted to the interior columns of the overlapping column interval.
      * If the blocks are side‑by‑side (their column intervals are disjoint
        and their row intervals overlap), the gap columns are filled with 8,
        restricted to the interior rows of the overlapping row interval.
    """
    # ----------------------------------------------------------------------
    # 1. Find the two most frequent non‑zero colours (the rectangular blocks)
    # ----------------------------------------------------------------------
    from collections import Counter
    cnt = Counter()
    for row in grid:
        for v in row:
            if v != 0:
                cnt[v] += 1

    if len(cnt) < 2:
        return grid                       # nothing to do

    # choose the two colours with the largest counts
    colour1, colour2 = (c1 := cnt.most_common(2)[0][0]), (c2 := cnt.most_common(2)[1][0])

    # ----------------------------------------------------------------------
    # 2. Compute bounding rectangles for the two colours
    # ----------------------------------------------------------------------
    def bounds(colour):
        min_r = max_r = min_c = max_c = None
        for r, row in enumerate(grid):
            for c, v in enumerate(row):
                if v == colour:
                    if min_r is None:
                        min_r = max_r = r
                        min_c = max_c = c
                    else:
                        if r < min_r: min_r = r
                        if r > max_r: max_r = r
                        if c < min_c: min_c = c
                        if c > max_c: max_c = c
        return min_r, max_r, min_c, max_c

    r1 = bounds(colour1)
    r2 = bounds(colour2)

    # ----------------------------------------------------------------------
    # 3. Decide whether the blocks are vertical (stacked) or horizontal
    # ----------------------------------------------------------------------
    # check for row‑overlap (blocks side‑by‑side)
    row_overlap = (r1[1] >= r2[0] and r2[1] >= r1[0])
    # check for column‑overlap (blocks stacked)
    col_overlap = (r1[3] >= r2[2] and r2[3] >= r1[2])

    # ----------------------------------------------------------------------
    # 4. Fill the appropriate gap with colour 8
    # ----------------------------------------------------------------------
    if row_overlap:
        # side‑by‑side: left‑right arrangement
        if r1[2] < r2[2]:
            left = r1
            right = r2
        else:
            left = r2
            right = r1

        # columns between the two blocks
        gap_c_start = left[3] + 1
        gap_c_end   = right[2] - 1
        if gap_c_start <= gap_c_end:
            gap_cols = range(gap_c_start, gap_c_end + 1)
        else:
            gap_cols = []

        # overlapping rows
        inter_r_top    = max(left[0], right[0])
        inter_r_bottom = min(left[1], right[1])

        # interior rows (exclude the outermost overlapping rows)
        interior_rows = range(inter_r_top + 1, inter_r_bottom)

        for rr in interior_rows:
            for cc in gap_cols:
                grid[rr][cc] = 8

    elif col_overlap:
        # stacked: top‑bottom arrangement
        if r1[0] < r2[0]:
            top = r1
            bottom = r2
        else:
            top = r2
            bottom = r1

        # rows between the two blocks
        gap_r_start = top[1] + 1
        gap_r_end   = bottom[0] - 1
        if gap_r_start <= gap_r_end:
            gap_rows = range(gap_r_start, gap_r_end + 1)
        else:
            gap_rows = []

        # overlapping columns
        inter_c_left  = max(top[2], bottom[2])
        inter_c_right = min(top[3], bottom[3])

        # interior columns (exclude the outermost overlapping columns)
        interior_cols = range(inter_c_left + 1, inter_c_right)

        for rr in gap_rows:
            for cc in interior_cols:
                grid[rr][cc] = 8

    # ----------------------------------------------------------------------
    # 5. Return the transformed grid
    # ----------------------------------------------------------------------
    return grid
