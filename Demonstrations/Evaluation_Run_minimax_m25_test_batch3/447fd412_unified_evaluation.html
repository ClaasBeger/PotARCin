2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import math
    from collections import deque

    rows = len(grid)
    cols = len(grid[0]) if rows else 0
    out = [row[:] for row in grid]

    # ---------- find connected components of colour 2 ----------
    visited = [[False] * cols for _ in range(rows)]
    components = []

    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 2 and not visited[r][c]:
                q = deque([(r, c)])
                visited[r][c] = True
                cells = []
                while q:
                    cr, cc = q.popleft()
                    cells.append((cr, cc))
                    for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                        nr, nc = cr + dr, cc + dc
                        if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == 2 and not visited[nr][nc]:
                            visited[nr][nc] = True
                            q.append((nr, nc))

                rows_set = {cell[0] for cell in cells}
                cols_set = {cell[1] for cell in cells}
                min_r = min(r for r, _ in cells)
                max_r = max(r for r, _ in cells)
                min_c = min(c for _, c in cells)
                max_c = max(c for _, c in cells)
                height = max_r - min_r + 1
                width = max_c - min_c + 1

                if width > 1 and height == 1:
                    orientation = 'horizontal'
                elif height > 1 and width == 1:
                    orientation = 'vertical'
                else:
                    orientation = 'horizontal'          # squares treated as horizontal

                components.append({
                    'id': len(components),
                    'cells': cells,
                    'rows': rows_set,
                    'cols': cols_set,
                    'min_r': min_r, 'max_r': max_r,
                    'min_c': min_c, 'max_c': max_c,
                    'height': height, 'width': width,
                    'size': len(cells),
                    'orientation': orientation
                })

    # ---------- helper to set cells ----------
    def set_cells(coords):
        for r, c in coords:
            if 0 <= r < rows and 0 <= c < cols and out[r][c] == 0:
                out[r][c] = 1

    # ---------- find aligned pairs ----------
    used_in_pair = set()
    processed_pairs = set()
    n = len(components)

    for i in range(n):
        for j in range(i + 1, n):
            a = components[i]
            b = components[j]
            aligned = False
            if a['orientation'] == 'horizontal' and b['orientation'] == 'horizontal':
                if a['rows'] & b['rows']:
                    aligned = True
            elif a['orientation'] == 'vertical' and b['orientation'] == 'vertical':
                if a['cols'] & b['cols']:
                    aligned = True
            if not aligned:
                continue

            pair_key = frozenset([a['id'], b['id']])
            if pair_key in processed_pairs:
                continue
            processed_pairs.add(pair_key)
            used_in_pair.add(a['id'])
            used_in_pair.add(b['id'])

            # ---------- horizontal pair ----------
            if a['orientation'] == 'horizontal':
                # order left‑right
                if a['min_c'] < b['min_c']:
                    left, right = a, b
                else:
                    left, right = b, a
                d = left['height']
                D = ((right['min_c'] + right['max_c']) - (left['min_c'] + left['max_c'])) // 2

                # rectangle (gap)
                rect_rows = range(left['min_r'], left['max_r'] + 1)
                rect_cols = range(left['max_c'] + 1, right['min_c'])

                # apex
                apex_rows = range(left['max_r'] + 1, left['max_r'] + d + 1)
                mid = (left['max_c'] + right['min_c']) / 2
                apex_col0 = int(math.ceil(mid))
                apex_cols = range(apex_col0, apex_col0 + d)

                # base
                base_rows = range(left['max_r'] + d + 1, left['max_r'] + 2 * d + 1)
                base_cols = range(left['max_c'] + 1, right['max_c'] + 1)

                # check for existing 1s
                already = False
                for r in rect_rows:
                    for c in rect_cols:
                        if 0 <= r < rows and 0 <= c < cols and out[r][c] == 1:
                            already = True
                for r in apex_rows:
                    for c in apex_cols:
                        if 0 <= r < rows and 0 <= c < cols and out[r][c] == 1:
                            already = True
                for r in base_rows:
                    for c in base_cols:
                        if 0 <= r < rows and 0 <= c < cols and out[r][c] == 1:
                            already = True
                if already:
                    continue

                set_cells((r, c) for r in rect_rows for c in rect_cols)
                set_cells((r, c) for r in apex_rows for c in apex_cols)
                set_cells((r, c) for r in base_rows for c in base_cols)

            # ---------- vertical pair ----------
            else:
                # order top‑bottom
                if a['min_r'] < b['min_r']:
                    top, bottom = a, b
                else:
                    top, bottom = b, a
                d = top['width']
                D = ((bottom['min_r'] + bottom['max_r']) - (top['min_r'] + top['max_r'])) // 2

                base_row = top['min_r'] + d
                base_col0 = top['min_c'] - D // 2
                base_cols = range(base_col0, base_col0 + D)

                apex_row = top['min_r'] + 2 * d
                apex_col = top['min_c']

                already = False
                for c in base_cols:
                    if 0 <= base_row < rows and 0 <= c < cols and out[base_row][c] == 1:
                        already = True
                if 0 <= apex_row < rows and 0 <= apex_col < cols and out[apex_row][apex_col] == 1:
                    already = True
                if already:
                    continue

                set_cells((base_row, c) for c in base_cols)
                set_cells([(apex_row, apex_col)])

    # ---------- isolated clusters (size > 2) ----------
    for comp in components:
        if comp['id'] in used_in_pair:
            continue
        if comp['size'] <= 2:
            continue

        left_gap = comp['min_c']
        right_gap = (cols - 1) - comp['max_c']

        if left_gap > right_gap:
            fill_cols = range(0, comp['min_c'])
        else:
            fill_cols = range(comp['max_c'] + 1, cols)

        fill_rows = range(comp['min_r'], comp['max_r'] + 1)

        # avoid overwriting existing 1s
        already = False
        for r in fill_rows:
            for c in fill_cols:
                if 0 <= r < rows and 0 <= c < cols and out[r][c] == 1:
                    already = True
        if already:
            continue

        set_cells((r, c) for r in fill_rows for c in fill_cols)

    return out
