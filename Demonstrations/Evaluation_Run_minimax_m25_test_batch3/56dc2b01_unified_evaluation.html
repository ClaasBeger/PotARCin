2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    n = len(grid)
    m = len(grid[0]) if n else 0

    # locate the line of 2s
    row_of_2 = None
    for i in range(n):
        if all(cell == 2 for cell in grid[i]):
            row_of_2 = i
            break

    col_of_2 = None
    if row_of_2 is None:
        for j in range(m):
            if all(grid[i][j] == 2 for i in range(n)):
                col_of_2 = j
                break

    orientation = 'horizontal' if row_of_2 is not None else 'vertical'

    # collect all cells with value 3
    shape = [(i, j) for i in range(n) for j in range(m) if grid[i][j] == 3]
    if not shape:
        return grid

    min_r = min(r for r, _ in shape)
    max_r = max(r for r, _ in shape)
    min_c = min(c for _, c in shape)
    max_c = max(c for _, c in shape)

    # compute translation to bring shape adjacent to the 2 line
    dr = dc = 0
    if orientation == 'horizontal':
        if max_r < row_of_2:          # shape is above the line
            dr = (row_of_2 - 1) - max_r
        elif min_r > row_of_2:        # shape is below the line
            dr = (row_of_2 + 1) - min_r
    else:  # vertical
        if max_c < col_of_2:          # shape is left of the line
            dc = (col_of_2 - 1) - max_c
        elif min_c > col_of_2:        # shape is right of the line
            dc = (col_of_2 + 1) - min_c

    new_shape = [(r + dr, c + dc) for r, c in shape]

    new_min_r = min(r for r, _ in new_shape)
    new_max_r = max(r for r, _ in new_shape)
    new_min_c = min(c for _, c in new_shape)
    new_max_c = max(c for _, c in new_shape)

    # create empty output grid
    out = [[0] * m for _ in range(n)]

    # place the line of 2s
    if orientation == 'horizontal':
        for j in range(m):
            out[row_of_2][j] = 2
    else:
        for i in range(n):
            out[i][col_of_2] = 2

    # place the shifted shape of 3s
    for r, c in new_shape:
        out[r][c] = 3

    # place the line of 8s on the side opposite the 2 line
    if orientation == 'horizontal':
        if max_r < row_of_2:          # shape above the 2 line
            row8 = new_min_r - 1
            if 0 <= row8 < n:
                for j in range(m):
                    out[row8][j] = 8
        else:                         # shape below the 2 line
            row8 = new_max_r + 1
            if 0 <= row8 < n:
                for j in range(m):
                    out[row8][j] = 8
    else:  # vertical
        if max_c < col_of_2:          # shape left of the 2 line
            col8 = new_min_c - 1
            if 0 <= col8 < m:
                for i in range(n):
                    out[i][col8] = 8
        else:                         # shape right of the 2 line
            col8 = new_max_c + 1
            if 0 <= col8 < m:
                for i in range(n):
                    out[i][col8] = 8

    return out
