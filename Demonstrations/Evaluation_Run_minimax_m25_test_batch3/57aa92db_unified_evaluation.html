2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

from collections import defaultdict

def solve(grid):
    # copy the input grid so we can modify it
    grid = [row[:] for row in grid]
    rows = len(grid)
    cols = len(grid[0]) if rows else 0

    # collect all non‑zero cells per colour
    colour_cells = defaultdict(list)
    for r in range(rows):
        for c in range(cols):
            val = grid[r][c]
            if val != 0:
                colour_cells[val].append((r, c))

    # bounding boxes for each colour
    colour_bbox = {}
    for col, cells in colour_cells.items():
        rs = [r for r, _ in cells]
        cs = [c for _, c in cells]
        colour_bbox[col] = (min(rs), max(rs), min(cs), max(cs))

    # examine every unordered pair of colours
    colours = list(colour_cells.keys())
    for i, c1 in enumerate(colours):
        for c2 in colours[i + 1:]:
            # obtain the bounding boxes
            r1min, r1max, c1min, c1max = colour_bbox[c1]
            r2min, r2max, c2min, c2max = colour_bbox[c2]

            # decide whether the two boxes touch in a single orthogonal direction
            # (they must be exactly one cell apart in the orthogonal dimension
            #  and overlap in the other dimension)
            direction = None          # direction in which the larger colour lies w.r.t. the smaller
            if c1max + 1 == c2min and not (r1max < r2min or r2max < r1min):
                # c1 is left of c2
                direction = ('left', c1, c2)
            elif c2max + 1 == c1min and not (r1max < r2min or r2max < r1min):
                # c2 is left of c1
                direction = ('left', c2, c1)
            elif r1max + 1 == r2min and not (c1max < c2min or c2max < c1min):
                # c1 is above c2
                direction = ('up', c1, c2)
            elif r2max + 1 == r1min and not (c1max < c2min or c2max < c1min):
                # c2 is above c1
                direction = ('up', c2, c1)

            if direction is None:
                continue

            # interpret the tuple
            dir_str, left_colour, right_colour = direction

            # decide which colour is larger (the one that will be expanded)
            if left_colour > right_colour:
                L, S = left_colour, right_colour
                # after the swap the direction has to be flipped
                if dir_str == 'left':
                    dir_str = 'right'   # L is now right of S
                else:  # dir_str == 'up'
                    dir_str = 'down'    # L is now below S
            else:
                L, S = right_colour, left_colour   # L is already on the side described by dir_str

            # recompute the bounding boxes for the chosen L and S
            rLmin, rLmax, cLmin, cLmax = colour_bbox[L]
            rSmin, rSmax, cSmin, cSmax = colour_bbox[S]

            # determine the final direction (L relative to S)
            if cLmax < cSmin and not (rLmax < rSmin or rSmax < rLmin):
                final_dir = 'left'
            elif cSmax < cLmin and not (rLmax < rSmin or rSmax < rLmin):
                final_dir = 'right'
            elif rLmax < rSmin and not (cLmax < cSmin or cSmax < cLmin):
                final_dir = 'up'
            elif rSmax < rLmin and not (cLmax < cSmin or cSmax < cLmin):
                final_dir = 'down'
            else:
                continue   # not a simple side‑by‑side adjacency

            # size of the larger shape
            h = rLmax - rLmin + 1
            w = cLmax - cLmin + 1
            cellsL = colour_cells[L]

            # generate the three orthogonal translation vectors
            offsets = []
            if final_dir == 'left':
                offsets = [(-w, 0), (0, -h), (0, h)]
                # corners are added only when the shape is a single cell
                if w == 1 or h == 1:
                    offsets += [(-w, -h), (-w, h)]
            elif final_dir == 'right':
                offsets = [(w, 0), (0, -h), (0, h)]
                if w == 1 or h == 1:
                    offsets += [(w, -h), (w, h)]
            elif final_dir == 'up':
                offsets = [(0, -h), (-w, 0), (w, 0)]
                if w == 1 or h == 1:
                    offsets += [(-w, -h), (w, -h)]
            elif final_dir == 'down':
                offsets = [(0, h), (-w, 0), (w, 0)]
                if w == 1 or h == 1:
                    offsets += [(-w, h), (w, h)]

            # apply every translation
            for dr, dc in offsets:
                for (r, c) in cellsL:
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] == 0:
                        grid[nr][nc] = L

    return grid
