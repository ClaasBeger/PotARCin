2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    # Copy grid to avoid modifying the input
    R = len(grid)
    C = len(grid[0])
    g = [row[:] for row in grid]

    # ------------------------------------------------------------
    # 1) Horizontal fill: fill gaps between the leftmost and
    #    rightmost 2 in each row.
    # ------------------------------------------------------------
    for r in range(R):
        cols = [c for c in range(C) if g[r][c] == 2]
        if len(cols) >= 2:
            mn = min(cols)
            mx = max(cols)
            for c in range(mn + 1, mx):
                if g[r][c] != 0:          # only fill non‑zero cells
                    g[r][c] = 8

    # ------------------------------------------------------------
    # 2) Vertical fill: fill gaps between the topmost and bottommost
    #    2 in each column.
    # ------------------------------------------------------------
    for c in range(C):
        rows = [r for r in range(R) if g[r][c] == 2]
        if len(rows) >= 2:
            mn = min(rows)
            mx = max(rows)
            for r in range(mn + 1, mx):
                if g[r][c] != 0:
                    g[r][c] = 8

    # ------------------------------------------------------------
    # 3) Locate the (unique) block of consecutive 2's in each row.
    #    If a row contains exactly one such block (length >= 2) we
    #    treat its leftmost column for “extension” (fill above/
    #    below, and possibly change the nearest 2).
    # ------------------------------------------------------------
    # Pre‑compute column -> list of rows containing a 2
    col2 = {c: [] for c in range(C)}
    for r in range(R):
        for c in range(C):
            if g[r][c] == 2:
                col2[c].append(r)
    for c in col2:
        col2[c].sort()

    for r in range(R):
        # find all runs of consecutive 2's
        runs = []
        c = 0
        while c < C:
            if g[r][c] == 2:
                start = c
                while c + 1 < C and g[r][c + 1] == 2:
                    c += 1
                end = c
                runs.append((start, end))
            c += 1

        # consider only runs of length >= 2
        big_runs = [run for run in runs if run[1] - run[0] + 1 >= 2]
        if len(big_runs) != 1:
            continue            # either none or more than one block -> skip

        start, _ = big_runs[0]   # leftmost column of the (only) block
        c = start
        rows2 = col2.get(c, [])

        # -----處理上方 (above) -----
        above = [row for row in rows2 if row < r]
        if above:
            nearest_above = max(above)          # highest 2 < r
            # only if it is directly above and also the topmost 2 in the column
            if nearest_above == r - 1 and nearest_above == min(above):
                g[nearest_above][c] = 8        # change the 2 to 8
                up = nearest_above - 1
                while up >= 0 and g[up][c] != 0:
                    g[up][c] = 8
                    up -= 1
            # otherwise do nothing
        else:
            # no 2 above – fill ordinary cells above the block
            up = r - 1
            while up >= 0 and g[up][c] != 0:
                g[up][c] = 8
                up -= 1

        # -----處理下方 (below) -----
        below = [row for row in rows2 if row > r]
        if below:
            nearest_below = min(below)          # lowest 2 > r
            if nearest_below == r + 1:
                # find the bottommost contiguous 2 in this column
                bottom = nearest_below
                while bottom + 1 < R and g[bottom + 1][c] == 2:
                    bottom += 1
                # fill only if the directly‑below 2 is also the bottommost one
                if bottom == max(below):
                    down = bottom + 1
                    while down < R and g[down][c] != 0:
                        g[down][c] = 8
                        down += 1
                # else: there is another 2 further down – do nothing
            # if the nearest 2 is not directly below, do nothing
        else:
            # no 2 below – fill ordinary cells below the block
            down = r + 1
            while down < R and g[down][c] != 0:
                g[down][c] = 8
                down += 1

    return g
