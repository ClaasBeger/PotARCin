2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    """
    Transform the input grid according to the discovered rule.

    The rule:
    - The grid contains a unique component of colour 5 that forms a corner
      (full bottom row and right column of its bounding box).
    - Exactly one other component has the same “corner” shape (bottom‑right cell
      present and at least one neighbour to its left or above).  This component
      lies in the region that is the mirror image of the 5‑component’s bounding
      box (mirrored both vertically and horizontally, or only vertically, or only
      horizontally).  The first mirror that contains such a component is used.
    - All cells of that component are recoloured to 5.
    """
    from collections import deque

    if not grid:
        return grid

    n = len(grid)
    m = len(grid[0])

    # ------------------------------------------------------------------
    # find all connected components (4‑neighbourhood)
    # ------------------------------------------------------------------
    visited = [[False] * m for _ in range(n)]
    components = []  # each element: {'color': int, 'cells': [(r,c)], 'bbox': (rmin,rmax,cmin,cmax)}

    for i in range(n):
        for j in range(m):
            if grid[i][j] != 0 and not visited[i][j]:
                col = grid[i][j]
                q = deque([(i, j)])
                visited[i][j] = True
                cells = []
                while q:
                    r, c = q.popleft()
                    cells.append((r, c))
                    for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nr, nc = r + dr, c + dc
                        if 0 <= nr < n and 0 <= nc < m and not visited[nr][nc] and grid[nr][nc] == col:
                            visited[nr][nc] = True
                            q.append((nr, nc))
                rmin = min(r for r, _ in cells)
                rmax = max(r for r, _ in cells)
                cmin = min(c for _, c in cells)
                cmax = max(c for _, c in cells)
                components.append({
                    'color': col,
                    'cells': cells,
                    'bbox': (rmin, rmax, cmin, cmax)
                })

    # ------------------------------------------------------------------
    # locate the 5‑component
    # ------------------------------------------------------------------
    comp5 = None
    for comp in components:
        if comp['color'] == 5:
            comp5 = comp
            break

    if comp5 is None:
        # no 5 component – nothing to do
        return grid

    rmin5, rmax5, cmin5, cmax5 = comp5['bbox']

    # ------------------------------------------------------------------
    # helper: does a component have the required corner shape?
    # ------------------------------------------------------------------
    def is_corner(comp):
        cells_set = set(comp['cells'])
        rmin, rmax, cmin, cmax = comp['bbox']
        # bottom‑right must be present
        if (rmax, cmax) not in cells_set:
            return False
        # at least one neighbour to the left or above the bottom‑right cell
        left = False
        above = False
        if cmax > cmin:
            left = (rmax, cmax - 1) in cells_set
        if rmax > rmin:
            above = (rmax - 1, cmax) in cells_set
        return left or above

    # ------------------------------------------------------------------
    # try the three possible mirrored positions (both axes, vertical, horizontal)
    # ------------------------------------------------------------------
    transforms = [
        ('both', lambda: (n - 1 - rmax5, n - 1 - rmin5, m - 1 - cmax5, m - 1 - cmin5)),
        ('vertical', lambda: (rmin5, rmax5, m - 1 - cmax5, m - 1 - cmin5)),
        ('horizontal', lambda: (n - 1 - rmax5, n - 1 - rmin5, cmin5, cmax5)),
    ]

    target = None
    for name, region_fn in transforms:
        r1, r2, c1, c2 = region_fn()
        # keep region inside the grid
        r1 = max(0, r1)
        r2 = min(n - 1, r2)
        c1 = max(0, c1)
        c2 = min(m - 1, c2)

        best_comp = None
        best_cnt = 0
        for comp in components:
            if comp['color'] == 5:
                continue
            if not is_corner(comp):
                continue
            cnt = 0
            for r, c in comp['cells']:
                if r1 <= r <= r2 and c1 <= c <= c2:
                    cnt += 1
            if cnt > best_cnt:
                best_cnt = cnt
                best_comp = comp

        if best_comp is not None:
            target = best_comp
            break

    # ------------------------------------------------------------------
    # recolour the target component to 5 (if any)
    # ------------------------------------------------------------------
    if target is not None:
        for r, c in target['cells']:
            grid[r][c] = 5

    return grid
