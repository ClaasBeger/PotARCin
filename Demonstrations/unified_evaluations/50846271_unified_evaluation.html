Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0]) if h else 0

    row_twos = {r: [c for c in range(w) if grid[r][c] == 2] for r in range(h)}
    col_twos = {c: [r for r in range(h) if grid[r][c] == 2] for c in range(w)}

    candidates = {}

    for r in range(h):
        if len(row_twos[r]) < 2:
            continue
        for c in range(w):
            if len(col_twos[c]) < 2:
                continue

            covered = set()
            for cc in row_twos[r]:
                covered.add((r, cc))
            for rr in col_twos[c]:
                covered.add((rr, c))

            if len(covered) < 4:
                continue

            # all covered cells must be actual 2s
            if not all(grid[rr][cc] == 2 for rr, cc in covered):
                continue

            # arm length determined by farthest 2 on center row/col
            L = 0
            for rr, cc in covered:
                L = max(L, abs(rr - r), abs(cc - c))

            # reject if any covered 2 lies outside the plus span
            ok = True
            for rr, cc in covered:
                if rr == r:
                    if abs(cc - c) > L:
                        ok = False
                        break
                elif cc == c:
                    if abs(rr - r) > L:
                        ok = False
                        break
                else:
                    ok = False
                    break
            if not ok:
                continue

            key = frozenset(covered)
            prev = candidates.get(key)
            if prev is None or L > prev[2]:
                candidates[key] = (r, c, L)

    out = [row[:] for row in grid]

    for covered, (r, c, L) in candidates.items():
        for cc in range(c - L, c + L + 1):
            if 0 <= cc < w and out[r][cc] == 5:
                out[r][cc] = 8
        for rr in range(r - L, r + L + 1):
            if 0 <= rr < h and out[rr][c] == 5:
                out[rr][c] = 8

    return out
