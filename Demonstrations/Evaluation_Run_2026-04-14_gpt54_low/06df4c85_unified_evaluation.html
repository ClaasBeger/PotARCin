2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    h = len(grid)
    w = len(grid[0])

    # Find separator color: the color that forms full rows and full columns.
    row_full = {}
    for r in range(h):
        if all(grid[r][c] == grid[r][0] for c in range(w)):
            v = grid[r][0]
            row_full[v] = row_full.get(v, 0) + 1

    col_full = {}
    for c in range(w):
        v = grid[0][c]
        ok = True
        for r in range(h):
            if grid[r][c] != v:
                ok = False
                break
        if ok:
            col_full[v] = col_full.get(v, 0) + 1

    sep_color = None
    best = -1
    for v in set(row_full) | set(col_full):
        score = row_full.get(v, 0) + col_full.get(v, 0)
        if score > best:
            best = score
            sep_color = v

    # Split into content row/col bands between separator rows/cols.
    sep_rows = [r for r in range(h) if all(grid[r][c] == sep_color for c in range(w))]
    sep_cols = []
    for c in range(w):
        if all(grid[r][c] == sep_color for r in range(h)):
            sep_cols.append(c)

    row_bands = []
    start = 0
    for r in sep_rows:
        if start < r:
            row_bands.append((start, r - 1))
        start = r + 1
    if start < h:
        row_bands.append((start, h - 1))

    col_bands = []
    start = 0
    for c in sep_cols:
        if start < c:
            col_bands.append((start, c - 1))
        start = c + 1
    if start < w:
        col_bands.append((start, w - 1))

    nr = len(row_bands)
    nc = len(col_bands)

    # Read cell colors.
    cells = [[0] * nc for _ in range(nr)]
    for i, (r0, r1) in enumerate(row_bands):
        for j, (c0, c1) in enumerate(col_bands):
            color = 0
            for r in range(r0, r1 + 1):
                for c in range(c0, c1 + 1):
                    v = grid[r][c]
                    if v != 0 and v != sep_color:
                        color = v
                        break
                if color:
                    break
            cells[i][j] = color

    # Group positions by color.
    positions = {}
    for i in range(nr):
        for j in range(nc):
            v = cells[i][j]
            if v != 0:
                positions.setdefault(v, []).append((i, j))

    # Connect same-colored cells horizontally within rows and vertically within columns.
    out_cells = [row[:] for row in cells]

    for color, pts in positions.items():
        by_row = {}
        by_col = {}
        for i, j in pts:
            by_row.setdefault(i, []).append(j)
            by_col.setdefault(j, []).append(i)

        for i, js in by_row.items():
            if len(js) >= 2:
                a, b = min(js), max(js)
                for j in range(a, b + 1):
                    if out_cells[i][j] in (0, color):
                        out_cells[i][j] = color

        for j, is_ in by_col.items():
            if len(is_) >= 2:
                a, b = min(is_), max(is_)
                for i in range(a, b + 1):
                    if out_cells[i][j] in (0, color):
                        out_cells[i][j] = color

    # Write back to full grid.
    out = [row[:] for row in grid]
    for i, (r0, r1) in enumerate(row_bands):
        for j, (c0, c1) in enumerate(col_bands):
            color = out_cells[i][j]
            fill = color if color != 0 else 0
            for r in range(r0, r1 + 1):
                for c in range(c0, c1 + 1):
                    out[r][c] = fill

    # Restore separator rows/cols exactly.
    for r in sep_rows:
        for c in range(w):
            out[r][c] = sep_color
    for c in sep_cols:
        for r in range(h):
            out[r][c] = sep_color

    return out
