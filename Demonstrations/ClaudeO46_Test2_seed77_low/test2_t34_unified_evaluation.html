def solve(grid):
    R = len(grid)
    C = len(grid[0])
    
    eight_cells = set()
    two_pos = None
    for r in range(R):

                return None

            for c in range(C):

                    continue

                    continue

                    return None

        for r in range(min_r, max_r+1):

                nr, nc = tf(r, c)
                result[r-min_r][c-min_c] = grid[nr][nc]
        return result
