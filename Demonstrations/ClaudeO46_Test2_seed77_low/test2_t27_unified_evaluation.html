Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows, cols = len(grid), len(grid[0])
    visited = [[False]*cols for _ in range(rows)]
    shape_colors, shape_cells = [], {}
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] != 0 and not visited[r][c]:
                color, cells, stack = grid[r][c], [], [(r,c)]
                visited[r][c] = True
                while stack:
                    cr,cc = stack.pop(); cells.append((cr,cc))
                    for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                        nr,nc = cr+dr, cc+dc
                        if 0<=nr<rows and 0<=nc<cols and not visited[nr][nc] and grid[nr][nc]==color:
                            visited[nr][nc]=True; stack.append((nr,nc))
                if color in shape_cells: shape_cells[color].extend(cells)
                else: shape_cells[color]=cells; shape_colors.append(color)
    def normalize(cells):
        mr=min(r for r,c in cells); mc=min(c for r,c in cells)
        return frozenset((r-mr,c-mc) for r,c in cells)
    def get_rotations(norm):
        rots, cur = [], set(norm)
        for _ in range(4):
            mr=min(r for r,c in cur); mc=min(c for r,c in cur)
            rots.append(frozenset((r-mr,c-mc) for r,c in cur))
            cur={(c,-r) for r,c in cur}
        return rots
    all_rots={}
    for color in shape_colors:
        rots=get_rotations(normalize(shape_cells[color])); seen=set(); u=[]
        for i,rot in enumerate(rots):
            if rot not in seen: seen.add(rot); u.append((i,rot))
        all_rots[color]=u
    def try_tile(rh,rw,pcs):
        board=[[0]*rw for _ in range(rh)]; sol=[None]*len(pcs); used=[False]*len(pcs)
        def ffe():
            for r in range(rh):
                for c in range(rw):
                    if board[r][c]==0: return(r,c)
            return None
        def bt():
            pos=ffe()
            if not pos: return True
            r0,c0=pos
            for i in range(len(pcs)):
                if used[i]: continue
                color,rotations=pcs[i]
                for oi,rot in rotations:
                    a=min(rot); ro,co=r0-a[0],c0-a[1]; ok=True
                    for cr,cc in rot:
                        rr,cc2=cr+ro,cc+co
                        if rr<0 or rr>=rh or cc2<0 or cc2>=rw or board[rr][cc2]!=0: ok=False; break
                    if ok:
                        for cr,cc in rot: board[cr+ro][cc+co]=color
                        used[i]=True; sol[i]=oi
                        if bt(): return True
                        for cr,cc in rot: board[cr+ro][cc+co]=0
                        used[i]=False
            return False
        return ([r[:] for r in board],list(sol)) if bt() else (None,None)
    def rcw(b):
        h,w=len(b),len(b[0])
        return [[b[h-1-c][r] for c in range(h)] for r in range(w)]
    def place(gc,board,sol,out):
        pcs=[(c,all_rots[c]) for c in gc]
        for idx,color in enumerate(gc):
            t=[r[:] for r in board]
            for _ in range((4-sol[idx])%4): t=rcw(t)
            th,tw=len(t),len(t[0])
            tc=sorted([(r,c) for r in range(th) for c in range(tw) if t[r][c]==color])
            oc=sorted(shape_cells[color])
            dr,dc=oc[0][0]-tc[0][0],oc[0][1]-tc[0][1]
            for r in range(th):
                for c in range(tw):
                    gr,gc2=r+dr,c+dc
                    if 0<=gr<rows and 0<=gc2<cols: out[gr][gc2]=t[r][c]
    def find(gc):
        total=sum(len(shape_cells[c]) for c in gc); pcs=[(c,all_rots[c]) for c in gc]
        dims=[]
        for h in range(1,total+1):
            if total%h==0:
                w=total//h
                if h<=w: dims.append((h,w))
        dims.sort(key=lambda d:d[1]-d[0])
        for h,w in dims:
            b,s=try_tile(h,w,pcs)
            if b: return b,s
            if h!=w:
                b,s=try_tile(w,h,pcs)
                if b: return b,s
        return None,None

            b1,s1=find(g1)
            if not b1: continue
            b2,s2=find(g2)
            if not b2: continue
            place(g1,b1,s1,out); place(g2,b2,s2,out); return out
    return grid
