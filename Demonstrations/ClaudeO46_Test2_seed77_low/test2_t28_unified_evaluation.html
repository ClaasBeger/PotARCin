Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    rows = len(grid)
    cols = len(grid[0])
    g = copy.deepcopy(grid)
    
    from collections import Counter
    color_counts = Counter()
    for r in range(rows):
        for c in range(cols):
            if g[r][c] not in (0, 5):
                color_counts[g[r][c]] += 1
    
    special_val = None
    shape_val = None
    for color, count in color_counts.items():
        if count == 1:
            special_val = color
        else:
            shape_val = color
    
    special_pos = None
    fives = []
    for r in range(rows):
        for c in range(cols):
            if g[r][c] == special_val:
                special_pos = (r, c)
            elif g[r][c] == 5:
                fives.append((r, c))
    
    fives.sort(key=lambda p: abs(p[0]-special_pos[0]) + abs(p[1]-special_pos[1]))
    nearest = fives[0]
    dr = nearest[0] - special_pos[0]
    dc = nearest[1] - special_pos[1]
    if dr: dr = dr // abs(dr)
    if dc: dc = dc // abs(dc)
    
    for r, c in fives:
        g[r][c] = 0
    g[special_pos[0]][special_pos[1]] = 0
    
    last_five = fives[-1]
    pr, pc = last_five
    
    for _ in range(10000):
        nr, nc = pr + dr, pc + dc
        if nr < 0 or nr >= rows:
            dr = -dr
            nr = pr + dr
        if nc < 0 or nc >= cols:
            dc = -dc
            nc = pc + dc
        
        if g[nr][nc] == shape_val:
            g[pr][pc] = special_val
            break
        
        pr, pc = nr, nc
    
    return g
