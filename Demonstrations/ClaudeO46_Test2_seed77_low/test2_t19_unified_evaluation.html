2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    fives = []
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 5:
                fives.append((r, c))
    
    r1, c1 = fives[0]
    r2, c2 = fives[1]
    
    def side(r, c):
        return (r2 - r1) * (c - c1) - (c2 - c1) * (r - r1)
    
    pos_cells = []
    neg_cells = []
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] not in (0, 5):
                s = side(r, c)
                if s > 0:
                    pos_cells.append((r, c, grid[r][c]))
                elif s < 0:
                    neg_cells.append((r, c, grid[r][c]))
    
    out = [[0] * cols for _ in range(rows)]
    for r, c in fives:
        out[r][c] = 5
    
    for cells in [pos_cells, neg_cells]:
        if not cells:
            continue
        colors = set(color for _, _, color in cells)
        all_r = [r for r, _, _ in cells]
        all_c = [c for _, c, _ in cells]
        min_r, max_r = min(all_r), max(all_r)
        min_c, max_c = min(all_c), max(all_c)
        
        if len(colors) > 1:
            for r in range(min_r, max_r + 1):
                for c in range(min_c, max_c + 1):
                    out[r][c] = 5
        else:
            color = colors.pop()
            h = max_r - min_r + 1
            w = max_c - min_c + 1
            
            if h == 1 or w == 1:
                size = max(h, w)
                if h == 1:
                    centroid_c = (min_c + max_c) / 2
                    d1 = abs(min_r - r1) + abs(centroid_c - c1)
                    d2 = abs(min_r - r2) + abs(centroid_c - c2)
                    far_5 = fives[1] if d2 > d1 else fives[0]
                    if far_5[0] < min_r:
                        min_r = max_r - size + 1
                    else:
                        max_r = min_r + size - 1
                elif w == 1:
                    centroid_r = (min_r + max_r) / 2
                    d1 = abs(centroid_r - r1) + abs(min_c - c1)
                    d2 = abs(centroid_r - r2) + abs(min_c - c2)
                    far_5 = fives[1] if d2 > d1 else fives[0]
                    if far_5[1] > min_c:
                        max_c = min_c + size - 1
                    else:
                        min_c = max_c - size + 1
            
            for r in range(min_r, max_r + 1):
                for c in range(min_c, max_c + 1):
                    out[r][c] = color
    
    return out
