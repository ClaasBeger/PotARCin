2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    visited = [[False]*cols for _ in range(rows)]
    components = []
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] != 0 and grid[r][c] != 9 and not visited[r][c]:
                color = grid[r][c]
                stack = [(r, c)]; visited[r][c] = True; cells = []
                while stack:
                    cr, cc = stack.pop(); cells.append((cr, cc))
                    for dr in [-1,0,1]:
                        for dc in [-1,0,1]:
                            if dr==0 and dc==0: continue
                            nr, nc = cr+dr, cc+dc
                            if 0<=nr<rows and 0<=nc<cols and not visited[nr][nc] and grid[nr][nc]==color:
                                visited[nr][nc] = True; stack.append((nr, nc))
                components.append((color, cells))
    
    shapes = sorted([c for c in components if len(c[1])>=3], key=lambda x:-len(x[1]))
    if len(shapes) < 1: return [r[:] for r in grid]
    
    nine_pos = {(r,c) for r in range(rows) for c in range(cols) if grid[r][c]==9}
    nz = {(r,c) for r in range(rows) for c in range(cols) if grid[r][c]!=0 and grid[r][c]!=9}
    
    def find_gap():
        for r in range(rows):
            if all(grid[r][c]==0 or grid[r][c]==9 for c in range(cols)):
                above = {p for p in nz if p[0]<r}; below = {p for p in nz if p[0]>r}
                if above and below: return 'row', r
        for c in range(cols):
            if all(grid[r][c]==0 or grid[r][c]==9 for r in range(rows)):
                left = {p for p in nz if p[1]<c}; right = {p for p in nz if p[1]>c}
                if left and right: return 'col', c
        for k in range(rows+cols):
            pts = [(r,k-r) for r in range(rows) if 0<=k-r<cols]
            if pts and all(grid[r][c]==0 or grid[r][c]==9 for r,c in pts):
                lo = {p for p in nz if p[0]+p[1]<k}; hi = {p for p in nz if p[0]+p[1]>k}
                if lo and hi: return 'dsum', k
        return None, None
    
    gt, gv = find_gap()
    if gt is None: return [r[:] for r in grid]
    
    def side(cells, gt, gv):
        if gt=='row': return 1 if sum(r for r,c in cells)/len(cells)>gv else -1
        if gt=='col': return 1 if sum(c for r,c in cells)/len(cells)>gv else -1
        if gt=='dsum': return 1 if sum(r+c for r,c in cells)/len(cells)>gv else -1
    
    def dist9(cells):
        if not nine_pos: return float('inf')
        return min(abs(r-nr)+abs(c-nc) for r,c in cells for nr,nc in nine_pos)
    
    moving = shapes[0]
    if len(shapes)>=2:
        if dist9(shapes[0][1])<=dist9(shapes[1][1]): moving=shapes[0]
        else: moving=shapes[1]
    
    mc = set(moving[1])
    s = side(moving[1], gt, gv)
    if gt=='row': step=(-s,0)
    elif gt=='col': step=(0,-s)
    else: step=(-s,-s)
    sr,sc=step
    
    max_n=0
    for n in range(1,rows+cols):
        shr,shc=n*sr,n*sc
        if not all(0<=r+shr<rows and 0<=c+shc<cols for r,c in mc): break
        tmp=[row[:] for row in grid]
        for r,c in mc: tmp[r][c]=0
        nps=[(r+shr,c+shc,grid[r][c]) for r,c in mc]
        nps.sort(key=lambda x:-(x[0]*sr+x[1]*sc))
        ok=True
        for nr,nc,val in nps:
            if tmp[nr][nc]!=0:
                cr,cc=nr,nc
                while 0<=cr<rows and 0<=cc<cols and tmp[cr][cc]!=0: cr+=sr;cc+=sc
                if not(0<=cr<rows and 0<=cc<cols): ok=False;break
                while(cr,cc)!=(nr,nc):
                    pr,pc=cr-sr,cc-sc;tmp[cr][cc]=tmp[pr][pc];cr,cc=pr,pc
            tmp[nr][nc]=val
        if ok: max_n=n
        else: break
    
    if max_n==0: return [r[:] for r in grid]
    shr,shc=max_n*sr,max_n*sc
    res=[row[:] for row in grid]
    for r,c in mc: res[r][c]=0
    nps=[(r+shr,c+shc,grid[r][c]) for r,c in mc]
    nps.sort(key=lambda x:-(x[0]*sr+x[1]*sc))
    for nr,nc,val in nps:
        if res[nr][nc]!=0:
            cr,cc=nr,nc
            while 0<=cr<rows and 0<=cc<cols and res[cr][cc]!=0: cr+=sr;cc+=sc
            while(cr,cc)!=(nr,nc):
                pr,pc=cr-sr,cc-sc;res[cr][cc]=res[pr][pc];cr,cc=pr,pc
        res[nr][nc]=val
    return res
