def solve(grid):
    # Find the separator row (all 5s)
    sep = next(i for i, row in enumerate(grid) if all(v == 5 for v in row))
    
    # Find the separator column (all 0s in top half, or column of 0s)
    cols = len(grid[0])
    sep_col = next(c for c in range(cols) if all(grid[r][c] == 0 for r in range(sep)))
    
    # Build mapping from top half: left_val <-> right_val
    left_to_right = {}
    right_to_left = {}
    for r in range(sep):
        lv = grid[r][0]
        rv = grid[r][sep_col + 1]
        left_to_right[lv] = rv
        right_to_left[rv] = lv
    
    result = [row[:] for row in grid]
    
    for r in range(sep + 1, len(grid)):
        left_vals = [grid[r][c] for c in range(sep_col) if grid[r][c] != 0]
        right_vals = [grid[r][c] for c in range(sep_col + 1, cols) if grid[r][c] != 0]
        
        left_zero = all(grid[r][c] == 0 for c in range(sep_col))
        right_zero = all(grid[r][c] == 0 for c in range(sep_col + 1, cols))
        
        if left_zero and right_zero:
            pass
        elif left_zero and not right_zero:
            rv = right_vals[0]
            lv = right_to_left[rv]
            for c in range(sep_col):
                result[r][c] = lv
        elif right_zero and not left_zero:
            lv = left_vals[0]
            rv = left_to_right[lv]
            for c in range(sep_col + 1, cols):
                result[r][c] = rv
    
    return result
