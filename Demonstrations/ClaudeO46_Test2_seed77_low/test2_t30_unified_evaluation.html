2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    sep = [i for i in range(rows) if all(grid[i][c] == 6 for c in range(cols))]
    
    sec_bounds = []
    prev = 0
    for s in sep:
        sec_bounds.append((prev, s))
        prev = s + 1
    sec_bounds.append((prev, rows))
    
    n = len(sec_bounds)
    sr = sec_bounds[0][1] - sec_bounds[0][0]
    
    color = 0
    sec_cells = []
    for start, end in sec_bounds:
        cells = set()
        for r in range(start, end):
            for c in range(cols):
                if grid[r][c] not in (0, 6):
                    cells.add((r - start, c))
                    color = grid[r][c]
        sec_cells.append(cells)
    
    empty = [i for i in range(n) if not sec_cells[i]]
    nonempty = sorted(i for i in range(n) if sec_cells[i])
    result = [s.copy() for s in sec_cells]
    
    if not empty:
        return grid
    
    def nb4(r, c):
        return [(r-1,c),(r+1,c),(r,c-1),(r,c+1)]
    def nb8(r, c):
        return [(r+dr,c+dc) for dr in [-1,0,1] for dc in [-1,0,1] if dr or dc]
    
    def bfs(seed, allowed, nf):
        dist = {c:0 for c in seed}
        q = list(seed)
        qi = 0
        while qi < len(q):
            cell = q[qi]; qi += 1
            for nr, nc in nf(*cell):
                if (nr,nc) in allowed and (nr,nc) not in dist:
                    dist[(nr,nc)] = dist[cell] + 1
                    q.append((nr,nc))
        return dist
    
    all_grid = {(r,c) for r in range(sr) for c in range(cols)}
    solved = False
    
    is_cum = len(nonempty) >= 2 and all(
        sec_cells[nonempty[i]].issubset(sec_cells[nonempty[i+1]])
        for i in range(len(nonempty)-1))
    
    if is_cum:
        target = sec_cells[nonempty[-1]]
        for nf in [nb4, nb8]:
            for root in sorted(target):
                dist = bfs({root}, target, nf)
                if len(dist) < len(target): continue
                d_map = {}
                ok = True
                for idx in nonempty:
                    mx = max(dist[c] for c in sec_cells[idx])
                    if {c for c in target if dist[c] <= mx} != sec_cells[idx]:
                        ok = False; break
                    d_map[idx] = mx
                if not ok: continue
                
                for idx in empty:
                    p = max((i for i in nonempty if i < idx), default=None)
                    nx = min((i for i in nonempty if i > idx), default=None)
                    if p is not None and nx is not None:
                        dk = d_map[p]+(d_map[nx]-d_map[p])*(idx-p)/(nx-p)
                    elif p is not None:
                        r2 = (d_map[nonempty[-1]]-d_map[nonempty[-2]])/(nonempty[-1]-nonempty[-2]) if len(nonempty)>=2 else 1
                        dk = d_map[p]+r2*(idx-p)
                    else:
                        r2 = (d_map[nonempty[1]]-d_map[nonempty[0]])/(nonempty[1]-nonempty[0]) if len(nonempty)>=2 else 1
                        dk = d_map[nx]-r2*(nx-idx)
                    dk = int(dk+1e-9)
                    result[idx] = {c for c in target if dist[c] <= max(0,dk)}
                solved = True; break
            if solved: break
    
    if not solved:
        seed = sec_cells[nonempty[0]]
        for nf in [nb4, nb8]:
            dist = bfs(seed, all_grid, nf)
            if all({c for c,d in dist.items() if d==idx} == sec_cells[idx] for idx in nonempty):
                for idx in empty:
                    result[idx] = {c for c,d in dist.items() if d==idx}
                solved = True; break
    
    if not solved:
        all_known = set()
        for i in nonempty: all_known |= sec_cells[i]
        seed = sec_cells[nonempty[0]]
        for nf in [nb4, nb8]:
            dist = bfs(seed, all_known, nf)
            if all({c for c,d in dist.items() if d==idx} == sec_cells[idx] for idx in nonempty):
                max_d = max(dist.values()) if dist else 0
                cr, cc = (sr-1)/2, (cols-1)/2
                for idx in empty:
                    if idx <= max_d:
                        result[idx] = {c for c,d in dist.items() if d==idx}
                    else:
                        last_ring = {c for c,d in dist.items() if d==max_d}
                        cum = set(dist.keys())
                        for step in range(idx - max_d):
                            new_ring = set()
                            for cell in last_ring:
                                for nr, nc in nf(*cell):
                                    if 0<=nr<sr and 0<=nc<cols and (nr,nc) not in cum:
                                        d_old = abs(cell[0]-cr)+abs(cell[1]-cc)
                                        d_new = abs(nr-cr)+abs(nc-cc)
                                        if d_new >= d_old:
                                            new_ring.add((nr,nc))
                            cum |= new_ring
                            last_ring = new_ring
                        result[idx] = last_ring
                solved = True; break
    
    output = [row[:] for row in grid]
    for idx in empty:
        start = sec_bounds[idx][0]
        for r in range(sec_bounds[idx][1]-sec_bounds[idx][0]):
            for c in range(cols):
                output[start+r][c] = color if (r,c) in result[idx] else 0
    return output
