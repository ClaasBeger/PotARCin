Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    grid = [row[:] for row in grid]
    rows, cols = len(grid), len(grid[0])
    
    # Find two groups of 2s
    groups = []  # each: (set of (r,c), orientation)
    
    # Find connected segments of 2s
    twos = set()
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 2:
                twos.add((r, c))
    
    visited = set()
    for t in twos:
        if t in visited:
            continue
        # BFS
        group = set()
        queue = [t]
        while queue:
            cur = queue.pop()
            if cur in visited:
                continue
            visited.add(cur)
            group.add(cur)
            r, c = cur
            for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                nr, nc = r+dr, c+dc
                if (nr, nc) in twos and (nr, nc) not in visited:
                    queue.append((nr, nc))
        groups.append(group)
    
    # For each group, determine: which side is the wall, row/col range
    def analyze_group(g):
        rs = [r for r, c in g]
        cs = [c for r, c in g]
        if len(set(cs)) == 1:  # vertical wall
            col = cs[0]
            min_r, max_r = min(rs), max(rs)
            side = 'left' if col == 0 else 'right'
            return {'type': 'vertical', 'col': col, 'min_r': min_r, 'max_r': max_r, 'side': side}
        else:
            row = rs[0]
            min_c, max_c = min(cs), max(cs)
            side = 'top' if row == 0 else 'bottom'
            return {'type': 'horizontal', 'row': row, 'min_c': min_c, 'max_c': max_c, 'side': side}
    
    infos = [analyze_group(g) for g in groups]
    
    # Find 8s and associate with groups
    eights = []
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 8:
                eights.append((r, c))
    
    # Determine which group has 8s nearby
    for i, info in enumerate(infos):
        info['eights'] = []
        for r8, c8 in eights:
            if info['type'] == 'vertical':
                if info['min_r'] <= r8 <= info['max_r']:
                    info['eights'].append((r8, c8))
    
    # Process
    source = None
    target = None
    for i, info in enumerate(infos):
        if info['eights']:
            source = i
        else:
            target = i
    
    if source is None or target is None:
        return grid
    
    si = infos[source]
    ti = infos[target]
    
    for r8, c8 in si['eights']:
        rel = r8 - si['min_r']
        
        # Fill source: from wall to 8 with 8s, 8 position becomes 4
        if si['type'] == 'vertical':
            if si['side'] == 'left':
                for c in range(si['col'] + 1, c8):
                    grid[r8][c] = 8
                grid[r8][c8] = 4
            else:
                for c in range(c8 + 1, si['col']):
                    grid[r8][c] = 8
                grid[r8][c8] = 4
        
        # Fill target: corresponding row fully with 8s
        tr = ti['min_r'] + rel
        if ti['type'] == 'vertical':
            if ti['side'] == 'left':
                for c in range(ti['col'] + 1, cols):
                    grid[tr][c] = 8
            else:
                for c in range(0, ti['col']):
                    grid[tr][c] = 8
    
    return grid
