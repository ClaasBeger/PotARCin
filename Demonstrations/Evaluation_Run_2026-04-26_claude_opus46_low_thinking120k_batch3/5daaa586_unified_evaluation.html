Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    h_lines = {}
    for r in range(rows):
        counts = {}
        for c in range(cols):
            v = grid[r][c]
            if v != 0:
                counts[v] = counts.get(v, 0) + 1
        for color, cnt in counts.items():
            if cnt >= cols * 0.6:
                h_lines[r] = color
                break
    
    v_lines = {}
    for c in range(cols):
        counts = {}
        for r in range(rows):
            v = grid[r][c]
            if v != 0:
                counts[v] = counts.get(v, 0) + 1
        for color, cnt in counts.items():
            if cnt >= rows * 0.6:
                v_lines[c] = color
                break
    
    h_rows = sorted(h_lines.keys())
    v_cols = sorted(v_lines.keys())
    r1, r2 = h_rows[0], h_rows[1]
    c1, c2 = v_cols[0], v_cols[1]
    
    line_positions = set()
    for r in h_rows:
        for c in range(cols):
            line_positions.add((r, c))
    for c in v_cols:
        for r in range(rows):
            line_positions.add((r, c))
    
    line_colors = set(h_lines.values()) | set(v_lines.values())
    marker_color = None
    for color in line_colors:
        found = False
        for r in range(rows):
            for c in range(cols):
                if grid[r][c] == color and (r, c) not in line_positions:
                    marker_color = color
                    found = True
                    break
            if found:
                break
        if found:
            break
    
    if h_lines[r1] == marker_color:
        gravity = 'up'
    elif h_lines[r2] == marker_color:
        gravity = 'down'
    elif v_lines[c1] == marker_color:
        gravity = 'left'
    else:
        gravity = 'right'
    
    out_rows = r2 - r1 + 1
    out_cols = c2 - c1 + 1
    output = [[0] * out_cols for _ in range(out_rows)]
    
    for oc in range(out_cols):
        output[0][oc] = grid[r1][c1 + oc]
        output[out_rows - 1][oc] = grid[r2][c1 + oc]
    for or_ in range(out_rows):
        output[or_][0] = grid[r1 + or_][c1]
        output[or_][out_cols - 1] = grid[r1 + or_][c2]
    
    interior_rows = range(r1 + 1, r2)
    interior_cols = range(c1 + 1, c2)
    
    if gravity == 'down':
        for ic, c in enumerate(interior_cols):
            topmost = None
            for r in interior_rows:
                if grid[r][c] == marker_color:
                    topmost = r
                    break
            if topmost is not None:
                for r in range(topmost, r2):
                    output[r - r1][ic + 1] = marker_color
    elif gravity == 'up':
        for ic, c in enumerate(interior_cols):
            bottommost = None
            for r in reversed(list(interior_rows)):
                if grid[r][c] == marker_color:
                    bottommost = r
                    break
            if bottommost is not None:
                for r in range(r1 + 1, bottommost + 1):
                    output[r - r1][ic + 1] = marker_color
    elif gravity == 'right':
        for ir, r in enumerate(interior_rows):
            leftmost = None
            for c in interior_cols:
                if grid[r][c] == marker_color:
                    leftmost = c
                    break
            if leftmost is not None:
                for c in range(leftmost, c2):
                    output[ir + 1][c - c1] = marker_color
    elif gravity == 'left':
        for ir, r in enumerate(interior_rows):
            rightmost = None
            for c in reversed(list(interior_cols)):
                if grid[r][c] == marker_color:
                    rightmost = c
                    break
            if rightmost is not None:
                for c in range(c1 + 1, rightmost + 1):
                    output[ir + 1][c - c1] = marker_color
    
    return output
