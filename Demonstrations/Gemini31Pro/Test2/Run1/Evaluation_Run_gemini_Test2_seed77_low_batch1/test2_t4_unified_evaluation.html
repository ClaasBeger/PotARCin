2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    # Find the 2-box
    min_r, max_r, min_c, max_c = rows, -1, cols, -1
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 2:
                min_r = min(min_r, r)
                max_r = max(max_r, r)
                min_c = min(min_c, c)
                max_c = max(max_c, c)
                
    # Extract the interior of the 2-box
    int_min_r = min_r + 1
    int_max_r = max_r - 1
    int_min_c = min_c + 1
    int_max_c = max_c - 1
    
    int_h = int_max_r - int_min_r + 1
    int_w = int_max_c - int_min_c + 1
    
    # Scale factor
    k_r = rows // int_h
    k_c = cols // int_w
    
    out_grid = [[0 for _ in range(cols)] for _ in range(rows)]
    
    for ir in range(int_h):
        for ic in range(int_w):
            color = grid[int_min_r + ir][int_min_c + ic]
            if color == 0:
                continue
                
            # Draw the KxK block
            r_start = ir * k_r
            r_end = r_start + k_r
            c_start = ic * k_c
            c_end = c_start + k_c
            
            # Draw color box
            for r in range(r_start, r_end):
                for c in range(c_start, c_end):
                    if r == r_start or r == r_end - 1 or c == c_start or c == c_end - 1:
                        out_grid[r][c] = color
                    else:
                        out_grid[r][c] = color # Fill interior first
            
            # Empty the interior
            for r in range(r_start + 1, r_end - 1):
                for c in range(c_start + 1, c_end - 1):
                    out_grid[r][c] = 0
                    
            # Draw the 2s
            # The 2s position inside the (K-2)x(K-2) interior matches the 2-box position in the original grid
            # Original grid is rows x cols. 2-box is min_r to max_r, min_c to max_c.
            # We map the grid to the interior of the box.
            in_h = k_r - 2
            in_w = k_c - 2
            
            # Calculate the bounding box for 2s
            # Scale min_r, max_r to 0..in_h
            start_r2 = r_start + 1 + (min_r * in_h) // rows
            end_r2 = r_start + 1 + ((max_r + 1) * in_h + rows - 1) // rows
            start_c2 = c_start + 1 + (min_c * in_w) // cols
            end_c2 = c_start + 1 + ((max_c + 1) * in_w + cols - 1) // cols
            
            # If the size is 0, make it 1
            if end_r2 == start_r2: end_r2 += 1
            if end_c2 == start_c2: end_c2 += 1
            
            for r in range(start_r2, end_r2):
                for c in range(start_c2, end_c2):
                    if r < r_end - 1 and c < c_end - 1:
                        out_grid[r][c] = 2
                        
    # Merge contiguous blocks of the same color
    for r in range(rows):
        for c in range(cols):
            if out_grid[r][c] == 0:
                continue
            color = out_grid[r][c]
            if color == 2: continue
            
            # Find boundaries to remove lines between merged blocks
            if r > 0 and out_grid[r-1][c] == color:
                pass # This requires complex merging, but standard grid overwriting usually solves this automatically
                
    # Re-fill the boxes properly to remove internal borders
    out_grid2 = [[0 for _ in range(cols)] for _ in range(rows)]
    
    # We will do a connected component for each color block in the scaled grid
    # For now, just fill based on the scaled interior
    for r in range(rows):
        for c in range(cols):
            ir = r // k_r
            ic = c // k_c
            color = grid[int_min_r + ir][int_min_c + ic]
            if color != 0:
                out_grid2[r][c] = color
                
    for r in range(rows):
        for c in range(cols):
            ir = r // k_r
            ic = c // k_c
            color = grid[int_min_r + ir][int_min_c + ic]
            if color != 0:
                # check if boundary of the whole color blob
                is_boundary = False
                if r == 0 or grid[int_min_r + (r-1)//k_r][int_min_c + c//k_c] != color:
                    if r % k_r == 0: is_boundary = True
                if r == rows - 1 or grid[int_min_r + (r+1)//k_r][int_min_c + c//k_c] != color:
                    if r % k_r == k_r - 1: is_boundary = True
                if c == 0 or grid[int_min_r + r//k_r][int_min_c + (c-1)//k_c] != color:
                    if c % k_c == 0: is_boundary = True
                if c == cols - 1 or grid[int_min_r + r//k_r][int_min_c + (c+1)//k_c] != color:
                    if c % k_c == k_c - 1: is_boundary = True
                    
                if not is_boundary:
                    out_grid2[r][c] = 0
                    
    # Place 2s
    for r in range(rows):
        for c in range(cols):
            ir = r // k_r
            ic = c // k_c
            color = grid[int_min_r + ir][int_min_c + ic]
            if color != 0:
                in_h = k_r - 2
                in_w = k_c - 2
                start_r2 = ir * k_r + 1 + (min_r * in_h) // rows
                end_r2 = ir * k_r + 1 + ((max_r + 1) * in_h + rows - 1) // rows
                start_c2 = ic * k_c + 1 + (min_c * in_w) // cols
                end_c2 = ic * k_c + 1 + ((max_c + 1) * in_w + cols - 1) // cols
                
                if start_r2 <= r < end_r2 and start_c2 <= c < end_c2:
                    out_grid2[r][c] = 2
                    
    return out_grid2
