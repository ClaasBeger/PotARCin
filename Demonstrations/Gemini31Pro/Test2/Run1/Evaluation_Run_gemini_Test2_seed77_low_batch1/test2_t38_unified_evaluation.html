2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    H = len(grid)
    W = len(grid[0])
    
    # Find all non-zero pixels and count their frequencies
    color_counts = {}
    non_zeros = []
    
    for r in range(H):
        for c in range(W):
            v = grid[r][c]
            if v != 0:
                color_counts[v] = color_counts.get(v, 0) + 1
                non_zeros.append((r, c, v))
                
    main_color = max(color_counts.keys(), key=lambda k: color_counts[k])
    
    # Try all possible block sizes
    best_r, best_c = H, W
    
    for r in range(1, H + 1):
        for c in range(1, W + 1):
            # Check if this could be the base fractal size
            # We assume the fractal has depth D > 1, so grid bounding box or size matches r^D x c^D
            # Actually, just check if the number of non-zero projected positions N satisfies N^D == len(non_zeros)
            
            projected = set()
            for x, y, v in non_zeros:
                projected.add((x % r, y % c))
                
            N = len(projected)
            if N == 0: continue
            
            # Check if there exists an integer D >= 1 such that N^D == len(non_zeros)
            # and r^D == bounding_box_height, etc. We can just use a simple heuristic:
            # D is log(len(non_zeros)) / log(N)
            D = 1
            n_pow = N
            while n_pow < len(non_zeros):
                n_pow *= N
                D += 1
                
            if n_pow == len(non_zeros) and D > 1:
                # Plausible base size
                best_r, best_c = r, c
                break
        if best_r != H:
            break

    if best_r == H and best_c == W:
        # Fallback if the clean power relation wasn't found
        for r in range(1, H):
            for c in range(1, W):
                projected = set((x % r, y % c) for x, y, v in non_zeros)
                if len(projected) > 0 and len(projected) * 2 <= r * c:
                    best_r, best_c = r, c
                    break
            if best_r != H:
                break

    output = [[0 for _ in range(best_c)] for _ in range(best_r)]
    
    # Fill with main color
    for x, y, v in non_zeros:
        output[x % best_r][y % best_c] = main_color
        
    # Overwrite with anomaly colors
    for x, y, v in non_zeros:
        if v != main_color:
            output[x % best_r][y % best_c] = v
            
    return output
