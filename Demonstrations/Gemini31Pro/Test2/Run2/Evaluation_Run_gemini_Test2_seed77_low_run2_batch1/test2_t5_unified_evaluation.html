2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    R = len(grid)
    C = len(grid[0])
    
    out = [row[:] for row in grid]
    
    # Simulate water (8) falling and flowing
    changed = True
    while changed:
        changed = False
        # Move down
        for r in range(R - 2, -1, -1):
            for c in range(C):
                if out[r][c] == 8 and out[r+1][c] == 0:
                    out[r][c] = 0
                    out[r+1][c] = 8
                    changed = True
                    
        if changed:
            continue
            
        # Spread horizontally to find lower ground or level out
        # To do this correctly, for each 8, we find if there's a reachable empty space 
        # in the same row that is supported, or can drop.
        # It's easier to just find the lowest available reachable cell.
        
        # We will try to move an 8 to an adjacent 0 in the same row if it helps it to drop 
        # or if it moves it to a "lower" level overall.
        # A simple BFS for each 8 to find the best target:
        # Best target: a 0 that is reachable by moving left/right and then optionally dropping down,
        # which is at a lower row, or same row but can level out.
        
        moves = []
        for r in range(R):
            for c in range(C):
                if out[r][c] == 8:
                    # Find reachable 0s by moving left/right
                    reachable_zeros = []
                    # check left
                    for dc in [-1, 1]:
                        nc = c + dc
                        while 0 <= nc < C and out[r][nc] in (0, 8):
                            if out[r][nc] == 0:
                                # check if it can drop from here, or just stop
                                drop_depth = 0
                                curr_r = r
                                while curr_r + 1 < R and out[curr_r+1][nc] == 0:
                                    curr_r += 1
                                    drop_depth += 1
                                reachable_zeros.append((curr_r, nc, drop_depth))
                            nc += dc
                    
                    if reachable_zeros:
                        # Sort by depth (descending), then distance
                        reachable_zeros.sort(key=lambda x: (-x[0], abs(x[1] - c)))
                        best_r, best_c, drop = reachable_zeros[0]
                        if best_r > r or (best_r == r and out[r][c] == 8): # It can drop!
                            # Wait, we should only move if it drops, OR to level out.
                            if best_r > r:
                                moves.append((r, c, best_r, best_c))
                                
        if moves:
            # Execute the best move that drops an 8
            moves.sort(key=lambda x: (-x[2], abs(x[3] - x[1]))) # move to lowest possible
            r, c, br, bc = moves[0]
            out[r][c] = 0
            out[br][bc] = 8
            changed = True
            continue
            
        # If no dropping moves, level out
        # For each row, if there are 8s and 0s connected horizontally, and 8s are stacked on top of 8s?
        # Actually, standard leveling: if there is an 8 at (r, c) and a 0 at (r, nc) reachable horizontally,
        # and moving it doesn't change depth, but reduces the "height" of the column.
        # Let's count heights.
        for r in range(R):
            for c in range(C):
                if out[r][c] == 8:
                    for dc in [-1, 1]:
                        nc = c + dc
                        while 0 <= nc < C and out[r][nc] in (0, 8):
                            if out[r][nc] == 0:
                                # move if height at c is greater than height at nc
                                # height is number of 8s above
                                h_c = 0
                                tr = r - 1
                                while tr >= 0 and out[tr][c] == 8:
                                    h_c += 1
                                    tr -= 1
                                    
                                h_nc = 0
                                tr = r - 1
                                while tr >= 0 and out[tr][nc] == 8:
                                    h_nc += 1
                                    tr -= 1
                                    
                                if h_c > 0 and h_nc == 0:
                                    # We can move the top of c to top of nc
                                    top_c = r - h_c
                                    out[top_c][c] = 0
                                    out[r][nc] = 8
                                    changed = True
                                    break
                            if changed: break
                            nc += dc
                        if changed: break
                if changed: break
            if changed: break

    return out
