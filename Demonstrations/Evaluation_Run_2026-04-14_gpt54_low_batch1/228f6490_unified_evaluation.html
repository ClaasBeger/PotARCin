2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import deque, defaultdict

    h = len(grid)
    w = len(grid[0]) if h else 0
    out = [row[:] for row in grid]

    def neighbors(r, c):
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < h and 0 <= nc < w:
                yield nr, nc

    def find_components(target_nonzero=True, exclude_color=None):
        vis = [[False] * w for _ in range(h)]
        comps = []
        for r in range(h):
            for c in range(w):
                val = grid[r][c]
                ok = (val != 0) if target_nonzero else (val == 0)
                if exclude_color is not None and val == exclude_color:
                    ok = False
                if ok and not vis[r][c]:
                    q = deque([(r, c)])
                    vis[r][c] = True
                    cells = []
                    colors = set()
                    while q:
                        cr, cc = q.popleft()
                        cells.append((cr, cc))
                        colors.add(grid[cr][cc])
                        for nr, nc in neighbors(cr, cc):
                            nval = grid[nr][nc]
                            nok = (nval != 0) if target_nonzero else (nval == 0)
                            if exclude_color is not None and nval == exclude_color:
                                nok = False
                            if nok and not vis[nr][nc]:
                                vis[nr][nc] = True
                                q.append((nr, nc))
                    comps.append((cells, colors))
        return comps

    def normalize(cells):
        min_r = min(r for r, c in cells)
        min_c = min(c for r, c in cells)
        return tuple(sorted((r - min_r, c - min_c) for r, c in cells))

    # Find all non-zero components.
    comps = find_components(target_nonzero=True)

    five_components = []
    other_components = []

    for cells, colors in comps:
        if colors == {5}:
            five_components.append(cells)
        elif len(colors) == 1:
            color = next(iter(colors))
            other_components.append({"cells": cells, "color": color, "shape": normalize(cells)})
        else:
            # Mixed-color components are not expected for movable shapes; ignore.
            pass

    # For each 5-component, find enclosed zero-hole components within its bounding box.
    holes = []
    for cells in five_components:
        rs = [r for r, c in cells]
        cs = [c for r, c in cells]
        r0, r1 = min(rs), max(rs)
        c0, c1 = min(cs), max(cs)

        five_set = set(cells)
        vis = set()

        for r in range(r0, r1 + 1):
            for c in range(c0, c1 + 1):
                if (r, c) in five_set or grid[r][c] != 0 or (r, c) in vis:
                    continue

                q = deque([(r, c)])
                vis.add((r, c))
                hole_cells = []
                touches_border = False

                while q:
                    cr, cc = q.popleft()
                    hole_cells.append((cr, cc))
                    if cr == r0 or cr == r1 or cc == c0 or cc == c1:
                        touches_border = True
                    for nr, nc in neighbors(cr, cc):
                        if r0 <= nr <= r1 and c0 <= nc <= c1:
                            if (nr, nc) not in five_set and grid[nr][nc] == 0 and (nr, nc) not in vis:
                                vis.add((nr, nc))
                                q.append((nr, nc))

                if not touches_border:
                    holes.append({"cells": hole_cells, "shape": normalize(hole_cells)})

    # Match each hole shape to a same-shaped non-5 component.
    shape_to_sources = defaultdict(list)
    for comp in other_components:
        shape_to_sources[comp["shape"]].append(comp)

    used_sources = set()

    for hole in holes:
        candidates = shape_to_sources.get(hole["shape"], [])
        source = None
        for comp in candidates:
            key = tuple(sorted(comp["cells"]))
            if key not in used_sources:
                source = comp
                used_sources.add(key)
                break
        if source is None:
            continue

        # Erase source.
        for r, c in source["cells"]:
            out[r][c] = 0

        # Paint hole with source color.
        color = source["color"]
        for r, c in hole["cells"]:
            out[r][c] = color

    return out
