2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    H = len(grid)
    W = len(grid[0])

    def sign(x):
        return (x > 0) - (x < 0)

    def neighbors(r, c):
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr, nc = r + dr, c + dc
            if 0 <= nr < H and 0 <= nc < W:
                yield nr, nc

    # Find connected components of same nonzero color
    seen = [[False] * W for _ in range(H)]
    comps = []
    for r in range(H):
        for c in range(W):
            if grid[r][c] != 0 and not seen[r][c]:
                color = grid[r][c]
                stack = [(r, c)]
                seen[r][c] = True
                cells = []
                while stack:
                    cr, cc = stack.pop()
                    cells.append((cr, cc))
                    for nr, nc in neighbors(cr, cc):
                        if not seen[nr][nc] and grid[nr][nc] == color:
                            seen[nr][nc] = True
                            stack.append((nr, nc))
                rs = [x for x, _ in cells]
                cs = [y for _, y in cells]
                r0, r1 = min(rs), max(rs)
                c0, c1 = min(cs), max(cs)
                comps.append({
                    "color": color,
                    "cells": cells,
                    "set": set(cells),
                    "r0": r0, "r1": r1, "c0": c0, "c1": c1,
                    "h": r1 - r0 + 1, "w": c1 - c0 + 1,
                    "size": len(cells),
                    "center_r": sum(rs) / len(rs),
                    "center_c": sum(cs) / len(cs),
                })

    if not comps:
        return grid

    # Largest component is the template
    template = max(comps, key=lambda comp: comp["size"])
    tr0, tc0 = template["r0"], template["c0"]
    th, tw = template["h"], template["w"]
    template_rel = {(r - tr0, c - tc0) for r, c in template["cells"]}

    out = [row[:] for row in grid]

    def rects_overlap(a, b):
        return not (a["r1"] < b["r0"] or b["r1"] < a["r0"] or a["c1"] < b["c0"] or b["c1"] < a["c0"])

    def bbox_distance(r0, c0):
        # distance between candidate bbox and template bbox (0 if touching/overlapping)
        r1, c1 = r0 + th - 1, c0 + tw - 1
        vr = 0
        if r1 < template["r0"]:
            vr = template["r0"] - r1
        elif template["r1"] < r0:
            vr = r0 - template["r1"]
        hc = 0
        if c1 < template["c0"]:
            hc = template["c0"] - c1
        elif template["c1"] < c0:
            hc = c0 - template["c1"]
        return vr + hc

    for comp in comps:
        if comp is template:
            continue

        dr = sign(comp["center_r"] - template["center_r"])
        dc = sign(comp["center_c"] - template["center_c"])
        if dr == 0 and dc == 0:
            continue

        # Enumerate placements of the full template shape that contain the marker component
        candidates = set()
        comp_cells = comp["cells"]
        for rr, cc in comp_cells:
            for pr, pc in template_rel:
                cand_r0 = rr - pr
                cand_c0 = cc - pc
                ok = True
                for xr, xc in comp_cells:
                    rr2 = xr - cand_r0
                    cc2 = xc - cand_c0
                    if (rr2, cc2) not in template_rel:
                        ok = False
                        break
                if ok:
                    candidates.add((cand_r0, cand_c0))

        if not candidates:
            continue

        filtered = []
        for r0, c0 in candidates:
            r1, c1 = r0 + th - 1, c0 + tw - 1
            cand_rect = {"r0": r0, "r1": r1, "c0": c0, "c1": c1}

            # Must be in the correct direction relative to template
            if dr < 0 and r1 >= template["r0"]:
                continue
            if dr > 0 and r0 <= template["r1"]:
                continue
            if dc < 0 and c1 >= template["c0"]:
                continue
            if dc > 0 and c0 <= template["c1"]:
                continue

            # Avoid overlapping the template
            if rects_overlap(cand_rect, template):
                continue

            filtered.append((r0, c0))

        if not filtered:
            # Fallback: only require non-overlap
            for r0, c0 in candidates:
                r1, c1 = r0 + th - 1, c0 + tw - 1
                cand_rect = {"r0": r0, "r1": r1, "c0": c0, "c1": c1}
                if not rects_overlap(cand_rect, template):
                    filtered.append((r0, c0))

        if not filtered:
            filtered = list(candidates)

        # Choose the candidate closest to the template
        start_r0, start_c0 = min(filtered, key=lambda p: (bbox_distance(p[0], p[1]), abs(p[0] - tr0) + abs(p[1] - tc0)))

        step_r = dr * (th + 1) if dr != 0 else 0
        step_c = dc * (tw + 1) if dc != 0 else 0

        r0, c0 = start_r0, start_c0
        while True:
            # Stop when bbox is completely outside
            if r0 >= H or c0 >= W or r0 + th - 1 < 0 or c0 + tw - 1 < 0:
                break

            for pr, pc in template_rel:
                rr, cc = r0 + pr, c0 + pc
                if 0 <= rr < H and 0 <= cc < W:
                    out[rr][cc] = comp["color"]

            r0 += step_r
            c0 += step_c
            if step_r == 0 and step_c == 0:
                break

    return out
