2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    rows, cols = len(grid), len(grid[0])
    
    # Find 2-cells and 5-cells
    twos = [(r,c) for r in range(rows) for c in range(cols) if grid[r][c] == 2]
    fives = [(r,c) for r in range(rows) for c in range(cols) if grid[r][c] == 5]
    
    # Find the bounding box of the 2-rectangle
    r2min = min(r for r,c in twos)
    r2max = max(r for r,c in twos)
    c2min = min(c for r,c in twos)
    c2max = max(c for r,c in twos)
    
    # The opening is on the side where 2s are missing
    # Find the interior of the rectangle
    # The 5s need to be reflected into the interior through the opening
    
    # Determine which side(s) the 5-pattern is on relative to the 2-box
    f_rcenter = sum(r for r,c in fives)/len(fives)
    f_ccenter = sum(c for r,c in fives)/len(fives)
    r2center = (r2min+r2max)/2
    c2center = (c2min+c2max)/2
    
    result = copy.deepcopy(grid)
    # Clear all 5s
    for r,c in fives:
        result[r][c] = 0
    
    # Determine direction of 5s relative to box center
    dr = f_rcenter - r2center
    dc = f_ccenter - c2center
    
    # Find the opening edge - the edge closest to the 5s
    # Mirror 5s through that edge into the box interior
    if abs(dr) > abs(dc):
        # top or bottom
        if dr < 0:  # 5s above, opening is top edge (r2min)
            for r,c in fives:
                nr = 2*r2min - r + 1  # reflect: distance from r to r2min is r2min-r, place at r2min + (r2min-r) +1? 
                # Actually: reflect into interior. The opening row is r2min+1 (first interior row)
                # Distance of 5 from r2min: d = r2min - r
                # New position: r2min + d = 2*r2min - r
                nr = 2*r2min - r
                if 0 <= nr < rows:
                    result[nr][c] = 5
        else:  # 5s below
            for r,c in fives:
                nr = 2*r2max - r
                if 0 <= nr < rows:
                    result[nr][c] = 5
    else:
        if dc < 0:  # 5s left
            for r,c in fives:
                nc = 2*c2min - c
                if 0 <= nc < cols:
                    result[r][nc] = 5
        else:  # 5s right
            for r,c in fives:
                nc = 2*c2max - c
                if 0 <= nc < cols:
                    result[r][nc] = 5
    
    return result
