def solve(grid):
    n = len(grid)
    m = len(grid[0])
    
    # Count unique values per row

    # Count occurrences of each value
    from collections import Counter

            counts[v] += 1
    
    most_common = counts.most_common()[0][0]
    
    # Create binary grid: 1 where most_common, 0 otherwise
    binary = []
    for row in grid:
        binary.append([1 if v == most_common else 0 for v in row])

    row_counts = [sum(r) for r in binary]
    
    # If all 9 (uniform grid) -> top row 5s, rest 0s
