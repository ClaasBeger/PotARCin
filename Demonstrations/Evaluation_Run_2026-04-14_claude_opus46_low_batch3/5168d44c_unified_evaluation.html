def solve(grid):

    rows = len(grid)
    cols = len(grid[0])
    
    # Find the 3x3 box of 2s (with 3 in center)
    box_r = box_c = None
    for r in range(rows-2):
        for c in range(cols-2):
            if (grid[r][c]==2 and grid[r][c+1]==2 and grid[r][c+2]==2 and
                grid[r+1][c]==2 and grid[r+1][c+1]==3 and grid[r+1][c+2]==2 and

                break

            break
    
    center_r = box_r + 1
    center_c = box_c + 1
    
    # Find trail of 3s from center - determine direction

    trails = {}

        count = 0

            if grid[r][c] == 3:
                count += 1
            r += dr * 2
            c += dc * 2
        if count > 0:
            trails[name] = count

    # The box moves toward the side with more trailing 3s, by 2 * (fewer count) steps
