2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    grid = [row[:] for row in grid]
    rows, cols = len(grid), len(grid[0])
    
    # Find all connected components of non-zero cells
    visited = [[False]*cols for _ in range(rows)]
    objects = []
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] != 0 and not visited[r][c]:
                # BFS
                stack = [(r,c)]
                visited[r][c] = True
                cells = []
                while stack:
                    cr, cc = stack.pop()
                    cells.append((cr, cc, grid[cr][cc]))
                    for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                        nr, nc = cr+dr, cc+dc
                        if 0<=nr<rows and 0<=nc<cols and not visited[nr][nc] and grid[nr][nc]!=0:
                            visited[nr][nc] = True
                            stack.append((nr,nc))
                objects.append(cells)
    
    # For each object, check if it has exactly 2 colors and one is a "template" pattern
    # The template is the small object with the same 2-color structure
    # Find the template: small object with specific pattern (like cross shape)
    
    # Separate objects by size - small ones are templates, large ones need expansion
    # Actually, look for objects that contain 2 colors. Those with 2 colors where one matches
    # a template color get expanded.
    
    for obj in objects:
        colors = set(v for _,_,v in obj)
        if len(colors) != 2:
            continue
        
        # Find the two color regions within this object
        color_cells = {}
        for r,c,v in obj:
            color_cells.setdefault(v, []).append((r,c))
        
        c1, c2 = list(colors)
        cells1, cells2 = color_cells[c1], color_cells[c2]
        
        # Determine block size: each color region dimensions
        def bbox(cells):
            rs = [r for r,c in cells]
            cs = [c for r,c in cells]
            return min(rs), max(rs), min(cs), max(cs)
        
        b1 = bbox(cells1)
        b2 = bbox(cells2)
        
        # Check if both regions are solid rectangles
        def is_rect(cells, bb):
            return len(cells) == (bb[1]-bb[0]+1)*(bb[3]-bb[2]+1)
        
        if not (is_rect(cells1, b1) and is_rect(cells2, b2)):
            # This might be a template, skip
            continue
        
        # This is a 2-color rectangular object. Find matching template.
        # The template has one of these colors plus a third color, arranged in a cross-like pattern
        
        # Determine the relative position of c2 w.r.t c1
        # Then find template that shows similar structure
        
        h1, w1 = b1[1]-b1[0]+1, b1[3]-b1[2]+1
        h2, w2 = b2[1]-b2[0]+1, b2[3]-b2[2]+1
        
        # Look for template with one of these colors
        for tmpl in objects:
            if tmpl is obj:
                continue
            tc = set(v for _,_,v in tmpl)
            common = tc & colors
            if len(common) != 1:
                continue
            anchor_color = list(common)[0]
            other_color = list(tc - common)[0]
            
            # Template defines the pattern. Get template bbox and shape
            t_cells = {}
            for r,c,v in tmpl:
                t_cells.setdefault(v, []).append((r,c))
            
            tb = bbox([p for r,c,v in tmpl for p in [(r,c)]])
            th = tb[1]-tb[0]+1
            tw = tb[3]-tb[2]+1
            
            # Scale factor: block size of the big object's color regions
            # The anchor_color region in big obj has some block size
            if anchor_color == c1:
                bh, bw = h1, w1
            else:
                bh, bw = h2, w2
            
            sx, sy = bh // 1, bw // 1  # need to figure scale
            # Template is th x tw, each cell maps to (bh/something) block
            # Actually scale = bh (or bw) since template cells are 1x1
            
            # Place scaled template centered on the big object
            obj_bb = bbox([p for r,c,v in obj for p in [(r,c)]])
            
            # Clear original big object
            for r,c,v in obj:
                grid[r][c] = 0
            
            # Scale each template cell by block size
            for r,c,v in tmpl:
                tr, tc2 = r - tb[0], c - tb[2]
                for dr in range(bh):
                    for dc in range(bw):
                        nr = obj_bb[0] + tr*bh + dr
                        nc = obj_bb[2] + tc2*bw + dc
                        if 0<=nr<rows and 0<=nc<cols:
                            if v == anchor_color:
                                grid[nr][nc] = v
                            else:
                                grid[nr][nc] = other_color
            break
    
    return grid
