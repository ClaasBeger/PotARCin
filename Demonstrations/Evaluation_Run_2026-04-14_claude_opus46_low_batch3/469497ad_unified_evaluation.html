Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    n = len(grid)
    # Extract the 4 quadrants info
    # Top-left is (n-1)x(n-1) with 0s and a colored shape
    # Right column (except last) and bottom row (except last) have border colors
    # Bottom-right corner has a color
    
    size = n - 1  # inner grid size
    
    # Find the non-zero color in the inner grid
    inner = [row[:size] for row in grid[:size]]
    
    # Find colored cells and their bounding box
    color_cells = {}
    for r in range(size):
        for c in range(size):
            v = inner[r][c]
            if v != 0:
                if v not in color_cells:
                    color_cells[v] = []
                color_cells[v].append((r, c))
    
    # Right border colors and bottom border colors
    right_colors = [grid[r][n-1] for r in range(size)]
    bottom_colors = [grid[n-1][c] for c in range(size)]
    corner_color = grid[n-1][n-1]
    
    # Scale factor: need to figure out. Looking at examples:
    # Ex1: 4x4 inner, 2x2 colored block, scale=2 -> 8x8 inner + 2 border cols/rows = 10x10

    obj_color = list(color_cells.keys())[0]
    positions = color_cells[obj_color]
    min_r = min(p[0] for p in positions)
    max_r = max(p[0] for p in positions)
    min_c = min(p[1] for p in positions)
    max_c = max(p[1] for p in positions)
    obj_h = max_r - min_r + 1

    out = [[0]*out_size for _ in range(out_size)]

                if v != 0:

                        for dc in range(scale):
