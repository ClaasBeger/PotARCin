2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    result = [row[:] for row in grid]
    
    # Find connected regions of 0s bounded by 2s and grid edges
    # The largest region gets filled with 1, small enclosed regions get 8
    
    from collections import deque
    
    visited = [[False]*cols for _ in range(rows)]
    regions = []
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 0 and not visited[r][c]:
                # BFS
                q = deque([(r,c)])
                visited[r][c] = True
                cells = []
                while q:
                    cr, cc = q.popleft()
                    cells.append((cr,cc))
                    for dr,dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                        nr,nc = cr+dr, cc+dc
                        if 0<=nr<rows and 0<=nc<cols and not visited[nr][nc] and grid[nr][nc]==0:
                            visited[nr][nc] = True
                            q.append((nr,nc))
                regions.append(cells)
    
    # Sort by size descending
    regions.sort(key=lambda x: len(x), reverse=True)
    
    # Largest gets 1, others that are "enclosed" (bounded by 2s forming a box) get 8
    # Looking at examples: the biggest region = 1, small regions enclosed by 2-lines = 8, 
    # regions touching only grid edges stay 0
    
    # Actually: regions that are fully bounded by 2s (not touching grid boundary without a 2 wall) get 8
    # The largest region gets 1
    # Regions that are just open areas stay 0
    
    # Simpler: largest region -> 1, smallest regions that are enclosed by 2-walls -> 8, rest -> 0
    
    # From examples: there's exactly one large region (1) and some small enclosed regions (8)
    # Regions touching corners or open edges remain 0
    
    # Let me check: a region is "enclosed" if it's surrounded by 2s on all sides (not grid edges)
    # A region touches grid edge if any cell is on boundary
    
    for i, cells in enumerate(regions):
        if i == 0:
            # largest -> 1
            for r,c in cells:
                result[r][c] = 1
        else:
            # Check if enclosed by 2s (no cell on grid boundary, OR boundary cells have 2 between them and edge)
            # Simpler: check if region is bounded by 2-lines forming a rectangle
            # From examples, small regions between 2-lines get 8
            touches_edge_without_wall = False
            for r,c in cells:
                if r == 0 or r == rows-1 or c == 0 or c == cols-1:
                    touches_edge_without_wall = True
                    break
            if not touches_edge_without_wall:
                for r,c in cells:
                    result[r][c] = 8
            # else stays 0
    
    return result
