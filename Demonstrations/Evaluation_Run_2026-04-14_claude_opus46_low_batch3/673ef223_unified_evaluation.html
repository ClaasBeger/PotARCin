2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    g = copy.deepcopy(grid)
    rows, cols = len(g), len(g[0])
    
    # Find vertical bars of 2s
    # Left bar: column 0, Right bar: last column
    # Find all 2-bars and 8s
    
    # Find connected vertical segments of 2s
    bars = []
    visited = [[False]*cols for _ in range(rows)]
    
    for c in range(cols):
        for r in range(rows):
            if g[r][c] == 2 and not visited[r][c]:
                # extend vertically
                rs = r
                while rs < rows and g[rs][c] == 2:
                    visited[rs][c] = True
                    rs += 1
                bars.append((r, rs-1, c))  # start_row, end_row, col
    
    # Find 8s
    eights = []
    for r in range(rows):
        for c in range(cols):
            if g[r][c] == 8:
                eights.append((r, c))
    
    for bar_start, bar_end, bar_col in bars:
        # Find 8s that are in rows within [bar_start, bar_end]
        bar_eights = [(r, c) for r, c in eights if bar_start <= r <= bar_end]
        
        if bar_eights:
            # Bar with 8s: draw from 8 position to bar, replacing 8 with 4, fill with 8s between
            for er, ec in bar_eights:
                if bar_col == 0:
                    # bar on left, 8 is to the right
                    g[er][ec] = 4
                    for cc in range(bar_col+1, ec):
                        g[er][cc] = 8
                else:
                    # bar on right, 8 is to the left
                    g[er][ec] = 4
                    for cc in range(ec+1, bar_col):
                        g[er][cc] = 8
        else:
            # Bar without 8s: mirror the other bar's 8 rows, fill entire row
            other_eights_rows = set()
            for ob_s, ob_e, ob_c in bars:
                if (ob_s, ob_e, ob_c) != (bar_start, bar_end, bar_col):
                    for er, ec in eights:
                        if ob_s <= er <= ob_e:
                            other_eights_rows.add(er)
            
            # Map rows: the 8-rows in other bar map to corresponding rows in this bar
            other_bar = None
            for ob_s, ob_e, ob_c in bars:
                if (ob_s, ob_e, ob_c) != (bar_start, bar_end, bar_col):
                    other_bar = (ob_s, ob_e, ob_c)
            
            if other_bar:
                ob_s, ob_e, ob_c = other_bar
                other_8_offsets = []
                for er, ec in eights:
                    if ob_s <= er <= ob_e:
                        other_8_offsets.append(er - ob_s)
                
                for off in other_8_offsets:
                    target_r = bar_start + off
                    if bar_start <= target_r <= bar_end:
                        if bar_col == 0:
                            for cc in range(1, cols):
                                if g[target_r][cc] != 2:
                                    g[target_r][cc] = 8
                        else:
                            for cc in range(0, cols-1):
                                if g[target_r][cc] != 2:
                                    g[target_r][cc] = 8
    
    return g
