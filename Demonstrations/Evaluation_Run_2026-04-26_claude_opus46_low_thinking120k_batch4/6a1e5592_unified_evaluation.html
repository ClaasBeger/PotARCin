Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import deque
    
    rows = len(grid)
    cols = len(grid[0])
    result = [row[:] for row in grid]
    
    two_rows = {r for r in range(rows) if 2 in grid[r]}
    
    def bfs(start_r, start_c, target, visited, region=None):
        q = deque([(start_r, start_c)])
        visited[start_r][start_c] = True
        cells = [(start_r, start_c)]
        while q:
            r, c = q.popleft()
            for dr, dc in [(-1,0),(1,0),(0,-1),(0,1)]:
                nr, nc = r+dr, c+dc
                if 0 <= nr < rows and 0 <= nc < cols and not visited[nr][nc] and grid[nr][nc] == target:
                    if region is None or nr in region:
                        visited[nr][nc] = True
                        q.append((nr, nc))
                        cells.append((nr, nc))
        return cells
    
    visited = [[False]*cols for _ in range(rows)]
    zero_holes = []
    for r in sorted(two_rows):
        for c in range(cols):
            if grid[r][c] == 0 and not visited[r][c]:
                zero_holes.append(bfs(r, c, 0, visited, two_rows))
    
    visited2 = [[False]*cols for _ in range(rows)]
    five_shapes = []
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 5 and not visited2[r][c]:
                five_shapes.append(bfs(r, c, 5, visited2))
    
    def normalize(cells):
        min_r = min(r for r, c in cells)
        min_c = min(c for r, c in cells if r == min_r)
        return sorted([(r - min_r, c - min_c) for r, c in cells])
    
    def get_anchor(cells):
        min_r = min(r for r, c in cells)
        min_c = min(c for r, c in cells if r == min_r)
        return min_r, min_c
    
    for hole in zero_holes:
        hole_norm = normalize(hole)
        hole_height = max(r for r, c in hole_norm) + 1
        anchor_r, anchor_c = get_anchor(hole)
        
        for shape in five_shapes:
            shape_norm = normalize(shape)
            top_part = sorted([(r, c) for r, c in shape_norm if r < hole_height])
            if top_part == hole_norm:
                for dr, dc in shape_norm:
                    nr, nc = anchor_r + dr, anchor_c + dc
                    if 0 <= nr < rows and 0 <= nc < cols:
                        result[nr][nc] = 1
                break
    
    for r in range(rows):
        for c in range(cols):
            if result[r][c] == 5:
                result[r][c] = 0
    
    return result
