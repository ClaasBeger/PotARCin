2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    # Find the 8-lines
    hline = next(r for r in range(rows) if all(grid[r][c] == 8 for c in range(cols)))
    vline = next(c for c in range(cols) if all(grid[r][c] == 8 for r in range(rows)))
    
    # Determine quadrants
    # Key is 2x2 with colors, pattern is 6x6 with 3s
    quads = {}
    for qr, rslice in enumerate([(0, hline), (hline+1, rows)]):
        for qc, cslice in enumerate([(0, vline), (vline+1, cols)]):
            r0, r1 = rslice
            c0, c1 = cslice
            cells = [grid[r][c] for r in range(r0, r1) for c in range(c0, c1) if grid[r][c] != 8]
            has3 = 3 in cells
            quads[(qr, qc)] = {'r': (r0, r1), 'c': (c0, c1), 'has3': has3}
    
    # Find pattern quad (has 3s) and key quad (has colors, not 0/3/8)
    pattern_pos = [k for k, v in quads.items() if v['has3']]
    # The pattern is the big one, key is small
    # Extract pattern and key
    pat_q = pattern_pos[0]  # should be one large quad with 3s
    
    # Find key quad (diagonal opposite or adjacent - the one with non-0 non-3 non-8 values)
    key_q = None
    for k, v in quads.items():
        if not v['has3']:
            r0, r1 = v['r']; c0, c1 = v['c']
            vals = [grid[r][c] for r in range(r0, r1) for c in range(c0, c1) if grid[r][c] not in (0, 8)]
            if vals:
                key_q = k
                break
    
    kr0, kr1 = quads[key_q]['r']
    kc0, kc1 = quads[key_q]['c']
    key = [[grid[kr0+r][kc0+c] for c in range(kr1-kr0)] for r in range(kc1-kc0)]
    # Actually just read 2x2 key
    key = [[grid[kr0][kc0], grid[kr0][kc1-1 if kc1-kc0>1 else kc0]],
           [grid[kr1-1][kc0], grid[kr1-1][kc1-1 if kc1-kc0>1 else kc0]]]
    
    # Extract pattern
    pr0, pr1 = quads[pat_q]['r']
    pc0, pc1 = quads[pat_q]['c']
    ph = pr1 - pr0
    pw = pc1 - pc0
    
    pattern = []
    for r in range(pr0, pr1):
        row = []
        for c in range(pc0, pc1):
            if grid[r][c] != 8:
                row.append(grid[r][c])
        pattern.append(row)
    
    h = len(pattern)
    w = len(pattern[0])
    
    output = [[0]*w for _ in range(h)]
    for r in range(h):
        for c in range(w):
            if pattern[r][c] == 3:
                kr = 0 if r < h//2 else 1
                kc = 0 if c < w//2 else 1
                output[r][c] = key[kr][kc]
    
    return output
