2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import Counter, deque
    
    rows = len(grid)
    cols = len(grid[0])
    
    flat = [grid[r][c] for r in range(rows) for c in range(cols)]
    bg = Counter(flat).most_common(1)[0][0]
    
    non_bg = {}
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] != bg:
                non_bg[(r,c)] = grid[r][c]
    
    visited = set()
    components = []
    for pos in non_bg:
        if pos not in visited:
            q = deque([pos])
            visited.add(pos)
            comp = [pos]
            while q:
                r, c = q.popleft()
                for dr in [-1,0,1]:
                    for dc in [-1,0,1]:
                        if dr == 0 and dc == 0: continue
                        nr, nc = r+dr, c+dc
                        if (nr,nc) in non_bg and (nr,nc) not in visited:
                            visited.add((nr,nc))
                            q.append((nr,nc))
                            comp.append((nr,nc))
            components.append(comp)
    
    template_comp = max(components, key=lambda comp: len(set(non_bg[p] for p in comp)))
    template_colors = set(non_bg[p] for p in template_comp)
    
    min_r = min(r for r,c in template_comp)
    min_c = min(c for r,c in template_comp)
    max_r = max(r for r,c in template_comp)
    max_c = max(c for r,c in template_comp)
    th, tw = max_r-min_r+1, max_c-min_c+1
    
    template = [[bg]*tw for _ in range(th)]
    for r,c in template_comp:
        template[r-min_r][c-min_c] = non_bg[(r,c)]
    
    tcells = [non_bg[p] for p in template_comp]
    fill_color = Counter(tcells).most_common(1)[0][0]
    corner_colors = [c for c in template_colors if c != fill_color]
    color_a, color_b = corner_colors[0], corner_colors[1]
    
    corner_pos = {}
    for color in corner_colors:
        for r in range(th):
            for c in range(tw):
                if template[r][c] == color and color not in corner_pos:
                    corner_pos[color] = (r,c)
    
    pa = corner_pos[color_a]
    pb = corner_pos[color_b]
    t_vec = (pb[0]-pa[0], pb[1]-pa[1])
    
    def apply_transform(ti, r, c):
        if ti==0: return (r,c)
        if ti==1: return (-c,r)
        if ti==2: return (-r,-c)
        if ti==3: return (c,-r)
        if ti==4: return (r,-c)
        if ti==5: return (-r,c)
        if ti==6: return (c,r)
        if ti==7: return (-c,-r)
    
    other_comps = [c for c in components if c is not template_comp]
    blocks = []
    for comp in other_comps:
        color = non_bg[comp[0]]
        rs = [r for r,c in comp]
        cs = [c for r,c in comp]
        size = max(max(rs)-min(rs)+1, max(cs)-min(cs)+1)
        blocks.append((color, min(rs), min(cs), size))
    
    result = [row[:] for row in grid]
    
    blocks_a = [b for b in blocks if b[0] == color_a]
    blocks_b = [b for b in blocks if b[0] == color_b]
    
    used_b = set()
    for ba in blocks_a:
        for i, bb in enumerate(blocks_b):
            if i in used_b or ba[3] != bb[3]:
                continue
            scale = ba[3]
            a_pos = (ba[1], ba[2])
            b_pos = (bb[1], bb[2])
            
            a_vec_r = (b_pos[0]-a_pos[0])/scale
            a_vec_c = (b_pos[1]-a_pos[1])/scale
            
            found_ti = None
            for ti in range(8):
                tv = apply_transform(ti, t_vec[0], t_vec[1])
                if tv[0] == a_vec_r and tv[1] == a_vec_c:
                    found_ti = ti
                    break
            
            if found_ti is None:
                continue
            
            for r in range(th):
                for c in range(tw):
                    v = template[r][c]
                    if v == bg:
                        continue
                    dr, dc = r-pa[0], c-pa[1]
                    tdr, tdc = apply_transform(found_ti, dr, dc)
                    for sr in range(scale):
                        for sc in range(scale):
                            pr = a_pos[0] + tdr*scale + sr
                            pc = a_pos[1] + tdc*scale + sc
                            if 0<=pr<rows and 0<=pc<cols:
                                result[pr][pc] = v
            
            used_b.add(i)
            break
    
    return result
