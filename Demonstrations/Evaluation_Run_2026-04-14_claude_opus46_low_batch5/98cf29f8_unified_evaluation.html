def solve(grid):
    import numpy as np
    g = np.array(grid)
    rows, cols = g.shape
    
    # Find all non-zero colors
    colors = set(np.unique(g)) - {0}
    
    # For each color, find its bounding box and cells
    color_info = {}
    for c in colors:
        positions = list(zip(*np.where(g == c)))
        r_min = min(p[0] for p in positions)
        r_max = max(p[0] for p in positions)
        c_min = min(p[1] for p in positions)
        c_max = max(p[1] for p in positions)
        color_info[c] = {
            'positions': positions,
            'bbox': (r_min, r_max, c_min, c_max),
            'cells': set(positions)
        }
    
    # Find the rectangle (all cells in bbox filled) vs the one with a line/connector
    # The rectangle is the "anchor", the other shape gets moved to be adjacent
    
    # Identify which is rectangle and which has a connector line
    rects = []
    others = []
    for c, info in color_info.items():
        r_min, r_max, c_min, c_max = info['bbox']
        expected = (r_max - r_min + 1) * (c_max - c_min + 1)
        if len(info['cells']) == expected:
            rects.append(c)
        else:
            others.append(c)
    
    # The "other" has a rectangular body + a line connecting to the rect
    # We need to find the body of the other shape (the largest rectangle within it)

    result = np.zeros_like(g)
    
    # Keep the rectangle as-is

        for (r, cl) in color_info[c]['cells']:
