2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows, cols = len(grid), len(grid[0])
    
    # Find the cross/plus pattern
    # Find center: the cell where horizontal and vertical arms intersect
    # Look for non-zero cells
    nz = {}
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] != 0:
                nz[(r,c)] = grid[r][c]
    
    # Find the center - cell that shares row with others AND column with others
    row_counts = {}
    col_counts = {}
    for (r,c) in nz:
        row_counts[r] = row_counts.get(r, 0) + 1
        col_counts[c] = col_counts.get(c, 0) + 1
    
    # The horizontal arm row and vertical arm column
    h_row = max(row_counts, key=row_counts.get)
    v_col = max(col_counts, key=col_counts.get)
    
    # Extract arms from center
    center_val = grid[h_row][v_col]
    
    # Horizontal arm: sequence from center going right (excluding center? no, including)
    # Get all cells in h_row sorted by column
    h_cells = sorted([(c, grid[h_row][c]) for (r,c) in nz if r == h_row], key=lambda x: x[0])
    # Vertical arm cells
    v_cells = sorted([(r, grid[r][v_col]) for (r,c) in nz if c == v_col], key=lambda x: x[0])
    
    # Horizontal pattern: from center rightward
    right_seq = [grid[h_row][c] for (c,_) in h_cells if c >= v_col]
    left_seq = [grid[h_row][c] for (c,_) in h_cells if c <= v_col]
    left_seq.reverse()  # from center going left
    
    # The repeating horizontal pattern is right_seq (which starts at center)
    # In example 2: center=3, right=3,8,4 and left=3,2 -> pattern repeats as ...2,3,8,4,2,3,8,4...
    # Full period: right_seq + left_seq[1:] but need to figure out the tile
    # Pattern: left_seq reversed (away from center) then right_seq... 
    # Actually the horizontal line repeats: the sequence from left to right across the row
    h_pattern = [v for (c,v) in h_cells]  # left to right
    
    # Vertical: top to bottom
    v_pattern = [v for (r,v) in v_cells]
    
    result = [[0]*cols for _ in range(rows)]
    
    # Fill horizontal row with repeating pattern
    # The pattern tiles with period len(h_pattern)
    # h_cells start at some column; we need to tile the whole row
    h_start_col = h_cells[0][0]
    period_h = len(h_pattern)
    for c in range(cols):
        idx = (c - h_start_col) % period_h
        result[h_row][c] = h_pattern[idx]
    
    # Fill vertical column with repeating pattern
    v_start_row = v_cells[0][0]
    period_v = len(v_pattern)
    for r in range(rows):
        idx = (r - v_start_row) % period_v
        result[r][v_col] = v_pattern[idx]
    
    # Center
    result[h_row][v_col] = center_val
    
    return result
