2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import numpy as np
    g = np.array(grid)
    rows, cols = g.shape
    
    # Find the two regions by background color
    # Detect background colors per region
    # One region has colored markers (non-background, non-1) as isolated points
    # The other has rectangular blocks with 1s and a special color
    
    # Find unique values to determine the two background colors
    vals = np.unique(g)
    
    # Identify two rectangular regions (could be side by side or top/bottom)
    # Check if there's a clear split
    bg_top = g[0, 0]
    bg_bot = g[-1, -1]
    
    if bg_top == bg_bot:
        bg_left = g[0, 0]
        bg_right = g[0, -1]
        # vertical split
        split_col = cols // 2
        region_a = g[:, :split_col]
        region_b = g[:, split_col:]
        vertical = True
    else:
        # horizontal split - find split row
        split_row = None
        for r in range(rows - 1):
            if g[r, 0] != g[r + 1, 0]:
                split_row = r + 1
                break
        region_a = g[:split_row, :]
        region_b = g[split_row:, :]
        vertical = False
    
    # One region has blocks (rectangles of non-background with 1s), other has isolated marker points
    # The markers in the point-region indicate where to place the blocks
    # The blocks contain the shape templates
    
    # Determine which region has blocks vs points
    # Blocks region: has rectangular clusters; Points region: has isolated non-bg cells
    
    if vertical:
        bg_a, bg_b = bg_left, bg_right
    else:
        bg_a = g[0, 0]
        bg_b = g[split_row, 0]
    
    # Find connected rectangular blocks in region_b (the one with 8s and 1s patterns)
    # and isolated points in region_a
    # Actually, let me look at the pattern more carefully.
    
    # The region with isolated colored points needs to have blocks placed around those points
    # The blocks come from the other region, matched by the special color in the block
    
    # For each block in the block-region, find its bounding box and the special color marker positions
    # For each isolated point in the point-region, find matching block and place it
    
    from scipy import ndimage
    
    def find_blocks_and_points(region, bg):
        mask = region != bg
        labeled, n = ndimage.label(mask)
        objects = []
        for i in range(1, n + 1):
            ys, xs = np.where(labeled == i)
            sub = region[ys.min():ys.max()+1, xs.min():xs.max()+1]
            # Find non-1, non-bg values in sub
            special = [v for v in np.unique(sub) if v != 1 and v != bg]
            objects.append((ys.min(), xs.min(), sub, special))
        return objects
    
    if vertical:
        objs_a = find_blocks_and_points(region_a, bg_a)
        objs_b = find_blocks_and_points(region_b, bg_b)
    else:
        objs_a = find_blocks_and_points(region_a, bg_a)
        objs_b = find_blocks_and_points(region_b, bg_b)
    
    # Separate blocks (large) from points (single cells)
    def classify(objs):
        blocks, points = [], []
        for o in objs:
            if o[2].size > 1:
                blocks.append(o)
            else:
                points.append(o)
        return blocks, points
    
    blocks_a, points_a = classify(objs_a)
    blocks_b, points_b = classify(objs_b)
    
    # Points region gets blocks placed; blocks come from block region matched by color
    if points_b:  # points in region_b, blocks in region_a
        target_bg, target_region, blocks, points = bg_b, region_b, blocks_a, points_b
    else:
        target_bg, target_region, blocks, points = bg_a, region_a, blocks_b, points_a
    
    out = np.full_like(target_region, target_bg)
    
    for pt in points:
        color = pt[2][0, 0]
        for blk in blocks:
            if color in blk[3]:
                sub = blk[2].copy()
                pr, pc = pt[0], pt[1]
                # Find marker position within block
                positions = list(zip(*np.where(sub == color)))
                for mr, mc in positions:
                    r0, c0 = pr - mr, pc - mc
                    for dr in range(sub.shape[0]):
                        for dc in range(sub.shape[1]):
                            out[r0+dr, c0+dc] = sub[dr, dc]
                break
    
    return out.tolist()
