2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import copy
    rows = len(grid)
    cols = len(grid[0])
    result = copy.deepcopy(grid)
    
    # Find horizontal and vertical lines of 8s
    h_lines = []  # row indices of horizontal lines
    v_lines = []  # col indices of vertical lines
    
    for r in range(rows):
        if all(grid[r][c] == 8 for c in range(cols)):
            h_lines.append(r)
    
    for c in range(cols):
        if all(grid[r][c] == 8 for r in range(rows)):
            v_lines.append(c)
    
    # Find 2s
    twos = []
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 2:
                twos.append((r, c))
    
    # For each 2, find nearest 8-line(s) and draw toward them
    for (r2, c2) in twos:
        # Find nearest horizontal line
        best_h = None
        best_h_dist = float('inf')
        for h in h_lines:
            d = abs(h - r2)
            if d < best_h_dist:
                best_h_dist = d
                best_h = h
        
        # Find nearest vertical line
        best_v = None
        best_v_dist = float('inf')
        for v in v_lines:
            d = abs(v - c2)
            if d < best_v_dist:
                best_v_dist = d
                best_v = v
        
        # Draw line of 2s from the 2 to each nearest 8-line (not overwriting the 8 line itself)
        # And place a 3x3 8-border around where it meets the line
        
        if best_h is not None:
            # Draw vertical line of 2s toward best_h
            dr = 1 if best_h > r2 else -1
            for r in range(r2 + dr, best_h, dr):
                result[r][c2] = 2
            # Place 2 on the 8-line
            result[best_h][c2] = 2
            # 3x3 box of 8s centered on (best_h, c2), but only on non-8-line cells
            for dr2 in [-1, 0, 1]:
                for dc2 in [-1, 0, 1]:
                    nr, nc = best_h + dr2, c2 + dc2
                    if 0 <= nr < rows and 0 <= nc < cols:
                        if nr == best_h and nc == c2:
                            continue  # keep the 2
                        if nr in h_lines and nc in v_lines:
                            continue
                        if nr in h_lines:
                            continue  # already 8
                        if nc in v_lines:
                            continue
                        result[nr][nc] = 8
        
        if best_v is not None:
            dc = 1 if best_v > c2 else -1
            for c in range(c2 + dc, best_v, dc):
                result[r2][c] = 2
            result[r2][best_v] = 2
            for dr2 in [-1, 0, 1]:
                for dc2 in [-1, 0, 1]:
                    nr, nc = r2 + dr2, best_v + dc2
                    if 0 <= nr < rows and 0 <= nc < cols:
                        if nr == r2 and nc == best_v:
                            continue
                        if nr in h_lines:
                            continue
                        if nc in v_lines:
                            continue
                        result[nr][nc] = 8
    
    return result
