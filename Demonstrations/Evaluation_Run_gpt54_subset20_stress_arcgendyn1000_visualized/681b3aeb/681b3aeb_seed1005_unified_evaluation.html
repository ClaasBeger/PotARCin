def solve(grid):
    from collections import defaultdict, deque

    h, w = len(grid), len(grid[0])

    # Collect cells by color
    cells_by_color = defaultdict(list)
    for r in range(h):
        for c in range(w):
            v = grid[r][c]
            if v != 0:
                cells_by_color[v].append((r, c))

    colors = list(cells_by_color.keys())
    if len(colors) != 2:
        # Fallback: crop non-zero bbox if not exactly two objects
        pts = [(r, c) for r in range(h) for c in range(w) if grid[r][c] != 0]
        if not pts:
            return [[0]]
        r0 = min(r for r, _ in pts)
        r1 = max(r for r, _ in pts)
        c0 = min(c for _, c in pts)
        c1 = max(c for _, c in pts)
        return [row[c0:c1 + 1] for row in grid[r0:r1 + 1]]

    # Build tight bbox matrices for the two colored objects
    shapes = []
    for color in colors:
        pts = cells_by_color[color]
        r0 = min(r for r, _ in pts)
        r1 = max(r for r, _ in pts)
        c0 = min(c for _, c in pts)
        c1 = max(c for _, c in pts)
        mat = [[0] * (c1 - c0 + 1) for _ in range(r1 - r0 + 1)]
        for r, c in pts:
            mat[r - r0][c - c0] = color
        shapes.append(mat)

    A, B = shapes
    ha, wa = len(A), len(A[0])
    hb, wb = len(B), len(B[0])

    best = None
    best_grid = None

    # Search relative placements of B with respect to A
    for dr in range(-hb, ha + 1):
        for dc in range(-wb, wa + 1):
            # Check for color collisions where both are non-zero
            ok = True
            occupied = []

            for r in range(ha):
                for c in range(wa):
                    if A[r][c] != 0:
                        occupied.append((r, c, A[r][c]))

            for r in range(hb):
                for c in range(wb):
                    if B[r][c] == 0:
                        continue
                    ar, ac = r + dr, c + dc
                    if 0 <= ar < ha and 0 <= ac < wa and A[ar][ac] != 0:
                        ok = False
                        break
                    occupied.append((ar, ac, B[r][c]))
                if not ok:
                    break

            if not ok:
                continue

            rs = [r for r, _, _ in occupied]
            cs = [c for _, c, _ in occupied]
            rmin, rmax = min(rs), max(rs)
            cmin, cmax = min(cs), max(cs)
            H = rmax - rmin + 1
            W = cmax - cmin + 1
            area = H * W

            # Build cropped union
            out = [[0] * W for _ in range(H)]
            for r, c, v in occupied:
                out[r - rmin][c - cmin] = v

            key = (area, H + W, H, W, tuple(tuple(row) for row in out))
            if best is None or key < best:
                best = key
                best_grid = out

    return best_grid
