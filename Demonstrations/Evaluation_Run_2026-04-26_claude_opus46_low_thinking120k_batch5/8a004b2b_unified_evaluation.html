2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    fours = [(r, c) for r in range(rows) for c in range(cols) if grid[r][c] == 4]
    r_min = min(r for r, c in fours)
    r_max = max(r for r, c in fours)
    c_min = min(c for r, c in fours)
    c_max = max(c for r, c in fours)
    rect_h = r_max - r_min + 1
    rect_w = c_max - c_min + 1
    
    color_cells = {}
    for r in range(r_min, r_max + 1):
        for c in range(c_min, c_max + 1):
            v = grid[r][c]
            if v not in (0, 4):
                color_cells.setdefault(v, []).append((r - r_min, c - c_min))
    
    first_cells = list(color_cells.values())[0]
    rs = [r for r, c in first_cells]
    block_size = max(rs) - min(rs) + 1
    
    all_colored = [(r, c) for cells in color_cells.values() for r, c in cells]
    origin_r = min(r for r, c in all_colored)
    origin_c = min(c for r, c in all_colored)
    
    template_cells = []
    for r in range(rows):
        for c in range(cols):
            if (r < r_min or r > r_max or c < c_min or c > c_max) and grid[r][c] != 0:
                template_cells.append((r, c, grid[r][c]))
    
    t_rmin = min(r for r, c, v in template_cells)
    t_cmin = min(c for r, c, v in template_cells)
    
    output = [[0] * rect_w for _ in range(rect_h)]
    for r, c in fours:
        output[r - r_min][c - c_min] = 4
    
    for r, c, color in template_cells:
        tr, tc = r - t_rmin, c - t_cmin
        for dr in range(block_size):
            for dc in range(block_size):
                out_r = origin_r + tr * block_size + dr
                out_c = origin_c + tc * block_size + dc
                if 0 <= out_r < rect_h and 0 <= out_c < rect_w:
                    output[out_r][out_c] = color
    
    return output
