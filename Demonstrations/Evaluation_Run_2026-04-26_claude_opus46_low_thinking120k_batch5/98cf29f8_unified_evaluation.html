2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import numpy as np
    g = np.array(grid)
    rows, cols = g.shape
    colors = set(g.flatten()) - {0}
    
    color_info = {}
    for c in colors:
        positions = set(zip(*np.where(g == c)))
        color_info[c] = positions
    
    rect_color = line_color = None
    for c in colors:
        pos = color_info[c]
        rs = [r for r, _ in pos]
        cs = [c2 for _, c2 in pos]
        if (max(rs)-min(rs)+1)*(max(cs)-min(cs)+1) == len(pos):
            rect_color = c
        else:
            line_color = c
    
    pos = color_info[line_color]
    row_counts = {}
    col_counts = {}
    for r, c in pos:
        row_counts[r] = row_counts.get(r, 0) + 1
        col_counts[c] = col_counts.get(c, 0) + 1
    
    line_cells = set()
    if any(v == 1 for v in row_counts.values()):
        for r, c in pos:
            if row_counts[r] == 1:
                line_cells.add((r, c))
    else:
        for r, c in pos:
            if col_counts[c] == 1:
                line_cells.add((r, c))
    
    rect_cells = pos - line_cells
    rect_rs = [r for r, _ in rect_cells]
    rect_cs = [c for _, c in rect_cells]
    rect_min_r, rect_max_r = min(rect_rs), max(rect_rs)
    rect_min_c, rect_max_c = min(rect_cs), max(rect_cs)
    
    line_rs = [r for r, _ in line_cells]
    line_cs = [c for _, c in line_cells]
    
    shift_r = shift_c = 0
    if len(set(line_cs)) == 1:
        if min(line_rs) < rect_min_r:
            shift_r = min(line_rs) - rect_min_r
        else:
            shift_r = max(line_rs) - rect_max_r
    else:
        if min(line_cs) < rect_min_c:
            shift_c = min(line_cs) - rect_min_c
        else:
            shift_c = max(line_cs) - rect_max_c
    
    out = [[0]*cols for _ in range(rows)]
    for r, c in color_info[rect_color]:
        out[r][c] = rect_color
    for r in range(rect_min_r, rect_max_r + 1):
        for c in range(rect_min_c, rect_max_c + 1):
            out[r + shift_r][c + shift_c] = line_color
    return out
