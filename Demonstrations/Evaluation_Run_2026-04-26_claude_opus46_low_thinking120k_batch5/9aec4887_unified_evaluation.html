2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    rows = len(grid)
    cols = len(grid[0])
    
    frame_cells = {}
    eight_cells = []
    
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == 8:
                eight_cells.append((r, c))
            elif grid[r][c] != 0:
                v = grid[r][c]
                if v not in frame_cells:
                    frame_cells[v] = []
                frame_cells[v].append((r, c))
    
    all_frame = []
    for v, cells in frame_cells.items():
        all_frame.extend(cells)
    
    min_r = min(r for r, c in all_frame)
    max_r = max(r for r, c in all_frame)
    min_c = min(c for r, c in all_frame)
    max_c = max(c for r, c in all_frame)
    
    top_color = bottom_color = left_color = right_color = None
    for v, cells in frame_cells.items():
        for r, c in cells:
            if r == min_r: top_color = v
            if r == max_r: bottom_color = v
            if c == min_c: left_color = v
            if c == max_c: right_color = v
    
    inner_h = max_r - min_r - 1
    inner_w = max_c - min_c - 1
    
    er_min = min(r for r, c in eight_cells)
    ec_min = min(c for r, c in eight_cells)
    
    pattern = [[0] * inner_w for _ in range(inner_h)]
    for r, c in eight_cells:
        pattern[r - er_min][c - ec_min] = 8
    
    out_h = inner_h + 2
    out_w = inner_w + 2
    out = [[0] * out_w for _ in range(out_h)]
    
    for c in range(1, out_w - 1):
        out[0][c] = top_color
        out[out_h - 1][c] = bottom_color
    for r in range(1, out_h - 1):
        out[r][0] = left_color
        out[r][out_w - 1] = right_color
    
    for r in range(inner_h):
        for c in range(inner_w):
            if pattern[r][c] == 8:
                dt = r + 1
                db = inner_h - r
                dl = c + 1
                dr = inner_w - c
                min_d = min(dt, db, dl, dr)
                cands = []
                if dt == min_d: cands.append(top_color)
                if db == min_d: cands.append(bottom_color)
                if dl == min_d: cands.append(left_color)
                if dr == min_d: cands.append(right_color)
                out[r + 1][c + 1] = cands[0] if len(cands) == 1 else 8
    
    return out
