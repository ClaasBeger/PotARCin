2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    H, W = len(grid), len(grid[0])

    # Locate the four corner markers (color 4)
    fours = [(r, c) for r in range(H) for c in range(W) if grid[r][c] == 4]
    rs = [r for r, _ in fours]
    cs = [c for _, c in fours]
    rmin, rmax = min(rs), max(rs)
    cmin, cmax = min(cs), max(cs)

    out_h = rmax - rmin + 1
    out_w = cmax - cmin + 1

    # Build output canvas as the cropped rectangle
    out = [[0 for _ in range(out_w)] for _ in range(out_h)]
    out[0][0] = 4
    out[0][out_w - 1] = 4
    out[out_h - 1][0] = 4
    out[out_h - 1][out_w - 1] = 4

    # Find colored components inside the 4-corner rectangle (excluding 0 and 4)
    inside = [[grid[r][c] for c in range(cmin, cmax + 1)] for r in range(rmin, rmax + 1)]
    h, w = len(inside), len(inside[0])

    visited = [[False] * w for _ in range(h)]
    comps = []

    for r in range(h):
        for c in range(w):
            if inside[r][c] != 0 and inside[r][c] != 4 and not visited[r][c]:
                color = inside[r][c]
                stack = [(r, c)]
                visited[r][c] = True
                cells = []
                while stack:
                    x, y = stack.pop()
                    cells.append((x, y))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx, ny = x + dx, y + dy
                        if 0 <= nx < h and 0 <= ny < w and not visited[nx][ny] and inside[nx][ny] == color:
                            visited[nx][ny] = True
                            stack.append((nx, ny))
                rr = [x for x, _ in cells]
                cc = [y for _, y in cells]
                top, left = min(rr), min(cc)
                bottom, right = max(rr), max(cc)
                comps.append({
                    "color": color,
                    "top": top,
                    "left": left,
                    "height": bottom - top + 1,
                    "width": right - left + 1
                })

    # Scale factor = size of the square blocks in the top region
    if comps:
        s = min(min(comp["height"], comp["width"]) for comp in comps)
    else:
        s = 1

    # Extract the miniature pattern outside the rectangle
    outside_cells = []
    for r in range(H):
        for c in range(W):
            if not (rmin <= r <= rmax and cmin <= c <= cmax):
                v = grid[r][c]
                if v != 0 and v != 4:
                    outside_cells.append((r, c))

    if not outside_cells:
        return out

    pr0 = min(r for r, _ in outside_cells)
    pr1 = max(r for r, _ in outside_cells)
    pc0 = min(c for _, c in outside_cells)
    pc1 = max(c for _, c in outside_cells)

    pattern = [[0 for _ in range(pc1 - pc0 + 1)] for _ in range(pr1 - pr0 + 1)]
    for r in range(pr0, pr1 + 1):
        for c in range(pc0, pc1 + 1):
            v = grid[r][c]
            if v != 4:
                pattern[r - pr0][c - pc0] = v

    ph, pw = len(pattern), len(pattern[0])

    # Precompute pattern positions by color
    pattern_pos = {}
    for r in range(ph):
        for c in range(pw):
            v = pattern[r][c]
            if v != 0:
                pattern_pos.setdefault(v, []).append((r, c))

    # Determine placement offset using top-region blocks
    def valid_offset(off_r, off_c):
        # Entire scaled pattern must fit inside crop
        if off_r < 0 or off_c < 0:
            return False
        if off_r + ph * s > out_h or off_c + pw * s > out_w:
            return False

        # Every existing top component must correspond to a same-colored miniature cell
        for comp in comps:
            tr, tc, color = comp["top"], comp["left"], comp["color"]
            if (tr - off_r) % s != 0 or (tc - off_c) % s != 0:
                return False
            mr = (tr - off_r) // s
            mc = (tc - off_c) // s
            if not (0 <= mr < ph and 0 <= mc < pw):
                return False
            if pattern[mr][mc] != color:
                return False
        return True

    offset_r = offset_c = None

    if comps:
        first = comps[0]
        for mr, mc in pattern_pos.get(first["color"], []):
            cand_r = first["top"] - mr * s
            cand_c = first["left"] - mc * s
            if valid_offset(cand_r, cand_c):
                offset_r, offset_c = cand_r, cand_c
                break

    if offset_r is None:
        # Fallback: place at top-left if possible
        offset_r, offset_c = 0, 0

    # Render scaled pattern
    for r in range(ph):
        for c in range(pw):
            color = pattern[r][c]
            if color == 0:
                continue
            rr0 = offset_r + r * s
            cc0 = offset_c + c * s
            for dr in range(s):
                for dc in range(s):
                    rr = rr0 + dr
                    cc = cc0 + dc
                    if 0 <= rr < out_h and 0 <= cc < out_w:
                        out[rr][cc] = color

    # Restore corner markers
    out[0][0] = 4
    out[0][out_w - 1] = 4
    out[out_h - 1][0] = 4
    out[out_h - 1][out_w - 1] = 4

    return out
