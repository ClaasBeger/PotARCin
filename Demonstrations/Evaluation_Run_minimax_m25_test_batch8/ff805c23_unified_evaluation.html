2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    """
    Receives a square grid (list of list of int) and returns the transformed 5x5 grid.
    The transformation extracts the interior of the colored region that contains the
    most cells different from its own colour, then down‑samples that interior to 5x5.
    """
    if not grid:
        return []

    n = len(grid)
    m = len(grid[0]) if n else 0

    # ------------------------------------------------------------------
    # 1) find connected components of each non‑zero colour (4‑directional)
    # ------------------------------------------------------------------
    visited = [[False] * m for _ in range(n)]
    components = []                     # each element: {'color', 'min_r','max_r','min_c','max_c'}

    for r in range(n):
        for c in range(m):
            if grid[r][c] != 0 and not visited[r][c]:
                col = grid[r][c]
                q = [(r, c)]
                visited[r][c] = True
                min_r = max_r = r
                min_c = max_c = c
                for cr, cc in q:
                    for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                        nr, nc = cr + dr, cc + dc
                        if 0 <= nr < n and 0 <= nc < m:
                            if not visited[nr][nc] and grid[nr][nc] == col:
                                visited[nr][nc] = True
                                q.append((nr, nc))
                                if nr < min_r: min_r = nr
                                if nr > max_r: max_r = nr
                                if nc < min_c: min_c = nc
                                if nc > max_c: max_c = nc
                components.append({
                    'color': col,
                    'min_r': min_r,
                    'max_r': max_r,
                    'min_c': min_c,
                    'max_c': max_c
                })

    # ------------------------------------------------------------------
    # 2) for each component, examine its interior (border removed)
    #    and count cells that are *not* the component's colour.
    # ------------------------------------------------------------------
    best_comp = None
    best_count = -1
    best interior = None   # (r1, r2, c1, c2)

    for comp in components:
        col = comp['color']
        r1 = comp['min_r'] + 1
        r2 = comp['max_r'] - 1
        c1 = comp['min_c'] + 1
        c2 = comp['max_c'] - 1
        if r1 > r2 or c1 > c2:
            continue                    # interior too small
        # count cells inside the interior that are not colour 'col'
        cnt = 0
        for rr in range(r1, r2 + 1):
            row = grid[rr]
            for cc in range(c1, c2 + 1):
                if row[cc] != col:
                    cnt += 1
        if cnt > best_count:
            best_count = cnt
            best_comp = comp
            best_interior = (r1, r2, c1, c2)

    # ------------------------------------------------------------------
    # 3) down‑sample the interior to a 5x5 grid
    # ------------------------------------------------------------------
    r1, r2, c1, c2 = best_interior
    h = r2 - r1 + 1
    w = c2 - c1 + 1
    colour = best_comp['color']

    bh = h // 5          # block height
    bw = w // 5          # block width

    out = [[0] * 5 for _ in range(5)]
    for i in range(5):
        rr = r1 + i * bh
        for j in range(5):
            cc = c1 + j * bw
            # look inside the block of size bh×bw
            non_colour = False
            for r in range(rr, rr + bh):
                for c in range(cc, cc + bw):
                    if grid[r][c] != colour:
                        non_colour = True
                        break
                if non_colour:
                    break
            out[i][j] = 0 if non_colour else colour

    return out
