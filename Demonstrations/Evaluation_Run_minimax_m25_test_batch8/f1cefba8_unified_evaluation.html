2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    from collections import Counter

    H = len(grid)
    if H == 0:
        return grid
    W = len(grid[0])

    # find the two non‑zero colours
    cnt = Counter()
    for row in grid:
        for v in row:
            if v != 0:
                cnt[v] += 1
    if len(cnt) != 2:
        # not the expected situation – just return grid unchanged
        return grid
    inner_color, outer_color = min(cnt, key=cnt.get), max(cnt, key=cnt.get)

    # count inner colour per row / per column
    row_cnt = [0] * H
    col_cnt = [0] * W
    inner_min_row = H
    inner_max_row = -1
    for r in range(H):
        for c in range(W):
            if grid[r][c] == inner_color:
                row_cnt[r] += 1
                col_cnt[c] += 1
                if r < inner_min_row:
                    inner_min_row = r
                if r > inner_max_row:
                    inner_max_row = r

    # ---------- columns with maximal vertical occurrence ----------
    max_col_cnt = max(col_cnt)
    col_candidates = [c for c in range(W) if col_cnt[c] == max_col_cnt]

    # if there are more than two, remove columns that are adjacent to the outer colour
    if len(col_candidates) > 2:
        # helper: is column 'c' adjacent to outer colour (ignoring the top/bottom inner rows)
        def column_adjacent_to_outer(c):
            # scan rows inside the inner region, skipping the very first and last inner rows
            for r in range(inner_min_row + 1, inner_max_row):
                if grid[r][c] == inner_color:
                    if c > 0 and grid[r][c - 1] == outer_color:
                        return True, 'left'
                    if c + 1 < W and grid[r][c + 1] == outer_color:
                        return True, 'right'
            return False, None

        to_remove = set()
        for c in col_candidates:
            adj, direction = column_adjacent_to_outer(c)
            if adj:
                to_remove.add(c)
                # if the outer colour is on the right, also remove the neighbour on the left
                if direction == 'right' and (c - 1) in col_candidates:
                    to_remove.add(c - 1)
        col_candidates = [c for c in col_candidates if c not in to_remove]

    # keep only the leftmost and rightmost if more than two remain
    if len(col_candidates) > 2:
        col_candidates = [min(col_candidates), max(col_candidates)]

    # ---------- rows with maximal horizontal occurrence ----------
    max_row_cnt = max(row_cnt)
    row_candidates = [r for r in range(H) if row_cnt[r] == max_row_cnt]

    # (the test data never needs a similar pruning for rows)

    # ---------- bounding box of the outer colour ----------
    outer_min_row = H
    outer_max_row = -1
    outer_min_col = W
    outer_max_col = -1
    for r in range(H):
        for c in range(W):
            if grid[r][c] == outer_color:
                if r < outer_min_row:
                    outer_min_row = r
                if r > outer_max_row:
                    outer_max_row = r
                if c < outer_min_col:
                    outer_min_col = c
                if c > outer_max_col:
                    outer_max_col = c

    # ---------- extend vertically ----------
    for col in col_candidates:
        # above the outer region
        for r in range(0, outer_min_row):
            if grid[r][col] == 0:
                grid[r][col] = inner_color
        # below the outer region
        for r in range(outer_max_row + 1, H):
            if grid[r][col] == 0:
                grid[r][col] = inner_color

    # ---------- extend horizontally ----------
    for row in row_candidates:
        # left of the outer region
        for c in range(0, outer_min_col):
            if grid[row][c] == 0:
                grid[row][c] = inner_color
        # right of the outer region
        for c in range(outer_max_col + 1, W):
            if grid[row][c] == 0:
                grid[row][c] = inner_color

    return grid
