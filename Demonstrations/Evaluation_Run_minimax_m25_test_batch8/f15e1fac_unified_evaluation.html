2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    # Copy input to output
    out = [row[:] for row in grid]
    h = len(grid)
    w = len(grid[0]) if h else 0

    # Find all cells with value 2
    anchor_rows = set()
    anchor_cols = set()
    for r in range(h):
        for c in range(w):
            if grid[r][c] == 2:
                anchor_rows.add(r)
                anchor_cols.add(c)

    # Find all cells with value 8 (the pattern)
    pattern = [(r, c) for r in range(h) for c in range(w) if grid[r][c] == 8]

    if not pattern:
        return out

    # Determine orientation of the 2‑line
    if len(anchor_cols) == 1:                     # vertical line of 2s
        anchor_col = next(iter(anchor_cols))
        step = 1
        # direction of horizontal shift: away from the 2‑column
        min_pat_c = min(c for (_, c) in pattern)
        sign = 1 if anchor_col < min_pat_c else -1

        # count how many 2s are above each row
        cnt = 0
        for r in range(h):
            if grid[r][anchor_col] == 2:
                cnt += 1
            offset = sign * step * cnt
            for (r0, c0) in pattern:
                nc = c0 + offset
                if 0 <= nc < w and out[r][nc] != 2:
                    out[r][nc] = 8

    elif len(anchor_rows) == 1:                  # horizontal line of 2s
        anchor_row = next(iter(anchor_rows))
        anc_cols = sorted(anchor_cols)
        step = anc_cols[1] - anc_cols[0] if len(anc_cols) >= 2 else 1

        # horizontal shift direction: try to move away from the 2‑columns
        min_pat_c = min(c for (_, c) in pattern)
        sign = 1 if anc_cols[0] < min_pat_c else -1

        # propagation direction (vertical): away from the 2‑row
        max_pat_r = max(r for (r, _) in pattern)
        min_pat_r = min(r for (r, _) in pattern)
        if anchor_row > max_pat_r:
            dir_v = -1          # upward
        elif anchor_row < min_pat_r:
            dir_v = 1           # downward
        else:
            dir_v = 0           # should not happen

        # Feasibility test: can we place a block in the chosen direction?
        feasible = True
        for (r0, c0) in pattern:
            if dir_v == -1:
                max_dist = r0
            else:
                max_dist = h - 1 - r0
            start = c0 + sign * step * max_dist
            if sign == 1:
                if start > w - step:
                    feasible = False
                    break
            else:
                if start - step + 1 < 0:
                    feasible = False
                    break
        if not feasible:
            sign = -sign

        # Propagate each seed upwards (or downwards)
        for (r0, c0) in pattern:
            k = 0
            while True:
                r = r0 + dir_v * k
                if r < 0 or r >= h:
                    break
                offset = step * k
                start = c0 + sign * offset

                if sign == 1:
                    col_l = start
                    col_r = start + step - 1
                else:
                    col_l = start - step + 1
                    col_r = start

                # clip to grid
                if col_l > w - 1 or col_r < 0:
                    if sign == 1 and col_l > w - 1:
                        break
                    if sign == -1 and col_r < 0:
                        break
                col_l = max(col_l, 0)
                col_r = min(col_r, w - 1)
                if col_l > col_r:
                    break

                for cc in range(col_l, col_r + 1):
                    if out[r][cc] != 2:
                        out[r][cc] = 8
                k += 1
    # If the 2s do not form a single row or column, leave the grid unchanged
    return out
