2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    # grid is a 16x16 list of lists of ints
    
    # 1. find the 5x5 uniform block (all cells equal)
    block_pos = None
    block_color = None
    for r in range(12):          # 0 .. 11
        for c in range(12):
            color = grid[r][c]
            ok = True
            for i in range(5):
                for j in range(5):
                    if grid[r+i][c+j] != color:
                        ok = False
                        break
                if not ok:
                    break
            if ok:
                block_pos = (r, c)
                block_color = color
                break
        if block_pos:
            break
    
    # If no uniform block is found (should not happen for the test), fall back to a simple heuristic:
    if block_pos is None:
        # fallback: choose the central 5x5 sub‑grid (rows 5‑9, cols 5‑9) after a horizontal flip
        r, c = 5, 5   # 0‑based indices (i.e. rows 6‑10, cols 6‑10 in 1‑based)
    else:
        r, c = block_pos
    
    # 2. Determine the background colour
    bg = grid[r][c] if block_pos else None
    
    # 3. generate the three possible mirrored positions of a 5x5 block
    candidates = []
    if block_pos:
        r0, c0 = block_pos
        # horizontal mirror
        rh, ch = r0, 11 - c0          # 11 = 15 - 4
        # vertical mirror
        rv, cv = 11 - r0, c0
        # both (rotate 180)
        rb, cb = 11 - r0, 11 - c0
        
        for rr, cc in ((rh, ch), (rv, cv), (rb, cb)):
            # check whether this candidate contains the background colour
            has_bg = any(grid[rr+i][cc+j] == bg for i in range(5) for j in range(5))
            if not has_bg:
                candidates.append((rr, cc))
    
        # select the candidate with smallest row, then column
        candidates.sort(key=lambda x: (x[0], x[1]))
        r, c = candidates[0]
    
    # 4. extract the 5x5 sub‑grid and reverse each row (horizontal flip)
    out = []
    for i in range(5):
        row = grid[r+i][c:c+5]
        out.append(row[::-1])
    
    return out
