2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    # Find the border colour (non‑zero constant that fills whole border rows/columns)
    h = len(grid)
    w = len(grid[0])
    border_color = None
    for r in range(h):
        row = grid[r]
        if len(set(row)) == 1 and row[0] != 0:
            border_color = row[0]
            break
    if border_color is None:          # should not happen for valid inputs
        border_color = 0

    # Number of interior blocks (each block is 2x2)
    nbr = (h + 1) // 3          # block rows
    nbc = (w + 1) // 3          # block columns

    # Reconstruct the block grid (colour for each 2x2 interior block)
    block = [[0] * nbc for _ in range(nbr)]
    for br in range(nbr):
        for bc in range(nbc):
            val = grid[br * 3][bc * 3]            # top‑left cell of the block
            if val != 0 and val != border_color:
                block[br][bc] = val

    # Build colour → list of block positions
    from collections import defaultdict
    colour_pos = defaultdict(list)
    for br in range(nbr):
        for bc in range(nbc):
            c = block[br][bc]
            if c != 0:
                colour_pos[c].append((br, bc))

    # Determine the blocks that must be filled
    fill = set()
    for col, positions in colour_pos.items():
        # original blocks
        for br, bc in positions:
            fill.add((br, bc, col))

        # horizontal filling
        rows_by_br = defaultdict(list)
        for br, bc in positions:
            rows_by_br[br].append(bc)
        for br, bcs in rows_by_br.items():
            if len(set(bcs)) > 1:
                mn = min(bcs)
                mx = max(bcs)
                for bc in range(mn, mx + 1):
                    fill.add((br, bc, col))

        # vertical filling
        cols_by_bc = defaultdict(list)
        for br, bc in positions:
            cols_by_bc[bc].append(br)
        for bc, brs in cols_by_bc.items():
            if len(set(brs)) > 1:
                mn = min(brs)
                mx = max(brs)
                for br in range(mn, mx + 1):
                    fill.add((br, bc, col))

    # Apply the filling to the block grid
    for br, bc, col in fill:
        block[br][bc] = col

    # Reconstruct the full grid from the block grid
    out = [[0] * w for _ in range(h)]
    for r in range(h):
        for c in range(w):
            if r % 3 == 2 or c % 3 == 2:
                out[r][c] = border_color
            else:
                out[r][c] = block[r // 3][c // 3]

    return out
