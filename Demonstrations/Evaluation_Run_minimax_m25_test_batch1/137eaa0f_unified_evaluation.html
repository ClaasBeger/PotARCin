def solve(grid):
    from collections import deque

    n = len(grid)
    # find all connected components (clusters) of same non‑zero value
    visited = [[False]*n for _ in range(n)]
    components = []  # each component: {'val':int, 'cells':list of (r,c), 'shape':'H'/'V'/'P'}

    for r in range(n):
        for c in range(n):
            if grid[r][c] != 0 and not visited[r][c]:
                val = grid[r][c]
                q = deque([(r,c)])
                visited[r][c] = True
                cells = []
                while q:
                    cr, cc = q.popleft()
                    cells.append((cr, cc))
                    for dr, dc in ((1,0),(-1,0),(0,1),(0,-1)):
                        nr, nc = cr+dr, cc+dc
                        if 0 <= nr < n and 0 <= nc < n and not visited[nr][nc] and grid[nr][nc] == val:
                            visited[nr][nc] = True
                            q.append((nr,nc))
                # determine shape
                rows = {cell[0] for cell in cells}
                cols = {cell[1] for cell in cells}
                if len(cells) == 1:
                    shape = 'P'
                elif len(rows) == 1:          # same row -> horizontal
                    shape = 'H'
                elif len(cols) == 1:          # same column -> vertical
                    shape = 'V'
                else:
                    shape = 'P'
                components.append({'val': val, 'cells': cells, 'shape': shape})
