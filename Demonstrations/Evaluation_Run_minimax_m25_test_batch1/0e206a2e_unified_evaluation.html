2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    import collections

    rows = len(grid)
    cols = len(grid[0]) if rows else 0

    # helper to copy grid
    new_grid = [row[:] for row in grid]

    # find distinct non‑zero values
    vals = set()
    for r in range(rows):
        for c in range(cols):
            if grid[r][c]:
                vals.add(grid[r][c])
    if not vals:
        return new_grid

    a = min(vals)                     # smallest non‑zero (anchor)
    b = a + 1 if (a + 1) in vals else None   # second smallest, if present

    # ----- locate anchors -------------------------------------------------
    def rightmost(pos_list):
        # return (row,col) of the cell with maximal column, breaking ties by larger row
        return max(pos_list, key=lambda p: (p[1], p[0]))

    a_positions = [(r, c) for r in range(rows) for c in range(cols) if grid[r][c] == a]
    anchor_a = rightmost(a_positions)

    if b is not None:
        b_positions = [(r, c) for r in range(rows) for c in range(cols) if grid[r][c] == b]
        anchor_b = rightmost(b_positions)
    else:
        anchor_b = None

    # ----- clear all non‑anchor a / b cells ------------------------------
    for r in range(rows):
        for c in range(cols):
            if new_grid[r][c] == a and (r, c) != anchor_a:
                new_grid[r][c] = 0
            if b is not None and new_grid[r][c] == b and (r, c) != anchor_b:
                new_grid[r][c] = 0

    # ----- find largest component among cells that are not a or b -------
    def bfs(start_r, start_c, visited, mask):
        q = collections.deque()
        q.append((start_r, start_c))
        visited[start_r][start_c] = True
        component = []
        while q:
            r, c = q.popleft()
            if mask[r][c]:
                component.append((r, c, new_grid[r][c]))
            for dr, dc in ((1,0),(-1,0),(0,1),(0,-1)):
                nr, nc = r+dr, c+dc
                if 0 <= nr < rows and 0 <= nc < cols and not visited[nr][nc] and mask[nr][nc]:
                    visited[nr][nc] = True
                    q.append((nr, nc))
        return component

    # build a boolean mask for “not a and not b”
    mask = [[False]*cols for _ in range(rows)]
    for r in range(rows):
        for c in range(cols):
            if new_grid[r][c] not in (a, b):
                mask[r][c] = True

    visited = [[False]*cols for _ in range(rows)]
    best_comp = []
    best_size = 0
    for r in range(rows):
        for c in range(cols):
            if mask[r][c] and not visited[r][c]:
                comp = bfs(r, c, visited, mask)
                if len(comp) > best_size:
                    best_size = len(comp)
                    best_comp = comp

    # ----- remove the original component ---------------------------------
    for r, c, _ in best_comp:
        new_grid[r][c] = 0

    # ----- shape processing -----------------------------------------------
    if best_comp:
        # bounding box of the component
        min_r = min(r for r,_,_ in best_comp)
        max_r = max(r for r,_,_ in best_comp)
        min_c = min(c for _,c,_ in best_comp)
        max_c = max(c for _,c,_ in best_comp)
        h = max_r - min_r + 1
        w = max_c - min_c + 1

        # relative coordinates and values
        shape = []
        for r, c, val in best_comp:
            shape.append((r - min_r, c - min_c, val))

        # rotate 90° clockwise
        rotated = []
        for rr, cc, val in shape:
            nr = cc
            nc = h - 1 - rr
            rotated.append((nr, nc, val))

        # bounding box of rotated shape
        min_rr = min(r for r,_,_ in rotated)
        max_rr = max(r for r,_,_ in rotated)
        min_cc = min(c for _,c,_ in rotated)
        max_cc = max(c for _,c,_ in rotated)
        new_h = max_rr - min_rr + 1
        new_w = max_cc - min_cc + 1

        # normalize coordinates
        norm = []
        for rr, cc, val in rotated:
            norm.append((rr - min_rr, cc - min_cc, val))

        # ----- placement ----------------------------------------------------
        if b is not None:
            # we have both anchors (a and b)
            ar, ac = anchor_a
            br, bc = anchor_b
            if ar == br:
                # same row – place horizontally
                left = ac + 1
                top = ar
            else:
                # different rows – place vertically, anchored to a
                left = ac + 1
                top = ar
            # write the rotated shape into the grid
            for rr, cc, val in norm:
                nr = top + rr
                nc = left + cc
                if 0 <= nr < rows and 0 <= nc < cols:
                    new_grid[nr][nc] = val
        else:
            # only a’s – place two copies (leftmost and rightmost a)
            left_a, right_a = None, None
            if a_positions:
                left_a = min(a_positions, key=lambda p: (p[1], p[0]))
                right_a = max(a_positions, key=lambda p: (p[1], p[0]))

            # copy for leftmost a  (to its right, centred)
            if left_a:
                left = left_a[1] - new_w // 2
                top = left_a[0]
                for rr, cc, val in norm:
                    nr = top + rr
                    nc = left + cc
                    if 0 <= nr < rows and 0 <= nc < cols:
                        new_grid[nr][nc] = val

            # copy for rightmost a (to its left)
            if right_a:
                left = right_a[1] - new_w
                top = right_a[0] - new_h + 1
                for rr, cc, val in norm:
                    nr = top + rr
                    nc = left + cc
                    if 0 <= nr < rows and 0 <= nc < cols:
                        new_grid[nr][nc] = val

    # finally restore the anchor cells themselves
    new_grid[anchor_a[0]][anchor_a[1]] = a
    if b is not None:
        new_grid[anchor_b[0]][anchor_b[1]] = b

    return new_grid
