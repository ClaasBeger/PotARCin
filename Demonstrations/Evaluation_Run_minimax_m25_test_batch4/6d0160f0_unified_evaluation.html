2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    n = len(grid)
    # create output grid with zeros, then place 5's where they belong
    out = [[0] * n for _ in range(n)]
    for i in range(n):
        for j in range(n):
            if grid[i][j] == 5:
                out[i][j] = 5

    # locate the unique digit 4
    four_pos = None
    for i in range(n):
        for j in range(n):
            if grid[i][j] == 4:
                four_pos = (i, j)
                break
        if four_pos:
            break
    if four_pos is None:
        # no 4 – nothing to do
        return out

    row4, col4 = four_pos
    # block coordinates (0,1,2)
    rs = row4 // 4
    cs = col4 // 4
    # offset inside its block
    i4 = row4 % 4
    j4 = col4 % 4

    # target block according to the observed rule
    if cs == 0:
        rt, ct = 0, 2
    elif cs == 1:
        rt, ct = 1, 2
    else:  # cs == 2
        rt, ct = 1, 0

    # helper: retrieve corner values of a block
    def block_corners(br, bc):
        start_r = br * 4
        start_c = bc * 4
        corners = {}
        for di, dj in ((0, 0), (0, 2), (2, 0), (2, 2)):
            v = grid[start_r + di][start_c + dj]
            if v != 0:
                corners[(di, dj)] = v
        return corners

    # mapping dictionaries
    above_map = {(0, 0): (1, 0), (0, 2): (2, 2), (2, 0): (0, 1)}
    left_map = {(0, 0): (0, 1)}          # only needed case
    opposite_map = {(0, 0): (1, 2)}      # for cs == 2
    target_map = above_map.copy()
    target_map[(2, 2)] = (2, 1)          # extra mapping for the target block itself

    # collect numbers to place
    target_cells = {}
    # place the 4 itself
    target_cells[(i4, j4)] = 4

    # source block corners (except the 4)
    for (i, j), v in block_corners(rs, cs).items():
        if (i, j) == (i4, j4):
            continue
        if v and (i, j) not in target_cells:
            target_cells[(i, j)] = v

    # block directly above the source block (if not the central cross)
    if not (rs - 1 == 1 and cs == 1):
        above_br = rs - 1
        above_bc = cs
        for (i, j), v in block_corners(above_br, above_bc).items():
            mapped = above_map.get((i, j))
            if mapped and mapped not in target_cells and v not in target_cells.values():
                target_cells[mapped] = v

    # left neighbour (only needed when cs == 1)
    if cs == 1:
        left_br = rs
        left_bc = cs - 1
        for (i, j), v in block_corners(left_br, left_bc).items():
            mapped = left_map.get((i, j))
            if mapped and mapped not in target_cells and v not in target_cells.values():
                target_cells[mapped] = v

    # opposite block (only needed when cs == 2)
    if cs == 2:
        opp_br, opp_bc = 0, 0
        for (i, j), v in block_corners(opp_br, opp_bc).items():
            mapped = opposite_map.get((i, j))
            if mapped and mapped not in target_cells and v not in target_cells.values():
                target_cells[mapped] = v

    # if we still have fewer than 4 numbers, use the target block's own corners
    if len(target_cells) < 4:
        for (i, j), v in block_corners(rt, ct).items():
            mapped = target_map.get((i, j))
            if mapped and mapped not in target_cells and v not in target_cells.values():
                target_cells[mapped] = v

    # write the collected numbers into the target block
    for (i_off, j_off), val in target_cells.items():
        r = rt * 4 + i_off
        c = ct * 4 + j_off
        out[r][c] = val

    return out
