2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    # find the middle zero block (all-zero rows)
    n = len(grid)
    m = len(grid[0])
    m_start = None
    for i in range(n):
        if all(cell == 0 for cell in grid[i]):
            m_start = i
            break
    if m_start is None:
        # no zero block – return copy as is (should not happen)
        return [row[:] for row in grid]

    # find last consecutive all‑zero row
    m_end = m_start
    for i in range(m_start + 1, n):
        if all(cell == 0 for cell in grid[i]):
            m_end = i
        else:
            break

    # --- compute sets ---
    # zeros just above the zero block (row m_start-1)
    Z2 = {c for c in range(m) if grid[m_start - 1][c] == 0}
    # zeros two rows above (row m_start-2)
    Z1 = {c for c in range(m) if grid[m_start - 2][c] == 0} if m_start >= 2 else set()

    # any five in rows below the zero block
    any_five = set()
    for r in range(m_end + 1, n):
        for c in range(m):
            if grid[r][c] == 5:
                any_five.add(c)

    # --- process runs of five columns ---
    sorted_five = sorted(any_five)
    runs = []
    if sorted_five:
        cur = [sorted_five[0]]
        for x in sorted_five[1:]:
            if x == cur[-1] + 1:
                cur.append(x)
            else:
                runs.append(cur)
                cur = [x]
        runs.append(cur)

    processed_five = set()
    excluded_runs = []
    for run in runs:
        right = run[-1]
        if right == m - 1 or right in Z2:
            excluded_runs.append(run)
        else:
            processed_five.update(run[:-1])   # all except rightmost

    # adjust Z2 for excluded runs
    for run in excluded_runs:
        # if any column of the run also appears in Z1, delete whole run from Z2
        if set(run) & Z1:
            Z2 -= set(run)
        # also delete the column right+1 if it lies inside Z2
        if right + 1 < m and (right + 1) in Z2:
            Z2.remove(right + 1)

    # row 3 (first middle row) columns
    row3_cols = Z2 | processed_five

    # --- row 4 (second middle row) ---
    zero_cnt = [(c, (c in Z1) + (c in Z2)) for c in range(m)]
    max_cnt = max(count for _, count in zero_cnt)
    candidates = [c for c, cnt in zero_cnt if cnt == max_cnt]
    candidates.sort(reverse=True)
    # take exactly max_cnt columns (rightmost)
    row4_cols = set(candidates[:max_cnt])

    # --- build result grid ---
    res = [row[:] for row in grid]

    # top rows: turn 0 into 1
    for r in range(m_start):
        for c in range(m):
            if res[r][c] == 0:
                res[r][c] = 1

    # middle rows
    for c in range(m):
        if c in row3_cols:
            res[m_start][c] = 1
        else:
            res[m_start][c] = 0
    if m_start + 1 <= m_end:
        for c in range(m):
            if c in row4_cols:
                res[m_start + 1][c] = 1
            else:
                res[m_start + 1][c] = 0
    # any further middle rows stay zero (they are already zero)

    # bottom rows: turn 5 into 0
    for r in range(m_end + 1, n):
        for c in range(m):
            if res[r][c] == 5:
                res[r][c] = 0

    return res
