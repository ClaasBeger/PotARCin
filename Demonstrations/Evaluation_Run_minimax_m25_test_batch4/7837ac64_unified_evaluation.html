2. Definition

Find the common rule that maps an input grid to an output grid, given the examples below. Generate a python script that takes a grid as input and transforms it according to the this rule. Anticipate that this program will be applied to several grids, not only the shown test grid.

Your script must define exactly one callable entrypoint: def solve(grid): where `grid` is passed as a list of lists of integers (list[list[int]]) and the function returns the transformed grid in the same format.

def solve(grid):
    # -------------------------------------------------------------
    # Find the period of the pattern (the distance between the
    # grid lines).  The line colour is the colour that appears at
    # the last position of each period.
    # -------------------------------------------------------------
    rows = len(grid)
    cols = len(grid[0])

    # guess period from the first row – it is a simple periodic row
    def find_period(row):
        for p in range(2, len(row)):
            ok = True
            for i in range(len(row) - p):
                if row[i] != row[i + p]:
                    ok = False
                    break
            if ok:
                return p
        return 2                     # fallback (should never happen)

    p = find_period(grid[0])
    line_colour = grid[0][p - 1]          # colour of the vertical grid lines

    # -------------------------------------------------------------
    # Determine all line rows and line columns (they appear every p cells)
    # -------------------------------------------------------------
    line_rows = [r for r in range(rows) if r % p == p - 1]
    line_cols = [c for c in range(cols) if c % p == p - 1]

    # -------------------------------------------------------------
    # Keep only the interior lines (exclude the outermost border)
    # -------------------------------------------------------------
    interior_rows = line_rows[1:-1]
    interior_cols = line_cols[1:-1]

    # -------------------------------------------------------------
    # Find the bounding box of the intersection points that contain a
    # colour different from the line colour.
    # -------------------------------------------------------------
    def is_non_line(v):
        return v != line_colour

    row_min = row_max = None
    for i, r in enumerate(interior_rows):
        for c in interior_cols:
            if is_non_line(grid[r][c]):
                if row_min is None:
                    row_min = row_max = i
                else:
                    if i < row_min:
                        row_min = i
                    if i > row_max:
                        row_max = i
                break                # one non‑line cell is enough to mark the row

    col_min = col_max = None
    for j, c in enumerate(interior_cols):
        for r in interior_rows:
            if is_non_line(grid[r][c]):
                if col_min is None:
                    col_min = col_max = j
                else:
                    if j < col_min:
                        col_min = j
                    if j > col_max:
                        col_max = j
                break

    # If there is no non‑line colour at all, return a 3×3 zero grid
    if row_min is None or col_min is None:
        return [[0, 0, 0], [0, 0, 0], [0, 0, 0]]

    sub_rows = interior_rows[row_min:row_max + 1]
    sub_cols = interior_cols[col_min:col_max + 1]

    K = len(sub_rows)
    L = len(sub_cols)

    # -------------------------------------------------------------
    # Scale the K×L sub‑grid down to 3×3.
    # For each cell we collect all original cells that map to the same
    # output position and keep the most frequent non‑line colour.
    # -------------------------------------------------------------
    out = [[0, 0, 0] for _ in range(3)]
    counts = [[{} for _ in range(3)] for _ in range(3)]

    for ii in range(K):
        r = sub_rows[ii]
        out_i = ii * 3 // K          # floor division – same as int(ii*3/K)
        for jj in range(L):
            c = sub_cols[jj]
            out_j = jj * 3 // L
            val = grid[r][c]
            if is_non_line(val):
                d = counts[out_i][out_j]
                d[val] = d.get(val, 0) + 1

    for i in range(3):
        for j in range(3):
            if counts[i][j]:
                # choose the colour with highest frequency
                out[i][j] = max(counts[i][j], key=counts[i][j].get)
            else:
                out[i][j] = 0

    return out
